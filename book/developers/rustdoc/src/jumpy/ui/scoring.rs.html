42#[derive(Serialize, Deserialize)]
43struct ScoringMessage {
44    pub data: ScoringMessageEnum,
45    pub magic: u8,
46}

48#[derive(Serialize, Deserialize)]
49enum ScoringMessageEnum {
50    PlayerReady(u32),
51}
52
53impl From<ScoringMessageEnum> for ScoringMessage {
54    fn from(value: ScoringMessageEnum) -> Self {

56            data: value,
57            magic: SCORING_MESSAGE_MAGIC,
58        }
59    }

71fn scoring_menu_system(
72    meta: Root<GameMeta>,
73    mut sessions: ResMut<Sessions>,
74    ctx: Res<EguiCtx>,
75    mut state: ResMut<ScoringMenuState>,
76    controls: Res<GlobalPlayerControls>,
77    world: &World,
78) {
79    if !state.active {
80        // Make sure state is cleared if menu is closed
81        state.reset();
82        return;
83    }
84
85    let mut continue_game = false;
86    let mut game_won = false;
87    if let Some(session) = sessions.get_mut(SessionNames::GAME) {
88        let player_indices = session.world.components.get::<PlayerIdx>();
89        let player_indices_ref = player_indices.borrow();
90        let game_entities = session.world.get_resource::<Entities>().unwrap();
91        let match_inputs = session.world.get_resource::<MatchInputs>().unwrap();
92
93        #[cfg(not(target_arch = "wasm32"))]
94        let network_socket: Option<Socket> = session
95            .world
96            .get_resource::<SyncingInfo>()
97            .and_then(|x| x.socket().cloned());
98
99        // Build Vec<PlayerScoreInfo> sorted by player indices
100        let mut player_entities: Vec<(Entity, &PlayerIdx)> =
101            game_entities.iter_with(&player_indices_ref).collect();
102        player_entities.sort_by_key(|x| x.1 .0);
103        let player_score_info: Vec<PlayerScoreInfo> = player_entities
104            .iter()
105            .map(|x| PlayerScoreInfo {
106                entity: x.0,
107                player_idx: *x.1,
108                score: state.match_score.score(*x.1),
109            })
110            .collect();
111
112        // Contains any players who have broken win threshold. If multiple players have winning scores,
113        // contains player with highest score. If players are tied, tied players are all included.
114        let mut winning_players = Vec::<PlayerIdx>::default();
115
116        let win_threshold = meta.core.config.winning_score_threshold;
117        let mut highest_score = 0;
118
119        for score_info in player_score_info.iter() {
120            let score = score_info.score;
121            if score >= win_threshold && score >= highest_score {
122                if score > highest_score {
123                    // New winning player, clear prev winners
124                    winning_players.clear();
125                    highest_score = score;
126                }
127
128                winning_players.push(score_info.player_idx);
129            }
130        }
131        if winning_players.len() == 1 {
132            game_won = true;
133        }
134
135        // Check for inputs from local players toggling ready state
136        for (_, player_idx) in player_entities.iter() {
137            if let Some(source) = match_inputs.get_control_source(player_idx.0 as usize) {
138                if let Some(control) = controls.get(&source) {
139                    if control.menu_confirm_just_pressed
140                        && !state.ready_players.contains(*player_idx)
141                    {
142                        #[cfg(not(target_arch = "wasm32"))]
143                        {
144                            if let Some(socket) = network_socket.as_ref() {
145                                socket.send_reliable(
146                                    SocketTarget::All,
147                                    &postcard::to_allocvec(&ScoringMessage::from(
148                                        ScoringMessageEnum::PlayerReady(player_idx.0),
149                                    ))
150                                    .unwrap(),
151                                );
152
153                                debug!("Send message local player {} ready", player_idx.0);
154                            }
155                        }
156
157                        state.ready_players.insert(**player_idx);
158                    }
159                }
160            }
161        }
162
163        #[cfg(not(target_arch = "wasm32"))]
164        if let Some(socket) = network_socket.as_ref() {
165            handle_scoring_messages(socket, &mut state);
166        }
167
168        // Check if all non-ai players are ready
169        let mut all_players_ready = true;
170        for (_, player_idx) in player_entities.iter() {
171            let is_ai = match_inputs
172                .players
173                .get(player_idx.0 as usize)
174                .unwrap()
175                .is_ai;
176            if !is_ai && !state.ready_players.contains(*player_idx) {
177                all_players_ready = false;
178            }
179        }
180        if all_players_ready {
181            continue_game = true;
182        }
183
184        egui::CentralPanel::default()
185            .frame(egui::Frame::none())
186            .show(&ctx, |ui| {
187                let screen_rect = ui.max_rect();
188
189                // Determine x margin based on size of player panels, some extra spacing, and screen space available.
190                // This way window does not cover whole screen if < 4 players, and has consistent layout.
191                let player_panel_spacing = 20.0;
192                let max_width =
193                    (PLAYER_PANEL_WIDTH + player_panel_spacing) * player_score_info.len() as f32;
194                let x_margin = (screen_rect.width() - max_width) / 2.0;
195
196                // x margin is dynamic, y margin percentage of screen
197                let outer_margin =
198                    egui::style::Margin::symmetric(x_margin, screen_rect.height() * 0.2);
199
200                ui.vertical_centered(|ui| {
201                    BorderedFrame::new(&meta.theme.panel.border)
202                        .margin(outer_margin)
203                        .padding(meta.theme.panel.padding)
204                        .show(ui, |ui| {
205                            world.run_system(
206                                scoring_menu,
207                                (
208                                    ui,
209                                    &player_score_info,
210                                    &*match_inputs,
211                                    &state,
212                                    &winning_players,
213                                ),
214                            );
215                        });
216                });
217            });
218    } else {
219        error!("Scoring menu failed to load score from existing game session. Closing scoring UI.");
220        state.reset();
221    }
222
223    if continue_game {
224        state.reset();
225        let next_maps = state.next_maps.clone();
226        let reset_score = game_won;
227        sessions.add_command(Box::new(move |sessions: &mut Sessions| {
228            sessions.restart_game(next_maps, reset_score);
229        }));
230    }
231}

233fn scoring_menu(
234    mut param: In<(
235        &mut egui::Ui,
236        &Vec<PlayerScoreInfo>,
237        &MatchInputs,
238        &ScoringMenuState,
239        &Vec<PlayerIdx>,
240    )>,
241    meta: Root<GameMeta>,
242    localization: Localization<GameMeta>,
243    world: &World,
244) {
245    let (ui, player_score_info, match_inputs, menu_state, winning_players) = &mut *param;
246
247    // Scoring heading label
248    ui.vertical_centered(|ui| {
249        let (text, color) = match winning_players.len() {
250            0 => ("intermission", meta.theme.panel.font_color),
251            1 => ("match-complete", meta.theme.colors.positive),
252            _ => ("tied-for-win", meta.theme.panel.font_color),
253        };
254
255        ui.label(
256            meta.theme
257                .font_styles
258                .heading
259                .rich(localization.get(text))
260                .color(color),
261        );
262    });
263
264    ui.vertical_centered(|ui| {
265        ui.horizontal_centered(|ui| {
266            let player_count = player_score_info.len();
267            let available_spacing = ui.available_width() - PLAYER_PANEL_WIDTH * player_count as f32;
268
269            // Compute how much space to use in gaps between panels
270            let spacing = available_spacing / (player_count + 1) as f32;
271            ui.add_space(spacing);
272            for (i, _) in player_score_info.iter().enumerate() {
273                let player_input = match_inputs.players.get(i).unwrap();
274                let player_score_info = player_score_info.get(i).unwrap();
275                let player_idx = player_score_info.player_idx;
276
277                let ready = menu_state.ready_players.contains(&player_idx);
278                let player_won =
279                    if winning_players.contains(&player_idx) && winning_players.len() == 1 {
280                        Some(PlayerWon::SoleWinner)
281                    } else if winning_players.contains(&player_idx) {
282                        Some(PlayerWon::Tied)
283                    } else {
284                        None
285                    };
286
287                world.run_system(
288                    player_score_panel,
289                    (ui, player_input, player_score_info, ready, player_won),
290                );
291                ui.add_space(spacing);
292            }
293        });
294    });
295
296    let match_complete = winning_players.len() == 1;
297    ui.horizontal(|ui| {
298        if match_complete {
299            ui.label(
300                meta.theme
301                    .font_styles
302                    .normal
303                    .rich(localization.get("press-confirm-to-play-again"))
304                    .color(meta.theme.panel.font_color),
305            );
306        } else {
307            ui.label(
308                meta.theme
309                    .font_styles
310                    .normal
311                    .rich(localization.get("press-confirm-to-ready-up"))
312                    .color(meta.theme.panel.font_color),
313            );
314        }
315    });
316}

318fn player_score_panel(
319    mut params: In<(
320        &mut egui::Ui,
321        &PlayerInput,
322        &PlayerScoreInfo,
323        bool,
324        Option<PlayerWon>,
325    )>,
326    meta: Root<GameMeta>,
327    assets: Res<AssetServer>,
328    localization: Localization<GameMeta>,
329    world: &World,
330) {
331    let (ui, player_input, player_score_info, ready, won) = &mut *params;
332    let panel = &meta.theme.panel;
333
334    BorderedFrame::new(&panel.border)
335        .padding(panel.padding)
336        .show(ui, |ui| {
337            // PLAYER_PANEL_WIDTH is total space for entire bordered frame, remove space lost to padding
338            // and use this for inner contents.
339            ui.set_width(PLAYER_PANEL_WIDTH - panel.padding.left - panel.padding.right);
340
341            ui.vertical_centered(|ui| {
342                ui.label(
343                    meta.theme
344                        .font_styles
345                        .bigger
346                        .rich(format!(
347                            "{} {}",
348                            localization.get("player"),
349                            player_score_info.player_idx.0
350                        ))
351                        .color(meta.theme.panel.font_color),
352                );
353
354                match won {
355                    Some(won) => {
356                        let text = match won {
357                            PlayerWon::Tied => localization.get("tied"),
358                            PlayerWon::SoleWinner => localization.get("won"),
359                        };
360
361                        ui.label(
362                            meta.theme
363                                .font_styles
364                                .bigger
365                                .rich(text)
366                                .color(meta.theme.colors.positive),
367                        );
368                    }
369                    None => {
370                        ui.add_space(meta.theme.font_styles.bigger.size);
371                    }
372                }
373
374                let player_meta = assets.get(player_input.selected_player);
375                let hat_meta = player_input.selected_hat.map(|x| assets.get(x));
376                world.run_system(player_image, (ui, &player_meta, hat_meta.as_deref()));
377
378                ui.label(
379                    meta.theme
380                        .font_styles
381                        .bigger
382                        .rich(format!(
383                            "{}: {}",
384                            localization.get("score"),
385                            player_score_info.score,
386                        ))
387                        .color(meta.theme.panel.font_color),
388                );
389
390                if !player_input.is_ai {
391                    let (ready_str, color) = match *ready {
392                        true => ("ready", meta.theme.colors.positive),
393                        false => ("not-ready", meta.theme.colors.negative),
394                    };
395                    let ready_localized = localization.get(ready_str);
396                    ui.label(
397                        meta.theme
398                            .font_styles
399                            .normal
400                            .rich(ready_localized)
401                            .color(color),
402                    );
403                } else {
404                    ui.label(
405                        meta.theme
406                            .font_styles
407                            .normal
408                            .rich(localization.get("ai"))
409                            .color(meta.theme.panel.font_color),
410                    );
411                }
412            });
413        });
414}

416#[cfg(not(target_arch = "wasm32"))]
417fn handle_scoring_messages(
418    network_socket: &(impl NetworkSocket + ?Sized),
419    state: &mut ScoringMenuState,
420) {
421    // TODO handle disconnects
422    let datas: Vec<(u32, Vec<u8>)> = network_socket.recv_reliable();
423    let local_player_idx = network_socket.player_idx();
424    for (_, data) in datas {
425        match postcard::from_bytes::<ScoringMessage>(&data) {
426            Ok(message) => match message.data {
427                ScoringMessageEnum::PlayerReady(player) => {
428                    if message.magic == SCORING_MESSAGE_MAGIC && player != local_player_idx {
429                        state.ready_players.insert(PlayerIdx(player));
430                        debug!("Received message player {} ready", player);
431                    }
432                }
433            },
434            Err(e) => warn!("Ignoring network message that was not understood: {e}"),
435        }
436    }
