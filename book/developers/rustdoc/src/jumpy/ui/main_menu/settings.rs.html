11#[derive(Clone, Default)]
12struct SettingsState {
13    tab: SettingsTab,
14    modified_settings: Settings,
15    settings_loaded: bool,
16    currently_binding_input_idx: Option<usize>,
17}
18
19/// Which settings tab we are on
20#[derive(Clone, Copy, PartialEq, Eq, Default)]
21enum SettingsTab {
22    #[default]
23    Controls,
24    Networking,
25    Audio,
26    Graphics,
27}
28
29impl SettingsTab {
30    const TABS: &'static [(Self, &'static str)] = &[
31        (Self::Controls, "controls"),
32        (Self::Networking, "networking"),
33        (Self::Audio, "audio"),
34        (Self::Graphics, "graphics"),
35    ];
36}
37
38pub fn widget(
39    mut ui: In<(&mut egui::Ui, &mut bool)>,
40    meta: Root<GameMeta>,
41    localization: Localization<GameMeta>,
42    input: Res<GlobalPlayerControls>,
43    mut mapping: ResMut<PlayerControlMapping>,
44    mut storage: ResMut<Storage>,
45    world: &World,
46) {
47    let (ui, close_settings_menu) = &mut *ui;
48    let mut state = ui.ctx().get_state::<SettingsState>();
49    let screen_rect = ui.max_rect();
50
51    if !state.settings_loaded {
52        state.modified_settings = storage.get::<Settings>().unwrap().clone();
53        state.settings_loaded = true;
54    }
55
56    let mut should_reset = false;
57
58    // Calculate a margin
59    let outer_margin = screen_rect.size() * 0.10;
60    let outer_margin = egui::Margin {
61        left: outer_margin.x,
62        right: outer_margin.x,
63        // Make top and bottom margins smaller
64        top: outer_margin.y / 1.5,
65        bottom: outer_margin.y / 1.5,
66    };
67
68    let heading_font = meta
69        .theme
70        .font_styles
71        .heading
72        .with_color(meta.theme.panel.font_color);
73
74    BorderedFrame::new(&meta.theme.panel.border)
75        .margin(outer_margin)
76        .padding(meta.theme.panel.padding)
77        .show(ui, |ui| {
78            // Disable all the buttons if we are currently binding an input
79            ui.set_enabled(state.currently_binding_input_idx.is_none());
80
81            ui.vertical_centered(|ui| {
82                // Settings Heading
83                ui.label(heading_font.rich(localization.get("settings")));
84
85                // Add tab list at the top of the panel
86                ui.horizontal(|ui| {
87                    for (i, (tab, name)) in SettingsTab::TABS.iter().enumerate() {
88                        let name = localization.get(name);
89                        let mut name = egui::RichText::new(name);
90
91                        // Underline the current tab
92                        if *tab == state.tab {
93                            name = name.underline();
94                        }
95
96                        let mut button =
97                            BorderedButton::themed(&meta.theme.buttons.normal, name).show(ui);
98
99                        // Focus the first tab by default
100                        if i == 0 {
101                            button = button.focus_by_default(ui);
102                        }
103
104                        // Change tab when clicked
105                        if button.clicked() {
106                            state.tab = *tab;
107                        }
108                    }
109                });
110
111                // Add buttons to the bottom
112                ui.with_layout(egui::Layout::bottom_up(egui::Align::Center), |ui| {
113                    let buttons_response = ui.horizontal(|ui| {
114                        // Calculate button size and spacing
115                        let width = ui.available_width();
116                        let button_width = width / 4.0;
117                        let button_min_size = vec2(button_width, 0.0);
118                        let button_spacing = (width - 3.0 * button_width) / 4.0;
119
120                        ui.add_space(button_spacing);
121
122                        // Cancel button
123                        if (BorderedButton::themed(
124                            &meta.theme.buttons.normal,
125                            localization.get("cancel"),
126                        )
127                        .min_size(button_min_size)
128                        .show(ui)
129                        .clicked()
130                            || input.values().any(|x| x.menu_back_just_pressed))
131                            && state.currently_binding_input_idx.is_none()
132                        {
133                            **close_settings_menu = true;
134                            // Reset the modified settings to their stored settings.
135                            state.modified_settings = storage.get::<Settings>().unwrap().clone();
136                            // Run cancel callbacks
137                            if state.tab == SettingsTab::Audio {
138                                world.run_system(audio::on_cancel, &state);
139                            }
140                        }
141
142                        ui.add_space(button_spacing);
143
144                        // Reset button
145                        if BorderedButton::themed(
146                            &meta.theme.buttons.normal,
147                            localization.get("reset"),
148                        )
149                        .min_size(button_min_size)
150                        .show(ui)
151                        .clicked()
152                        {
153                            should_reset = true;
154                        }
155
156                        ui.add_space(button_spacing);
157
158                        // Save button
159                        if BorderedButton::themed(
160                            &meta.theme.buttons.normal,
161                            localization.get("save"),
162                        )
163                        .min_size(button_min_size)
164                        .show(ui)
165                        .clicked()
166                        {
167                            // Save the settings to disk.
168                            *mapping = state.modified_settings.player_controls.clone();
169                            storage.insert(state.modified_settings.clone());
170                            storage.save();
171                            **close_settings_menu = true;
172                        }
173                    });
174
175                    let buttons_rect = buttons_response.response.rect;
176                    ui.add_space(buttons_rect.height() / 2.0);
177
178                    ui.vertical(|ui| match state.tab {
179                        SettingsTab::Controls => {
180                            world.run_system(controls::widget, (ui, &mut state, should_reset))
181                        }
182                        SettingsTab::Networking => {
183                            world.run_system(networking::widget, (ui, &mut state, should_reset))
184                        }
185                        SettingsTab::Audio => {
186                            world.run_system(audio::widget, (ui, &mut state, should_reset))
187                        }
188                        SettingsTab::Graphics => {
189                            world.run_system(graphics::widget, (ui, &mut state, should_reset))
190                        }
191                    });
192                });
193            });
194        });
195
196    // Update the state with our modified one
197    ui.ctx().set_state(state);
198}
