15/// Network message that may be sent when selecting a map.
16#[derive(Serialize, Deserialize)]
17pub enum MapSelectMessage {
18    SelectMap(MapPoolNetwork),
19}
20
21pub fn widget(
22    ui: In<&mut egui::Ui>,
23    world: &World,
24    meta: Root<GameMeta>,
25    mut sessions: ResMut<Sessions>,
26    mut session_options: ResMut<SessionOptions>,
27    assets: Res<AssetServer>,
28
29    #[cfg(not(target_arch = "wasm32"))] network_socket: Option<Res<NetworkMatchSocket>>,
30) {
31    let mut select_action = MapSelectAction::None;
32
33    // Get map select action from network
34    #[cfg(not(target_arch = "wasm32"))]
35    if let Some(MapSelectAction::SelectMap(map_meta)) =
36        handle_match_setup_messages(&network_socket, &assets)
37    {
38        select_action = MapSelectAction::SelectMap(map_meta);
39    }
40
41    // If the `TEST_MAP` debug env var is present start the game with the map
42    // matching the provided name.
43    #[cfg(debug_assertions)]
44    'test: {
45        use std::env::{var, VarError};
46        use std::sync::atomic::{AtomicBool, Ordering};
47        static DEBUG_DID_CHECK_ENV_VARS: AtomicBool = AtomicBool::new(false);
48        if DEBUG_DID_CHECK_ENV_VARS
49            .compare_exchange(false, true, Ordering::Acquire, Ordering::Relaxed)
50            .is_ok()
51        {
52            let test_map = match var("TEST_MAP") {
53                Ok(name) => name,
54                Err(VarError::NotPresent) => break 'test,
55                Err(VarError::NotUnicode(err)) => {
56                    warn!("Invalid TEST_MAP, not unicode: {err:?}");
57                    break 'test;
58                }
59            };
60
61            let asset_server = world.resource::<AssetServer>();
62            let game_meta = asset_server.root::<GameMeta>();
63
64            let get_map_handles = || {
65                let mut map_handles = Vec::new();
66                map_handles.extend(game_meta.core.stable_maps.iter().copied());
67                for pack in asset_server.packs() {
68                    let pack_meta = asset_server.get(pack.root.typed::<crate::PackMeta>());
69                    map_handles.extend(pack_meta.maps.iter().copied());
70                }
71                map_handles
72            };
73
74            let Some(test_map) = get_map_handles()
75                .into_iter()
76                .find(|h| asset_server.get(*h).name == test_map)
77            else {
78                warn!("TEST_MAP not found: {test_map}");
79                let available_names = super::handle_names_to_string(get_map_handles(), |h| {
80                    asset_server.get(h).name.as_str()
81                });
82                warn!("Available map names: {available_names}");
83                break 'test;
84            };
85
86            select_action = MapSelectAction::SelectMap(MapPool::from_single_map(test_map));
87        }
88    }
89
90    // If no network action - update action from UI
91    if matches!(select_action, MapSelectAction::None) {
92        select_action = world.run_system(map_select_menu, ());
93
94        #[cfg(not(target_arch = "wasm32"))]
95        // Replicate local action
96        replicate_map_select_action(&select_action, &network_socket, &assets);
97    }
98
99    match select_action {
100        MapSelectAction::None => (),
101        MapSelectAction::SelectMap(maps) => {
102            session_options.delete = true;
103            ui.ctx().set_state(MenuPage::Home);
104
105            #[cfg(not(target_arch = "wasm32"))]
106            let session_runner: Box<dyn SessionRunner> = match network_socket {
107                Some(socket) => {
108                    let random_seed = ui.ctx().get_state::<NetworkGameState>().random_seed();
109
110                    Box::new(GgrsSessionRunner::<NetworkInputConfig>::new(
111                        Some(FPS),
112                        GgrsSessionRunnerInfo::new(
113                            socket.ggrs_socket(),
114                            Some(meta.network.max_prediction_window),
115                            Some(meta.network.local_input_delay),
116                            random_seed,
117                        ),
118                    ))
119                }
120                None => Box::<JumpyDefaultMatchRunner>::default(),
121            };
122            #[cfg(target_arch = "wasm32")]
123            let session_runner = Box::<JumpyDefaultMatchRunner>::default();
124
125            let player_select_state = ui.ctx().get_state::<PlayerSelectState>();
126            sessions.start_game(MatchPlugin {
127                maps,
128                player_info: std::array::from_fn(|i| {
129                    let slot = player_select_state.slots[i];
130
131                    PlayerInput {
132                        active: !slot.is_empty(),
133                        selected_player: slot
134                            .selected_player()
135                            .unwrap_or(player_select_state.players[0]),
136                        selected_hat: slot.selected_hat(),
137                        control_source: slot.user_control_source(),
138                        editor_input: default(),
139                        control: default(),
140                        is_ai: slot.is_ai(),
141                    }
142                }),
143                plugins: meta.get_plugins(&assets),
144                session_runner,
145                score: default(),
146            });
147            ui.ctx().set_state(PlayerSelectState::default());
148        }
149        MapSelectAction::GoBack => ui.ctx().set_state(MenuPage::PlayerSelect),
150    }
151}

153/// Send a MapSelectMessage over network if local player has selected a map.
154#[cfg(not(target_arch = "wasm32"))]
155fn replicate_map_select_action(
156    action: &MapSelectAction,
157    socket: &Option<Res<NetworkMatchSocket>>,
158    asset_server: &AssetServer,
159) {
160    use bones_framework::networking::SocketTarget;
161    if let Some(socket) = socket {
162        if let MapSelectAction::SelectMap(maps) = action {
163            info!("Sending network SelectMap message.");
164            socket.send_reliable(
165                SocketTarget::All,
166                &postcard::to_allocvec(&MapSelectMessage::SelectMap(
167                    maps.into_network(asset_server),
168                ))
169                .unwrap(),
170            );
171        }
172    }
173}

175#[cfg(not(target_arch = "wasm32"))]
176fn handle_match_setup_messages(
177    socket: &Option<Res<NetworkMatchSocket>>,
178    asset_server: &AssetServer,
179) -> Option<MapSelectAction> {
180    if let Some(socket) = socket {
181        let datas: Vec<(u32, Vec<u8>)> = socket.recv_reliable();
182
183        for (_player, data) in datas {
184            match postcard::from_bytes::<MapSelectMessage>(&data) {
185                Ok(message) => match message {
186                    MapSelectMessage::SelectMap(maps) => {
187                        info!("Map select message received, starting game");
188
189                        return Some(MapSelectAction::SelectMap(MapPool::from_network(
190                            maps,
191                            asset_server,
192                        )));
193                    }
194                },
195                Err(e) => {
196                    // TODO: The second player in an online match is having this triggered by
197                    // picking up a `ConfirmSelection` message, that might have been sent to
198                    // _itself_.
199                    warn!("Ignoring network message that was not understood: {e} data: {data:?}");
200                }
201            }
202        }
203    }
204
205    None
