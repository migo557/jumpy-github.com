5trait SettingsExts {
6    fn volume_change_event(&self) -> AudioEvent;
7}
8
9impl SettingsExts for Settings {
10    fn volume_change_event(&self) -> AudioEvent {
11        AudioEvent::VolumeScaleUpdate {
12            main_volume_scale: self.main_volume,
13            music_volume_scale: self.music_volume,
14            effects_volume_scale: self.effects_volume,
15        }
16    }
17}
18
19pub(super) fn on_cancel(In(state): In<&SettingsState>, mut audio_center: ResMut<AudioCenter>) {
20    audio_center.push_event(state.modified_settings.volume_change_event());
21}

23pub(super) fn widget(
24    In((ui, state, should_reset)): In<(&mut egui::Ui, &mut SettingsState, bool)>,
25    meta: Root<GameMeta>,
26    localization: Localization<GameMeta>,
27    mut audio_center: ResMut<AudioCenter>,
28) {
29    let bigger_font = meta
30        .theme
31        .font_styles
32        .bigger
33        .with_color(meta.theme.panel.font_color);
34    let normal_font = meta
35        .theme
36        .font_styles
37        .normal
38        .with_color(meta.theme.panel.font_color);
39
40    if should_reset {
41        state.modified_settings.main_volume = meta.default_settings.main_volume;
42        audio_center.push_event(state.modified_settings.volume_change_event());
43    }
44
45    ui.add_space(normal_font.size);
46
47    ui.scope(|ui| {
48        let style = ui.style_mut();
49        style.drag_value_text_style = egui::TextStyle::Monospace;
50        style.spacing.indent = normal_font.size / 2.5;
51        style.spacing.button_padding = egui::vec2(6.0, 3.0);
52        style.spacing.slider_width = 200.0;
53        style.visuals.indent_has_left_vline = false;
54
55        ui.indent("audio-controls", |ui| {
56            ui.label(bigger_font.rich(localization.get("volume")));
57            ui.add_space(normal_font.size / 2.0);
58
59            egui::Grid::new("volume-grid").show(ui, |ui| {
60                // Main
61                let main_changed = volume_control_widget(
62                    ui,
63                    normal_font.rich(localization.get("volume-main")),
64                    &mut state.modified_settings.main_volume,
65                )
66                .changed();
67                ui.end_row();
68
69                // Music
70                let music_changed = volume_control_widget(
71                    ui,
72                    normal_font.rich(localization.get("volume-music")),
73                    &mut state.modified_settings.music_volume,
74                )
75                .changed();
76                ui.end_row();
77
78                // Effects
79                let effects_changed = volume_control_widget(
80                    ui,
81                    normal_font.rich(localization.get("volume-effects")),
82                    &mut state.modified_settings.effects_volume,
83                )
84                .changed();
85                ui.end_row();
86
87                if main_changed || music_changed || effects_changed {
88                    audio_center.push_event(state.modified_settings.volume_change_event());
89                }
90            });
91        });
92    });
93}
94
95fn volume_control_widget(
96    ui: &mut egui::Ui,
97    label: impl Into<egui::WidgetText>,
98    value: &mut f32,
99) -> egui::Response {
100    ui.label(label);
101    let slider = egui::Slider::new(value, 0.0..=1.0)
102        .smallest_positive(0.0)
103        .step_by(0.01)
104        .fixed_decimals(3)
105        .custom_formatter(|x, _| format!("{:5.1}", x * 100.0))
106        .custom_parser(|s| s.parse::<f64>().ok().map(|f| f / 100.0));
107    ui.add(slider)
108}
