333/// The kind of input binding to listen for.
334enum BindingKind {
335    Keyboard,
336    Gamepad,
337}
338
339fn get_input(
340    kind: BindingKind,
341    keyboard: &KeyboardInputs,
342    gamepad: &GamepadInputs,
343) -> Option<InputKind> {
344    match kind {
345        BindingKind::Keyboard => keyboard.key_events.iter().next().and_then(|event| {
346            event
347                .button_state
348                .pressed()
349                .then(|| event.key_code.option().map(InputKind::Keyboard))
350                .flatten()
351        }),
352        BindingKind::Gamepad => {
353            gamepad
354                .gamepad_events
355                .iter()
356                .next()
357                .and_then(|event| match event {
358                    GamepadEvent::Button(e) => {
359                        (e.value.abs() > 0.1).then_some(InputKind::Button(e.button))
360                    }
361                    GamepadEvent::Axis(e) => {
362                        if e.value > 0.1 {
363                            Some(InputKind::AxisPositive(e.axis))
364                        } else if e.value < -0.1 {
365                            Some(InputKind::AxisNegative(e.axis))
366                        } else {
367                            None
368                        }
369                    }
370                    _ => None,
371                })
372        }
373    }
