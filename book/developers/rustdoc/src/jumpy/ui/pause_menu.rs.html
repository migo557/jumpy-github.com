24fn pause_menu_system(
25    meta: Root<GameMeta>,
26    mut sessions: ResMut<Sessions>,
27    ctx: Res<EguiCtx>,
28    controls: Res<GlobalPlayerControls>,
29    world: &World,
30    assets: Res<AssetServer>,
31    mut pause_menu: ResMutInit<PauseMenu>,
32) {
33    let mut back_to_menu = false;
34    let mut restart_game = false;
35    let mut close_pause_menu = false;
36    let mut close_settings_menu = false;
37    let mut select_map = None;
38    if let Some(session) = sessions.get_mut(SessionNames::GAME) {
39        let pause_pressed = controls.values().any(|x| x.pause_just_pressed);
40
41        #[cfg(not(target_arch = "wasm32"))]
42        let is_online = session
43            .world
44            .get_resource::<SyncingInfo>()
45            .is_some_and(|x| x.is_online());
46
47        #[cfg(target_arch = "wasm32")]
48        let is_online = false;
49
50        if pause_menu.menu_open {
51            let page = ctx.get_state::<PauseMenuPage>();
52
53            match page {
54                PauseMenuPage::Pause => {
55                    egui::CentralPanel::default()
56                        .frame(egui::Frame::none())
57                        .show(&ctx, |ui| {
58                            let screen_rect = ui.max_rect();
59
60                            let pause_menu_width = meta.main_menu.menu_width;
61                            let x_margin = (screen_rect.width() - pause_menu_width) / 2.0;
62                            let outer_margin = egui::style::Margin::symmetric(
63                                x_margin,
64                                screen_rect.height() * 0.2,
65                            );
66
67                            BorderedFrame::new(&meta.theme.panel.border)
68                                .margin(outer_margin)
69                                .padding(meta.theme.panel.padding)
70                                .show(ui, |ui| {
71                                    ui.set_min_width(ui.available_width());
72
73                                    world.run_system(
74                                        main_pause_menu,
75                                        (
76                                            ui,
77                                            session,
78                                            &mut restart_game,
79                                            &mut back_to_menu,
80                                            &mut close_pause_menu,
81                                            is_online,
82                                        ),
83                                    );
84                                });
85                        });
86                }
87                PauseMenuPage::MapSelect => {
88                    let action = world.run_system(crate::ui::map_select::map_select_menu, ());
89
90                    match action {
91                        super::map_select::MapSelectAction::None => (),
92                        super::map_select::MapSelectAction::SelectMap(map) => {
93                            select_map = Some(map);
94                            ctx.set_state(PauseMenuPage::Pause);
95                        }
96                        super::map_select::MapSelectAction::GoBack => {
97                            ctx.set_state(PauseMenuPage::Pause);
98                        }
99                    }
100                }
101                PauseMenuPage::Settings => {
102                    egui::CentralPanel::default()
103                        .frame(egui::Frame::none())
104                        .show(&ctx, |ui| {
105                            world.run_system(
106                                super::main_menu::settings::widget,
107                                (ui, &mut close_settings_menu),
108                            );
109                        });
110                }
111            }
112
113            // When pause menu is open (and not about to be closed), we -re-pause every frame regardless of if just pressed
114            // because in online, if menu is open during a map transition, we need to make sure new net session
115            // disables input to re-pause itself.
116            if !close_pause_menu {
117                pause_session(true, is_online, session, false);
118            }
119        } else if pause_pressed {
120            pause_menu.menu_open = true;
121        }
122    }
123
124    if back_to_menu {
125        sessions.end_game();
126        sessions.start_menu();
127        pause_menu.menu_open = false;
128    } else if restart_game {
129        sessions.restart_game(None, false);
130        pause_menu.menu_open = false;
131    } else if let Some(maps) = select_map {
132        let match_info = sessions
133            .get(SessionNames::GAME)
134            .unwrap()
135            .world
136            .resource::<MatchInputs>()
137            .deref()
138            .clone();
139        sessions.end_game();
140        sessions.create_with(SessionNames::GAME, |builder: &mut SessionBuilder| {
141            builder.install_plugin(crate::core::MatchPlugin {
142                maps,
143                player_info: std::array::from_fn(|i| PlayerInput {
144                    control: default(),
145                    editor_input: default(),
146                    ..match_info.players[i]
147                }),
148                plugins: meta.get_plugins(&assets),
149                session_runner: Box::<JumpyDefaultMatchRunner>::default(),
150                score: default(),
151            });
152        });
153        pause_menu.menu_open = false;
154    }
155
156    if close_pause_menu {
157        pause_menu.menu_open = false;
158    }
159
160    if close_settings_menu {
161        ctx.set_state(PauseMenuPage::Pause);
162    }
163}

165fn main_pause_menu(
166    mut param: In<(
167        &mut egui::Ui,
168        &mut Session,
169        &mut bool,
170        &mut bool,
171        &mut bool,
172        bool,
173    )>,
174    meta: Root<GameMeta>,
175    localization: Localization<GameMeta>,
176    controls: Res<GlobalPlayerControls>,
177    scoring_menu: Res<ScoringMenuState>,
178) {
179    let (ui, session, restart_game, back_to_menu, close_pause_menu, is_online) = &mut *param;
180
181    // Unpause the game
182    if controls.values().any(|x| x.pause_just_pressed) {
183        // Do not unpause game session if scoring menu open.
184        // TODO: Use some kind of pause stack to track what different systems
185        // might want session to remain inactive.
186        pause_session(false, *is_online, session, scoring_menu.active);
187        **close_pause_menu = true;
188    }
189
190    ui.vertical_centered(|ui| {
191        let width = ui.available_width();
192
193        // Heading
194        ui.label(
195            meta.theme
196                .font_styles
197                .heading
198                .rich(localization.get("paused"))
199                .color(meta.theme.panel.font_color),
200        );
201
202        // Map title
203        if let Some(map_meta) = session.world.get_resource::<SpawnedMapMeta>() {
204            ui.label(
205                meta.theme
206                    .font_styles
207                    .bigger
208                    .rich(map_meta.name.to_string()),
209            );
210        }
211
212        ui.add_space(10.0);
213
214        // Continue button
215        if BorderedButton::themed(&meta.theme.buttons.normal, localization.get("continue"))
216            .min_size(vec2(width, 0.0))
217            .show(ui)
218            .focus_by_default(ui)
219            .clicked()
220        {
221            pause_session(false, *is_online, session, false);
222            **close_pause_menu = true;
223        }
224
225        // Local game buttons
226        ui.scope(|ui| {
227            // Map select button
228            if BorderedButton::themed(
229                &meta.theme.buttons.normal,
230                localization.get("map-select-title"),
231            )
232            .min_size(vec2(width, 0.0))
233            .show(ui)
234            .clicked()
235            {
236                ui.ctx().set_state(PauseMenuPage::MapSelect);
237            }
238
239            // Settings button
240            if BorderedButton::themed(&meta.theme.buttons.normal, localization.get("settings"))
241                .min_size(vec2(width, 0.0))
242                .show(ui)
243                .clicked()
244            {
245                ui.ctx().set_state(PauseMenuPage::Settings);
246            }
247
248            // Restart button
249            if BorderedButton::themed(&meta.theme.buttons.normal, localization.get("restart"))
250                .min_size(vec2(width, 0.0))
251                .show(ui)
252                .clicked()
253            {
254                **restart_game = true;
255            }
256        });
257
258        // Re-add edit button once map editor is back in game.
259        //
260        // Edit button
261        // ui.scope(|ui| {
262        //     if BorderedButton::themed(&meta.theme.buttons.normal, localization.get("edit"))
263        //         .min_size(vec2(width, 0.0))
264        //         .show(ui)
265        //         .clicked()
266        //     {
267        //         // TODO: show editor.
268        //         pause_session(false, *is_online, session, false);
269        //         **close_pause_menu = true;
270        //     }
271        // });
272
273        // Main menu button
274        if BorderedButton::themed(&meta.theme.buttons.normal, localization.get("main-menu"))
275            .min_size(vec2(width, 0.0))
276            .show(ui)
277            .clicked()
278        {
279            **back_to_menu = true;
280        }
281    });
282}

284/// Helper for pausing session depending on if online or offline match.
285/// `remain_inactive` is used if other systems like scoring have set session inactive and wish to
286/// prevent that change.
287fn pause_session(paused: bool, is_online: bool, session: &mut Session, remain_inactive: bool) {
288    if is_online {
289        // Online session remains active so that it will not timeout with other players.
290        // Instead we disable the input so it is not captured by game while paused.
291        session.runner.disable_local_input(paused);
292    } else if !paused && !remain_inactive {
293        session.active = true;
294    } else if paused {
295        session.active = false;
296    }
297}
