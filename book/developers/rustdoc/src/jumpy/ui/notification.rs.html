11pub fn network_disconnect_notify(
12    meta: Root<GameMeta>,
13    ctx: Res<EguiCtx>,
14    mut sessions: ResMut<Sessions>,
15    world: &World,
16) {
17    let sessions = sessions.borrow_mut();
18    #[allow(unused_assignments, unused_mut)]
19    let mut all_players_disconnected = false;
20
21    #[cfg(not(target_arch = "wasm32"))]
22    {
23        all_players_disconnected = if let Some(game_session) = sessions.get(SessionNames::GAME) {
24            if let Some(disconnected_players) =
25                game_session.world.resources.get::<DisconnectedPlayers>()
26            {
27                // Determine if all remote players have been disconnected
28                // (Disconnected players should be player count - 1, can not have local player)
29                let player_indices = game_session.world.components.get::<PlayerIdx>().borrow();
30
31                !disconnected_players.disconnected_players.is_empty()
32                    && player_indices.bitset().bit_count() - 1
33                        == disconnected_players.disconnected_players.len()
34            } else {
35                false
36            }
37        } else {
38            false
39        };
40    }
41
42    if all_players_disconnected {
43        egui::CentralPanel::default()
44            .frame(egui::Frame::none())
45            .show(&ctx, |ui| {
46                ui.vertical_centered(|ui| {
47                    // Calculate a margin
48                    let screen_rect = ui.max_rect();
49                    let outer_margin = screen_rect.size() * 0.10;
50                    let outer_margin = egui::Margin {
51                        left: outer_margin.x,
52                        right: outer_margin.x,
53                        top: outer_margin.y * 3.0,
54                        bottom: outer_margin.y,
55                    };
56
57                    BorderedFrame::new(&meta.theme.panel.border)
58                        .margin(outer_margin)
59                        .padding(meta.theme.panel.padding)
60                        .show(ui, |ui| {
61                            world.run_system(
62                                network_disconnect_notify_ui,
63                                (ui, sessions.borrow_mut()),
64                            )
65                        });
66                });
67            });
68    }
69}

71fn network_disconnect_notify_ui(
72    mut param: In<(&mut egui::Ui, &mut Sessions)>,
73    localization: Localization<GameMeta>,
74    meta: Root<GameMeta>,
75) {
76    let (ui, sessions) = &mut *param;
77    ui.vertical_centered(|ui| {
78        ui.label(
79            meta.theme
80                .font_styles
81                .heading
82                .rich(localization.get("disconnected")),
83        );
84
85        ui.add_space(meta.theme.font_styles.normal.size);
86        ui.label(
87            meta.theme
88                .font_styles
89                .normal
90                .rich(localization.get("disconnected-from-all")),
91        );
92        ui.add_space(meta.theme.font_styles.normal.size);
93
94        if BorderedButton::themed(&meta.theme.buttons.normal, localization.get("exit-match"))
95            .show(ui)
96            .focus_by_default(ui)
97            .clicked()
98        {
99            sessions.add_command(Box::new(move |sessions: &mut Sessions| {
100                sessions.end_game();
101                sessions.start_menu();
102            }));
103        }
104    });
