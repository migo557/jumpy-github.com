26/// Resource containing the physics debug line entity.
27#[derive(HasSchema)]

31    debug_pipeline: rapier::DebugRenderPipeline,

34/// An implementation of the rapier `DebugRenderingBackend` that we use to create bones `Path2d`
35/// entities with.
36struct RapierDebugBackend<'a> {
37    points: &'a mut Vec<Vec2>,
38    line_breaks: &'a mut Vec<usize>,
39}
40
41impl<'a> rapier::DebugRenderBackend for RapierDebugBackend<'a> {
42    fn draw_line(
43        &mut self,
44        object: rapier::DebugRenderObject,
45        a: rapier::Point<rapier::Real>,
46        b: rapier::Point<rapier::Real>,
47        // TODO: implement multi-colored rendering
48        _color: [f32; 4],
49    ) {
50        let render = match object {
51            rapier::DebugRenderObject::RigidBody(_, body) => body.is_enabled(),
52            rapier::DebugRenderObject::Collider(_, collider) => collider.is_enabled(),
53            rapier::DebugRenderObject::ImpulseJoint(_, _) => true,
54            rapier::DebugRenderObject::MultibodyJoint(_, _, _) => true,
55            rapier::DebugRenderObject::ColliderAabb(_, _, _) => true,
56            rapier::DebugRenderObject::ContactPair(_, _, _) => true,
57        };
58        if render {
59            self.points.push(vec2(a.x, a.y));
60            self.points.push(vec2(b.x, b.y));
61            self.line_breaks.push(self.points.len());
62        }
63    }
64}

66impl Clone for RapierDebugContext {

70            debug_pipeline: default(),

75impl FromWorld for RapierDebugContext {
76    fn from_world(world: &World) -> Self {
77        let path_entity = world.resource_mut::<Entities>().create();
78
79        let transforms = world.components.get::<Transform>();
80        let mut transforms = transforms.borrow_mut();
81        transforms.insert(
82            path_entity,
83            Transform::from_translation(vec3(0.0, 0.0, -1.0)),
84        );
85
86        Self {
87            path_entity,
88            debug_pipeline: default(),
89        }
90    }
91}

93/// Renders debug lines for rapier colliders.
94fn debug_render_colliders(
95    settings: ResInit<DebugSettings>,
96    mut collision_world: CollisionWorld,
97    transforms: Comp<Transform>,
98    mut dynamic_bodies: CompMut<DynamicBody>,
99    mut paths: CompMut<Path2d>,
100    mut debug_context: ResMutInit<RapierDebugContext>,
101) {
102    if settings.show_kinematic_colliders {
103        // TODO: It's unfortunate that we are doing an extra sync here, just for debug rendering. We
104        // should try find a way to avoid this. Without this, the collider body positions will be
105        // out of sync when they are rendered.
106        collision_world.sync_bodies(&transforms, &mut dynamic_bodies);
107
108        let mut points = Vec::new();
109        let mut line_breaks = Vec::new();
110
111        debug_context.debug_pipeline.render_colliders(
112            &mut RapierDebugBackend {
113                points: &mut points,
114                line_breaks: &mut line_breaks,
115            },
116            &collision_world.ctx.rigid_body_set,
117            &collision_world.ctx.collider_set,
118        );
119
120        // TODO: Provide a way to change the collider colors
121        paths.insert(
122            debug_context.path_entity,
123            Path2d {
124                // An orange-y color
125                color: Color::from([205.0 / 255.0, 94.0 / 255.0, 15.0 / 255.0, 1.0]),
126                points,
127                line_breaks,
128                ..default()
129            },
130        );
131    } else {
132        paths.remove(debug_context.path_entity);
133    }
134}

136/// Renders debug lines for damage regions.
137fn debug_render_damage_regions(
138    settings: ResInit<DebugSettings>,
139    entities: Res<Entities>,
140    regions: Comp<DamageRegion>,
141    transforms: Comp<Transform>,
142    mut paths: CompMut<Path2d>,
143) {
144    let path_for_region = |rotation: f32, region: &DamageRegion| {
145        let rect = Rect::new(0.0, 0.0, region.size.x, region.size.y);
146
147        // The collision boxes don't rotate, so apply the opposite rotation of the object to the
148        // debug lines to keep it upright.
149        let angle = Vec2::from_angle(-rotation);
150
151        Path2d {
152            color: Color::RED,
153            points: vec![
154                angle.rotate(rect.top_left()),
155                angle.rotate(rect.top_right()),
156                angle.rotate(rect.bottom_right()),
157                angle.rotate(rect.bottom_left()),
158                angle.rotate(rect.top_left()),
159            ],
160            thickness: 1.0,
161            ..default()
162        }
163    };
164
165    if settings.show_damage_regions {
166        for (ent, (region, transform)) in entities.iter_with((&regions, &transforms)) {
167            paths.insert(
168                ent,
169                path_for_region(transform.rotation.to_euler(EulerRot::XYZ).2, region),
170            );
171        }
172    } else {
173        for ent in entities.iter_with_bitset(regions.bitset()) {
174            paths.remove(ent);
175        }
176    }
177}
