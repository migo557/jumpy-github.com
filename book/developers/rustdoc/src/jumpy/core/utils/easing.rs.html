5/// Simple easing calculator
6#[derive(Clone, Debug)]
7pub struct Ease {
8    pub ease_in: bool,
9    pub ease_out: bool,
10    pub function: EaseFunction,
11    pub progress: f32,
12}
13
14#[derive(HasSchema, Clone, Default)]
15#[type_data(metadata_asset("ease"))]
16#[repr(C)]
17pub struct EaseMeta {
18    pub ease_in: bool,
19    pub ease_out: bool,
20    pub function: EaseFunction,
21}
22
23#[derive(HasSchema, Copy, Clone, Debug, Default)]
24#[repr(u8)]
25pub enum EaseFunction {
26    #[default]
27    Quadratic,
28    Cubic,
29    Sinusoidial,
30}
31
32impl Default for Ease {
33    fn default() -> Self {
34        Self {
35            ease_in: true,
36            ease_out: true,
37            function: EaseFunction::Quadratic,
38            progress: 0.0,
39        }
40    }
41}
42
43impl Ease {
44    pub fn output(&self) -> f32 {
45        let mut k = self.progress;
46
47        // Reference for easing functions:
48        // https://echarts.apache.org/examples/en/editor.html?c=line-easing&lang=ts
49        //
50        // TODO: Add tests to make sure easings are correct
51        match (&self.function, self.ease_in, self.ease_out) {
52            (EaseFunction::Quadratic, true, true) => {
53                k *= 2.0;
54                if k < 1.0 {
55                    0.5 * k * k
56                } else {
57                    k -= 1.0;
58                    -0.5 * (k * (k - 2.0) - 1.0)
59                }
60            }
61            (EaseFunction::Quadratic, true, false) => k * k,
62            (EaseFunction::Quadratic, false, true) => k * (2.0 - k),
63            (EaseFunction::Cubic, true, true) => {
64                k *= 2.0;
65                if k < 1.0 {
66                    0.5 * k * k * k
67                } else {
68                    k -= 2.0;
69                    0.5 * (k * k * k + 2.0)
70                }
71            }
72            (EaseFunction::Cubic, true, false) => k * k * k,
73            (EaseFunction::Cubic, false, true) => {
74                k -= 1.0;
75                k * k * k + 1.0
76            }
77            (EaseFunction::Sinusoidial, true, true) => 0.5 * (1.0 - f32::cos(PI * k)),
78            (EaseFunction::Sinusoidial, true, false) => 1.0 - f32::cos((k * PI) / 2.0),
79            (EaseFunction::Sinusoidial, false, true) => f32::sin((k * PI) / 2.0),
80            (_, false, false) => k,
81        }
82    }
83}
