1//! Utilities for asset and meta data handling.
2
3use crate::prelude::*;
4
5/// Error type for failure to resolve [`UntypedHandle`] to metadata asset from  `Handle<T>` or `Handle<ElementMeta>`
6/// (where `T` is metadata type).
7#[derive(thiserror::Error, Debug, Clone)]
8pub enum MetaHandleCastError {
9    #[error("UntypedHandle does not represent Handle<{0}> or Handle<ElementMeta>")]
10    BadHandleType(String),
11    #[error("UntypedHandle maps to Handle<ElementMeta> but ElementMeta inner handle does not cast to {0}")]
12    ElementDataMismatch(String),
13    #[error("Failed to retrieve asset for handle (required for cast type valdiation)")]
14    InvalidHandle,
15}
16
17/// Try to get metadata handle from [`UntypedHandle`] that may represent direct handle to meta (`Handle<T>`)
18/// or `Handle<ElementMeta>` where [`ElementMeta`]'s data is castable to `Handle<T>`.
19///
20/// [`Handle::untyped`] can be used to convert to [`UntypedHandle`].
21///
22/// This is useful for code that wants to spawn an item and take UntypedHandle to allow either direct meta handle
23/// or `Handle<ElementMeta>` as argument.
24pub fn try_cast_meta_handle<T: HasSchema>(
25    handle: UntypedHandle,
26    assets: &AssetServer,
27) -> Result<Handle<T>, MetaHandleCastError> {
28    // Get untyped asset from handle or error on failure
29    let asset = assets
30        .try_get_untyped(handle)
31        .ok_or(MetaHandleCastError::InvalidHandle)?;
32
33    // If asset casts to T, return it as Handle<T>
34    if asset.try_cast_ref::<T>().is_ok() {
35        return Ok(handle.typed::<T>());
36    }
37
38    // Check if handle type is ElementMeta
39    if let Ok(element_meta) = asset.try_cast_ref::<ElementMeta>() {
40        // Does element data cast to T?
41        if assets.get(element_meta.data).try_cast_ref::<T>().is_ok() {
42            // Return ElementMeta's data as Handle<T>
43            Ok(element_meta.data.untyped().typed())
44        } else {
45            // ElementMeta data does not cast to T.
46            Err(MetaHandleCastError::ElementDataMismatch(
47                std::any::type_name::<T>().to_string(),
48            ))
49        }
50    } else {
51        // UntypedHandle is neither Handle<T> or Handle<ElementMeta>.
52        Err(MetaHandleCastError::BadHandleType(
53            std::any::type_name::<T>().to_string(),
54        ))
55    }
56}
