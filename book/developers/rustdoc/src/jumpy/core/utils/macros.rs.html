1/// Macro to "derive" ( not really a derive macro ) SystemParam for a struct.
2#[macro_export]
3macro_rules! impl_system_param {
4    (
5        $( #[$m:meta] )*
6        pub struct $t:ident<'a> {
7            $(
8                $( #[$attrs:meta] )*
9                $f_name:ident: $f_ty:ty
10            ),*
11            $(,)?
12        }
13    ) => {
14        $( #[$m] )*
15        pub struct $t<'a> {
16            $(
17                $( #[$attrs] )*
18                pub $f_name: $f_ty
19            ),*
20        }
21
22        impl<'a> SystemParam for $t<'a> {
23            type State = (
24                $(
25                    <$f_ty as SystemParam>::State
26                ),*
27            );
28            type Param<'p> = $t<'p>;
29
30            fn get_state(world: &World) -> Self::State {
31                (
32                    $(
33                        <$f_ty as SystemParam>::get_state(world)
34                    ),*
35                )
36            }
37
38            fn borrow<'s>(world: &'s World, state: &'s mut Self::State) -> Self::Param<'s> {
39                let (
40                    $(
41                        $f_name
42                    ),*
43                ) = state;
44                let (
45                    $(
46                        $f_name
47                    ),*
48                ) = (
49                    $(
50                        <$f_ty as SystemParam>::borrow(world, $f_name)
51                    ),*
52                );
53
54                Self::Param {
55                    $(
56                        $f_name
57                    ),*
58                }
59            }
60        }
61    };
62}
