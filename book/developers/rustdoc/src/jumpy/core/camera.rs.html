15/// A sprite that is spawned as a part of the parallax background.
16#[derive(Clone, HasSchema, Default)]
17pub struct ParallaxBackgroundSprite {
18    /// The sprite with `idx` of `0` will be centered, with indexes increasing and decreasing
19    /// representing repeated tiles to the right and left respectively.
20    pub idx: i32,
21    /// Information about the parallax layer it is a part of.
22    pub meta: ParallaxLayerMeta,
23}

25/// A subject of the camera.
26///
27/// The camera will move and zoom to ensure that all subjects remain visible. Entities must also
28/// have a `Transform` component and a `KinematicBody` component for this to work properly.
29#[derive(Clone, Copy, Debug, Default, HasSchema)]
30pub struct CameraSubject {
31    /// A rectangle around the subject that is larger than the subject, and will always move to
32    /// contain it. The camera will seek to contain this rectangle, instead of the subject itself.
33    ///
34    /// The advantage of this processing method is that the larger rect doesn't move as much as the
35    /// subject, because, for instance, a player jumping up and down in place will still be inside
36    /// of their camera rect, so the camera will not move around annoyingly.
37    rect: Rect,
38}
39
40/// The state of the camera.
41#[derive(Clone, Debug, HasSchema, Default)]
42pub struct CameraState {
43    /// Disables the default camera controller. Useful, for example, when taking over the camera
44    /// from the editor.
45    pub disable_controller: bool,
46}

48/// Implemenets the camera controller.
49fn camera_controller(
50    meta: Root<GameMeta>,
51    entities: Res<Entities>,
52    map: Res<LoadedMap>,
53    mut cameras: CompMut<Camera>,
54    mut camera_shakes: CompMut<CameraShake>,
55    camera_states: Comp<CameraState>,
56    mut camera_subjects: CompMut<CameraSubject>,
57    transforms: Comp<Transform>,
58    bodies: Comp<KinematicBody>,
59    window: Res<Window>,
60) {
61    let meta = &meta.core.camera;
62
63    let Some((_ent, (camera, camera_shake, camera_state))) = entities
64        .iter_with((&mut cameras, &mut camera_shakes, &camera_states))
65        .next()
66    else {
67        return;
68    };
69    if camera_state.disable_controller {
70        return;
71    }
72
73    // Update player camera rects
74    for (_ent, (camera_subj, transform, body)) in
75        entities.iter_with((&mut camera_subjects, &transforms, &bodies))
76    {
77        let camera_box_size = meta.player_camera_box_size;
78
79        // Get the player's camera box
80        let camera_box = &mut camera_subj.rect;
81
82        // If it's not be initialized.
83        if camera_box.min == Vec2::ZERO && camera_box.max == Vec2::ZERO {
84            // Set it's size and position according to the player
85            *camera_box = Rect::new(
86                transform.translation.x,
87                transform.translation.y,
88                camera_box_size.x,
89                camera_box_size.y,
90            );
91        }
92
93        // Get the player's body rect
94        let rect = body.bounding_box(*transform);
95
96        // Push the camera rect just enough to contain the player's body rect
97        if rect.min.x < camera_box.min.x {
98            camera_box.min.x = rect.min.x;
99            camera_box.max.x = camera_box.min.x + camera_box_size.x;
100        }
101        if rect.max.x > camera_box.max.x {
102            camera_box.max.x = rect.max.x;
103            camera_box.min.x = rect.max.x - camera_box_size.x;
104        }
105        if rect.min.y < camera_box.min.y {
106            camera_box.min.y = rect.min.y;
107            camera_box.max.y = camera_box.min.y + camera_box_size.y;
108        }
109        if rect.max.y > camera_box.max.y {
110            camera_box.max.y = rect.max.y;
111            camera_box.min.y = rect.max.y - camera_box_size.y;
112        }
113    }
114
115    let viewport_size = camera
116        .viewport
117        .option()
118        .map(|x| x.size.as_vec2())
119        .unwrap_or(window.size);
120    let viewport_aspect = viewport_size.x / viewport_size.y;
121    let default_height = meta.default_height;
122    let default_width = viewport_aspect * default_height;
123    let camera_height = if let CameraSize::FixedHeight(height) = camera.size {
124        height
125    } else {
126        400.0
127    };
128    let mut scale = camera_height / default_height;
129    let map_size = map.grid_size.as_vec2() * map.tile_size;
130
131    let mut min = Vec2::MAX;
132    let mut max = Vec2::MIN;
133
134    for CameraSubject { rect } in camera_subjects.iter_mut() {
135        min = (rect.min - vec2(meta.border_left, meta.border_bottom))
136            .min(min)
137            .max(Vec2::ZERO);
138        max = (rect.max + vec2(meta.border_right, meta.border_top)).max(max);
139        max.x = max.x.min(map_size.x)
140    }
141
142    let camera_pos = &mut camera_shake.center;
143
144    let subject_count = camera_subjects.iter().count();
145    let mut middle_point = if subject_count == 0 {
146        camera_pos.truncate()
147    } else {
148        Rect { min, max }.center()
149    };
150
151    let size = max - min;
152    let size = size.max(meta.min_camera_size);
153
154    let rh = size.y / default_height;
155    let rw = size.x / default_width;
156    let r_target = if rh > rw { rh } else { rw };
157    let r_diff = r_target - scale;
158    if r_diff > 0.0 {
159        scale += r_diff * meta.zoom_out_lerp_factor;
160    } else {
161        scale += r_diff * meta.zoom_in_lerp_factor;
162    }
163
164    // Keep camera above the map floor
165    if middle_point.y - size.y / 2. < 0.0 {
166        middle_point.y = size.y / 2.0;
167    }
168
169    let delta = camera_pos.truncate() - middle_point;
170    let dist = delta * meta.move_lerp_factor;
171    camera.size = CameraSize::FixedHeight(scale * default_height);
172    *camera_pos -= dist.extend(0.0);
173}

175/// Implements the background layer parallax.
176fn camera_parallax(
177    entities: Res<Entities>,
178    mut transforms: CompMut<Transform>,
179    parallax_bg_sprites: Comp<ParallaxBackgroundSprite>,
180    cameras: Comp<Camera>,
181    map: Res<LoadedMap>,
182) {
183    // TODO: This constant represents that maximum camera-visible distance, and should be moved
184    // somewhere more appropriate.
185    const FAR_PLANE: f32 = 1000.0;
186
187    let map_size = map.grid_size.as_vec2() * map.tile_size;
188
189    let camera_transform = entities
190        .iter_with((&transforms, &cameras))
191        .next()
192        .map(|x| x.1 .0)
193        .copied()
194        .unwrap();
195    let camera_offset = map_size / 2.0 - camera_transform.translation.truncate();
196
197    for (_ent, (transform, bg)) in entities.iter_with((&mut transforms, &parallax_bg_sprites)) {
198        transform.scale.x = bg.meta.scale;
199        transform.scale.y = bg.meta.scale;
200        let display_size = transform.scale.truncate() * bg.meta.size;
201        transform.translation.x = bg.idx as f32 * display_size.x;
202        transform.translation.y = map_size.y / 2.0;
203        transform.translation.z = -FAR_PLANE + 1.0 - bg.meta.depth / 100.0;
204        transform.translation += bg.meta.offset.extend(1.0);
205
206        transform.translation.x -= camera_offset.x * bg.meta.depth * map.background.speed.x;
207        transform.translation.y += camera_offset.y * bg.meta.depth * map.background.speed.y;
208    }
209}
