3/// A crab roaming on the ocean floor
4#[derive(HasSchema, Default, Debug, Clone)]
5#[type_data(metadata_asset("crab"))]
6#[repr(C)]
7pub struct CrabMeta {
8    pub body_size: Vec2,
9    pub walk_frames: SVec<u32>,
10    pub spawn_frames: SVec<u32>,
11    pub fps: f32,
12    pub comfortable_spawn_distance: f32,
13    pub comfortable_scared_distance: f32,
14    /// How long a crab has to be away from it's spawn point before it digs into the ground and
15    /// digs back out in his spawn point.
16    pub uncomfortable_respawn_time: Duration,
17    pub same_level_threshold: f32,
18    pub walk_speed: f32,
19    pub run_speed: f32,
20    // TODO: migrate this to a duration like `uncomfortable_respawn_time`.
21    pub timer_delay_max: u8,
22    pub atlas: Handle<Atlas>,
23}

36#[derive(Default, Clone, HasSchema, Debug, Copy)]
37pub enum CrabState {
38    #[default]
39    Paused,
40    Walking {
41        left: bool,
42    },
43    Fleeing {
44        scared_of: Entity,
45    },
46    Spawning,
47    Despawning,
48}
49
50impl CrabState {
51    fn is_moving(&self) -> bool {
52        matches!(self, CrabState::Walking { .. } | CrabState::Fleeing { .. })
53    }
54}

56#[derive(HasSchema, Clone, Default)]
57pub struct CrabCritter {
58    state: CrabState,
59    start_pos: Vec2,
60    state_count: u8,
61    state_count_max: u8,
62    respawn_timer: Timer,
63}

65fn hydrate(
66    game_meta: Root<GameMeta>,
67    mut entities: ResMutInit<Entities>,
68    mut hydrated: CompMut<MapElementHydrated>,
69    mut despawns: CompMut<DehydrateOutOfBounds>,
70    mut element_handles: CompMut<ElementHandle>,
71    mut crabs: CompMut<CrabCritter>,
72    mut atlas_sprites: CompMut<AtlasSprite>,
73    mut bodies: CompMut<KinematicBody>,
74    mut transforms: CompMut<Transform>,
75    mut animated_sprites: CompMut<AnimatedSprite>,
76    mut spawner_manager: SpawnerManager,
77    assets: Res<AssetServer>,
78) {
79    let mut not_hydrated_bitset = hydrated.bitset().clone();
80    not_hydrated_bitset.bit_not();
81    not_hydrated_bitset.bit_and(element_handles.bitset());
82
83    let spawner_entities = entities
84        .iter_with_bitset(&not_hydrated_bitset)
85        .collect::<Vec<_>>();
86
87    for spawner_ent in spawner_entities {
88        let transform = *transforms.get(spawner_ent).unwrap();
89        let element_handle = *element_handles.get(spawner_ent).unwrap();
90        let element_meta = assets.get(element_handle.0);
91
92        if let Ok(CrabMeta {
93            atlas,
94            body_size,
95            spawn_frames,
96            uncomfortable_respawn_time,
97            ..
98        }) = assets.get(element_meta.data).try_cast_ref()
99        {
100            hydrated.insert(spawner_ent, MapElementHydrated);
101
102            let entity = entities.create();
103            crabs.insert(
104                entity,
105                CrabCritter {
106                    state_count: 0,
107                    state_count_max: 60,
108                    state: CrabState::Spawning,
109                    start_pos: Vec2::new(transform.translation.x, transform.translation.y),
110                    respawn_timer: Timer::from_seconds(
111                        uncomfortable_respawn_time.as_secs_f32(),
112                        TimerMode::Once,
113                    ),
114                },
115            );
116            atlas_sprites.insert(entity, AtlasSprite::new(*atlas));
117            transforms.insert(entity, transform);
118            element_handles.insert(entity, element_handle);
119            hydrated.insert(entity, MapElementHydrated);
120            despawns.insert(entity, DehydrateOutOfBounds(spawner_ent));
121
122            bodies.insert(
123                entity,
124                KinematicBody {
125                    gravity: game_meta.core.physics.gravity,
126                    has_mass: true,
127                    has_friction: true,
128                    shape: ColliderShape::Rectangle { size: *body_size },
129                    ..default()
130                },
131            );
132
133            animated_sprites.insert(
134                entity,
135                AnimatedSprite {
136                    repeat: false,
137                    frames: spawn_frames.iter().cloned().collect(),
138                    ..default()
139                },
140            );
141
142            spawner_manager.create_spawner(spawner_ent, vec![entity])
143        }
144    }
145}

147fn update_crabs(
148    rng: Res<GlobalRng>,
149    time: Res<Time>,
150    entities: Res<Entities>,
151    player_indexes: Comp<PlayerIdx>,
152    mut crabs: CompMut<CrabCritter>,
153    mut sprites: CompMut<AtlasSprite>,
154    mut animated_sprites: CompMut<AnimatedSprite>,
155    mut bodies: CompMut<KinematicBody>,
156    mut transforms: CompMut<Transform>,
157    mut hydrated: CompMut<MapElementHydrated>,
158    mut commands: Commands,
159    spawners: Comp<DehydrateOutOfBounds>,
160    element_handles: Comp<ElementHandle>,
161    assets: Res<AssetServer>,
162) {
163    for (entity, (crab, element_handle, spawner)) in
164        entities.iter_with((&mut crabs, &element_handles, &spawners))
165    {
166        let element_meta = assets.get(element_handle.0);
167
168        let asset = assets.get(element_meta.data);
169        let CrabMeta {
170            fps,
171            run_speed,
172            walk_speed,
173            walk_frames,
174            spawn_frames,
175            comfortable_spawn_distance,
176            comfortable_scared_distance,
177            same_level_threshold,
178            timer_delay_max,
179            ..
180        } = asset.cast_ref();
181
182        let body = bodies.get_mut(entity).unwrap();
183        let sprite = sprites.get_mut(entity).unwrap();
184        let animated_sprite = animated_sprites.get_mut(entity).unwrap();
185        let transform = transforms.get_mut(entity).unwrap();
186        let pos = Vec2::new(transform.translation.x, transform.translation.y);
187
188        if (crab.start_pos - pos).length() > *comfortable_spawn_distance {
189            crab.respawn_timer.tick(time.delta());
190        } else {
191            crab.respawn_timer.reset();
192        }
193
194        if let CrabState::Despawning = crab.state {
195            if finished_playing(animated_sprite).unwrap() {
196                crab.state = CrabState::Spawning;
197                hydrated.remove(**spawner);
198                commands.add(move |mut entities: ResMutInit<Entities>| entities.kill(entity));
199            }
200            continue;
201        } else if body.is_on_ground && crab.respawn_timer.finished() {
202            crab.state = CrabState::Despawning;
203            body.is_deactivated = true;
204            *animated_sprite = AnimatedSprite {
205                frames: spawn_frames.iter().rev().cloned().collect(),
206                repeat: false,
207                fps: *fps,
208                ..default()
209            };
210            continue;
211        }
212
213        if let CrabState::Spawning = crab.state {
214            if body.is_on_ground {
215                body.velocity = Vec2::ZERO;
216                animated_sprite.fps = *fps;
217                if finished_playing(animated_sprite).unwrap() {
218                    crab.state = CrabState::Paused;
219                    *animated_sprite = AnimatedSprite {
220                        frames: walk_frames.iter().cloned().collect(),
221                        repeat: true,
222                        fps: *fps,
223                        ..default()
224                    };
225                }
226            }
227            continue;
228        }
229
230        crab.state_count += 1;
231        let rand_bool = |true_bias: u8| -> bool { rng.u8(0..(1_u8 + true_bias)) > 0 };
232        let rand_timer_delay = |max: u8| rng.u8(0..max);
233
234        let next_scary_thing = || {
235            for (scary_entity, (_player_idx, scary_thing_transform)) in
236                entities.iter_with((&player_indexes, &transforms))
237            {
238                let scary_thing_pos = Vec2::new(
239                    scary_thing_transform.translation.x,
240                    scary_thing_transform.translation.y,
241                );
242
243                if pos.distance(scary_thing_pos) < *comfortable_scared_distance
244                    && (pos.y - scary_thing_pos.y).abs() < *same_level_threshold
245                {
246                    return Some(scary_entity);
247                }
248            }
249            None
250        };
251
252        let pick_next_move = |crab: &CrabCritter| {
253            let pause_bias = if crab.state.is_moving() { 2 } else { 0 };
254
255            if rand_bool(pause_bias) {
256                (CrabState::Paused, rand_timer_delay(*timer_delay_max))
257            } else {
258                let distance_from_home = pos.x - crab.start_pos.x;
259                let left = if distance_from_home.abs() > *comfortable_spawn_distance && rand_bool(2)
260                {
261                    distance_from_home > 0.0
262                } else {
263                    rand_bool(0)
264                };
265                (
266                    CrabState::Walking { left },
267                    rand_timer_delay(*timer_delay_max),
268                )
269            }
270        };
271
272        let get_scared_of_pos = |scared_of: Entity| {
273            transforms
274                .get(scared_of)
275                .map(|x| x.translation.xy())
276                .unwrap_or_default()
277        };
278
279        if crab.state_count >= crab.state_count_max {
280            crab.state_count = 0;
281
282            if let Some(scared_of_pos) = next_scary_thing() {
283                crab.state = CrabState::Fleeing {
284                    scared_of: scared_of_pos,
285                };
286                crab.state_count_max = rand_timer_delay(*timer_delay_max);
287            } else {
288                match &crab.state {
289                    CrabState::Paused | CrabState::Walking { .. } => {
290                        let (state, timer) = pick_next_move(crab);
291                        crab.state = state;
292                        crab.state_count_max = timer;
293                    }
294                    CrabState::Fleeing { scared_of } => {
295                        let scared_of_pos = get_scared_of_pos(*scared_of);
296
297                        if pos.distance(scared_of_pos) > *comfortable_scared_distance {
298                            if let Some(scared_of) = next_scary_thing() {
299                                crab.state = CrabState::Fleeing { scared_of };
300                                crab.state_count_max = rand_timer_delay(*timer_delay_max / 3);
301                            } else {
302                                let (state, timer) = pick_next_move(crab);
303                                crab.state = state;
304                                crab.state_count_max = timer;
305                            }
306                        }
307                    }
308                    _ => {}
309                }
310            }
311        }
312
313        match &crab.state {
314            CrabState::Paused => {
315                animated_sprite.fps = 0.0;
316                body.velocity.x = 0.0;
317            }
318            CrabState::Walking { left } => {
319                animated_sprite.fps = *fps;
320                sprite.flip_x = *left;
321
322                let direction = if *left { -1.0 } else { 1.0 };
323                body.velocity.x = *walk_speed * direction;
324            }
325            CrabState::Fleeing { scared_of } => {
326                animated_sprite.fps = *fps;
327                let scared_of_pos = get_scared_of_pos(*scared_of);
328                let direction = (pos.x - scared_of_pos.x).signum();
329                body.velocity.x = direction * *run_speed;
330            }
331            _ => {}
332        }
333    }
334}

336fn finished_playing(
337    AnimatedSprite {
338        index,
339        frames,
340        fps,
341        timer,
342        repeat,
343    }: &AnimatedSprite,
344) -> Option<bool> {
345    (!repeat)
346        .then(|| *index == frames.len() as u32 - 1 && *timer > 1.0 / fps.max(f32::MIN_POSITIVE))
