31fn hydrate(
32    entities: Res<Entities>,
33    mut hydrated: CompMut<MapElementHydrated>,
34    element_handles: Comp<ElementHandle>,
35    assets: Res<AssetServer>,
36    mut slippery: CompMut<Slippery>,
37    mut atlas_sprites: CompMut<AtlasSprite>,
38    mut bodies: CompMut<KinematicBody>,
39) {
40    let mut not_hydrated_bitset = hydrated.bitset().clone();
41    not_hydrated_bitset.bit_not();
42    not_hydrated_bitset.bit_and(element_handles.bitset());
43
44    for entity in entities.iter_with_bitset(&not_hydrated_bitset) {
45        let element_handle = element_handles.get(entity).unwrap();
46        let element_meta = assets.get(element_handle.0);
47
48        if let Ok(SlipperyMeta {
49            atlas,
50            body_size,
51            player_slide,
52            body_friction,
53        }) = assets.get(element_meta.data).try_cast_ref()
54        {
55            hydrated.insert(entity, MapElementHydrated);
56            atlas_sprites.insert(entity, AtlasSprite::new(*atlas));
57            bodies.insert(
58                entity,
59                KinematicBody {
60                    shape: ColliderShape::Rectangle { size: *body_size },
61                    has_mass: false,
62                    ..default()
63                },
64            );
65            slippery.insert(
66                entity,
67                Slippery {
68                    player_slide: *player_slide,
69                    body_friction: *body_friction,
70                },
71            );
72        }
73    }
74}

76pub fn update(
77    entities: Res<Entities>,
78    slippery: CompMut<Slippery>,
79    collision_world: CollisionWorld,
80    mut bodies: CompMut<KinematicBody>,
81) {
82    for (slippery_ent, slippery) in entities.iter_with(&slippery) {
83        for (p_ent, body) in entities.iter_with(&mut bodies) {
84            if collision_world
85                .actor_collisions(p_ent)
86                .contains(&slippery_ent)
87            {
88                body.frame_friction_override = Some(slippery.body_friction);
89            }
90        }
91    }
