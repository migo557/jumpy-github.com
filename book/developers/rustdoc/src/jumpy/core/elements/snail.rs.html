3#[derive(HasSchema, Default, Debug, Clone)]
4#[type_data(metadata_asset("snail"))]

6pub struct SnailMeta {
7    pub atlas: Handle<Atlas>,
8    pub fps: f32,
9    pub body_diameter: f32,
10    pub bounciness: f32,
11    pub gravity: f32,
12    pub hit_speed: f32,
13    /// The animation frames for when the snail is crawling
14    pub crawl_frames: SVec<u32>,
15    /// The `crawl_frames` indexes in which to move the snail
16    pub move_frame_indexes: SVec<u32>,
17    /// The animation frames for when the snail is fleeing into its shell.
18    ///
19    /// **Note:** This is reversed for the snail coming out of its shell.
20    pub hide_frames: SVec<u32>,
21    pub hide_time: f32,
22}

36#[derive(Clone, HasSchema, Debug, Copy, Default)]
37pub enum Snail {
38    Hiding(f32),
39    #[default]
40    UnHiding,
41    Moving {
42        attempted_x: Option<f32>,
43        frame_timer: f32,
44        anim_index: u32,
45    },
46}

48fn hydrate(
49    mut entities: ResMutInit<Entities>,
50    mut hydrated: CompMut<MapElementHydrated>,
51    mut dehrydate_bounds: CompMut<DehydrateOutOfBounds>,
52    mut element_handles: CompMut<ElementHandle>,
53    assets: Res<AssetServer>,
54    mut snails: CompMut<Snail>,
55    mut atlas_sprites: CompMut<AtlasSprite>,
56    mut animated_sprites: CompMut<AnimatedSprite>,
57    mut animation_banks: CompMut<AnimationBankSprite>,
58    mut bodies: CompMut<KinematicBody>,
59    mut transforms: CompMut<Transform>,
60    mut spawner_manager: SpawnerManager,
61) {
62    let mut not_hydrated_bitset = hydrated.bitset().clone();
63    not_hydrated_bitset.bit_not();
64    not_hydrated_bitset.bit_and(element_handles.bitset());
65
66    let spawner_entities = entities
67        .iter_with_bitset(&not_hydrated_bitset)
68        .collect::<Vec<_>>();
69
70    for spawner_ent in spawner_entities {
71        let transform = *transforms.get(spawner_ent).unwrap();
72        let element_handle = *element_handles.get(spawner_ent).unwrap();
73        let element_meta = assets.get(element_handle.0);
74
75        if let Ok(SnailMeta {
76            atlas,
77            fps,
78            body_diameter,
79            gravity,
80            bounciness,
81            hide_time,
82            hide_frames,
83            ..
84        }) = assets.get(element_meta.data).try_cast_ref()
85        {
86            hydrated.insert(spawner_ent, MapElementHydrated);
87
88            let entity = entities.create();
89            snails.insert(entity, Snail::Hiding(*hide_time));
90            transforms.insert(entity, transform);
91            element_handles.insert(entity, element_handle);
92            hydrated.insert(entity, MapElementHydrated);
93            dehrydate_bounds.insert(entity, DehydrateOutOfBounds(spawner_ent));
94
95            bodies.insert(
96                entity,
97                KinematicBody {
98                    gravity: *gravity,
99                    has_mass: true,
100                    has_friction: true,
101                    bounciness: *bounciness,
102                    shape: ColliderShape::Circle {
103                        diameter: *body_diameter,
104                    },
105                    ..default()
106                },
107            );
108            atlas_sprites.insert(entity, AtlasSprite::new(*atlas));
109            animated_sprites.insert(entity, default());
110
111            let mut animations = SMap::default();
112
113            animations.insert("disabled".into(), default());
114            animations.insert(
115                "hide".into(),
116                AnimatedSprite {
117                    frames: hide_frames.iter().cloned().collect(),
118                    fps: *fps,
119                    repeat: false,
120                    ..default()
121                },
122            );
123            animations.insert(
124                "unhide".into(),
125                AnimatedSprite {
126                    frames: hide_frames.iter().rev().cloned().collect(),
127                    fps: *fps,
128                    repeat: false,
129                    ..default()
130                },
131            );
132            animation_banks.insert(
133                entity,
134                AnimationBankSprite {
135                    current: "hide".into(),
136                    animations,
137                    last_animation: "hide".into(),
138                },
139            );
140            spawner_manager.create_spawner(spawner_ent, vec![entity])
141        }
142    }
143}

145fn update_snails(
146    entities: Res<Entities>,
147    mut snails: CompMut<Snail>,
148    mut sprites: CompMut<AtlasSprite>,
149    mut animated_sprites: CompMut<AnimatedSprite>,
150    mut animation_banks: CompMut<AnimationBankSprite>,
151    time: Res<Time>,
152    mut bodies: CompMut<KinematicBody>,
153    mut transforms: CompMut<Transform>,
154    element_handles: Comp<ElementHandle>,
155    assets: Res<AssetServer>,
156    damage_regions: CompMut<DamageRegion>,
157) {
158    for (entity, (snail, element_handle, body, sprite)) in
159        entities.iter_with((&mut snails, &element_handles, &mut bodies, &mut sprites))
160    {
161        let element_meta = assets.get(element_handle.0);
162
163        let asset = assets.get(element_meta.data);
164        let Ok(SnailMeta {
165            fps,
166            bounciness,
167            hide_time,
168            hit_speed,
169            crawl_frames,
170            move_frame_indexes,
171            hide_frames,
172            ..
173        }) = asset.try_cast_ref()
174        else {
175            unreachable!();
176        };
177        let Some(animated_sprite) = animated_sprites.get_mut(entity) else {
178            continue;
179        };
180        let Some(animation_bank) = animation_banks.get_mut(entity) else {
181            continue;
182        };
183
184        let mut hit = false;
185
186        for (_, (damage_region, region_transform)) in
187            entities.iter_with((&damage_regions, &transforms))
188        {
189            let transform = transforms.get(entity).unwrap();
190            let region_pos = region_transform.translation;
191            if damage_region
192                .collider_rect(region_pos)
193                .overlaps(&body.bounding_box(*transform))
194            {
195                hit = true;
196                body.velocity = -Vec2::from_angle(
197                    Vec2::X.angle_between(region_pos.xy() - transform.translation.xy()),
198                ) * *hit_speed;
199            }
200        }
201
202        match &snail {
203            Snail::UnHiding | Snail::Moving { .. } => {
204                if hit {
205                    *snail = Snail::Hiding(0.0);
206                    body.bounciness = *bounciness;
207                    animation_bank.current = "hide".into();
208                }
209            }
210            _ => {}
211        }
212
213        match snail {
214            Snail::Hiding(time) => {
215                if hit {
216                    *time = 0.0;
217                    sprite.flip_x = !sprite.flip_x;
218                } else {
219                    *time += 1.0 / crate::FPS;
220                    if *time >= *hide_time {
221                        *snail = Snail::UnHiding;
222                        animation_bank.current = "unhide".into();
223                    }
224                }
225            }
226            Snail::UnHiding => {
227                if animated_sprite.index == hide_frames.len() as u32 - 1 {
228                    *snail = Snail::Moving {
229                        attempted_x: None,
230                        frame_timer: 0.0,
231                        anim_index: 0,
232                    };
233                    sprite.index = 0;
234                    animation_bank.current = "disabled".into();
235                    body.bounciness = 0.0;
236                }
237            }
238            Snail::Moving {
239                attempted_x,
240                frame_timer,
241                anim_index,
242            } => {
243                let translation = &mut transforms.get_mut(entity).unwrap().translation;
244
245                if let Some(attempted_x) = attempted_x {
246                    if attempted_x != &translation.x {
247                        *snail = Snail::Hiding(*hide_time);
248                        body.bounciness = *bounciness;
249                        animation_bank.current = "hide".into();
250                        sprite.flip_x = !sprite.flip_x;
251                        continue;
252                    }
253                }
254
255                *frame_timer += time.delta_seconds();
256
257                if *frame_timer > 1.0 / fps.max(f32::MIN_POSITIVE) {
258                    *frame_timer = 0.0;
259                    *anim_index = (*anim_index + 1) % crawl_frames.len() as u32;
260                    sprite.index = crawl_frames[*anim_index as usize];
261
262                    if move_frame_indexes.iter().any(|i| i == anim_index) {
263                        let next_x = translation.x - 1.0 * if sprite.flip_x { -1.0 } else { 1.0 };
264                        *attempted_x = Some(next_x);
265                        translation.x = next_x;
266                    }
267                } else {
268                    *attempted_x = None
269                }
270            }
271        }
272    }
