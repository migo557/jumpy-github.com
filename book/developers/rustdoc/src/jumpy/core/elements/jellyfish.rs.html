63fn hydrate(
64    game_meta: Root<GameMeta>,
65    mut hydrated: CompMut<MapElementHydrated>,
66    mut element_handles: CompMut<ElementHandle>,
67    mut entities: ResMutInit<Entities>,
68    assets: Res<AssetServer>,
69    mut jellyfishes: CompMut<Jellyfish>,
70    mut atlas_sprites: CompMut<AtlasSprite>,
71    mut spawners: CompMut<DehydrateOutOfBounds>,
72    mut items: CompMut<Item>,
73    mut item_grabs: CompMut<ItemGrab>,
74    mut item_throws: CompMut<ItemThrow>,
75    mut transforms: CompMut<Transform>,
76    mut bodies: CompMut<KinematicBody>,
77    mut spawner_manager: SpawnerManager,
78) {
79    let mut not_hydrated_bitset = hydrated.bitset().clone();
80    not_hydrated_bitset.bit_not();
81    not_hydrated_bitset.bit_and(element_handles.bitset());
82
83    let spawner_entities = entities
84        .iter_with_bitset(&not_hydrated_bitset)
85        .collect::<Vec<_>>();
86
87    for spawner_ent in spawner_entities {
88        let element_handle = *element_handles.get(spawner_ent).unwrap();
89        let element_meta = assets.get(element_handle.0);
90        let transform = *transforms.get(spawner_ent).unwrap();
91
92        if let Ok(JellyfishMeta {
93            atlas,
94            body_size,
95            throw_velocity,
96            angular_velocity,
97            fin_anim,
98            grab_offset,
99            max_ammo,
100            ..
101        }) = assets.get(element_meta.data).try_cast_ref()
102        {
103            hydrated.insert(spawner_ent, MapElementHydrated);
104
105            let entity = entities.create();
106            spawners.insert(entity, DehydrateOutOfBounds(spawner_ent));
107            hydrated.insert(entity, MapElementHydrated);
108            jellyfishes.insert(entity, Jellyfish::new(*max_ammo));
109            items.insert(entity, Item);
110            item_grabs.insert(
111                entity,
112                ItemGrab {
113                    fin_anim: *fin_anim,
114                    sync_animation: false,
115                    grab_offset: *grab_offset,
116                },
117            );
118            item_throws.insert(
119                entity,
120                ItemThrow::strength(*throw_velocity).with_spin(*angular_velocity),
121            );
122            element_handles.insert(entity, element_handle);
123            atlas_sprites.insert(entity, AtlasSprite::new(*atlas));
124            transforms.insert(entity, transform);
125            bodies.insert(
126                entity,
127                KinematicBody {
128                    shape: ColliderShape::Rectangle { size: *body_size },
129                    has_mass: true,
130                    has_friction: true,
131                    gravity: game_meta.core.physics.gravity,
132                    ..default()
133                },
134            );
135            spawner_manager.create_spawner(spawner_ent, vec![entity]);
136        }
137    }
138}

140/// A marker component for jellyfish items to indicate that it has been used and
141/// should dehydrate. This will also remove and despawn the jellyfish from the
142/// player's inventory.
143#[derive(Clone, Debug, Default, HasSchema)]
144pub struct DehydrateJellyfish {
145    pub owner: Entity,
146}
147
148fn dehydrate(
149    entities: Res<Entities>,
150    dehydrate_jellyfish: Comp<DehydrateJellyfish>,
151    spawners: Comp<DehydrateOutOfBounds>,
152    mut player_driving: CompMut<PlayerDrivingJellyfish>,
153    mut player_inventories: CompMut<Inventory>,
154    mut commands: Commands,
155    mut hydrated: CompMut<MapElementHydrated>,
156) {
157    for (jellyfish_ent, (dehydrate, spawner)) in
158        entities.iter_with((&dehydrate_jellyfish, &spawners))
159    {
160        if player_inventories
161            .get(dehydrate.owner)
162            .and_then(|inv| inv.0)
163            .filter(|item| *item == jellyfish_ent)
164            .is_some()
165        {
166            player_driving.remove(dehydrate.owner);
167            player_inventories.insert(dehydrate.owner, Inventory(None));
168        }
169        commands.add(move |mut entities: ResMut<Entities>| entities.kill(jellyfish_ent));
170        hydrated.remove(**spawner);
171    }
172}

174fn update_player_driving(
175    entities: Res<Entities>,
176    player_indexes: Comp<PlayerIdx>,
177    player_inventories: Comp<Inventory>,
178    player_states: Comp<PlayerState>,
179    mut jellyfishes: CompMut<Jellyfish>,
180    player_inputs: Res<MatchInputs>,
181    mut player_driving: CompMut<PlayerDrivingJellyfish>,
182    mut commands: Commands,
183) {
184    for (player_ent, (player_idx, player_inv, player_state)) in
185        entities.iter_with((&player_indexes, &player_inventories, &player_states))
186    {
187        let Some(jellyfish_ent) = player_inv.0 else {
188            continue;
189        };
190        let Some(jellyfish) = jellyfishes.get_mut(jellyfish_ent) else {
191            continue;
192        };
193
194        let player_control = player_inputs.players[player_idx.0 as usize].control;
195
196        if player_control.shoot_pressed {
197            if player_state.current != *idle::ID || player_driving.contains(player_ent) {
198                continue;
199            }
200            commands.add(flappy_jellyfish::spawn_or_take_control(
201                player_ent,
202                jellyfish_ent,
203                jellyfish.flappy,
204            ));
205        } else {
206            player_driving.remove(player_ent);
207        }
208    }
