50fn hydrate(
51    game_meta: Root<GameMeta>,
52    mut entities: ResMutInit<Entities>,
53    mut hydrated: CompMut<MapElementHydrated>,
54    mut element_handles: CompMut<ElementHandle>,
55    assets: Res<AssetServer>,
56    mut periscopes: CompMut<Periscope>,
57    mut atlas_sprites: CompMut<AtlasSprite>,
58    mut bodies: CompMut<KinematicBody>,
59    mut transforms: CompMut<Transform>,
60    mut items: CompMut<Item>,
61    mut item_throws: CompMut<ItemThrow>,
62    mut item_grabs: CompMut<ItemGrab>,
63    mut respawn_points: CompMut<DehydrateOutOfBounds>,
64    mut spawner_manager: SpawnerManager,
65) {
66    let mut not_hydrated_bitset = hydrated.bitset().clone();
67    not_hydrated_bitset.bit_not();
68    not_hydrated_bitset.bit_and(element_handles.bitset());
69
70    let spawner_entities = entities
71        .iter_with_bitset(&not_hydrated_bitset)
72        .collect::<Vec<_>>();
73
74    for spawner_ent in spawner_entities {
75        let transform = *transforms.get(spawner_ent).unwrap();
76        let element_handle = *element_handles.get(spawner_ent).unwrap();
77        let element_meta = assets.get(element_handle.0);
78
79        if let Ok(PeriscopeMeta {
80            atlas,
81            fin_anim,
82            grab_offset,
83            max_ammo,
84            body_size,
85            can_rotate,
86            bounciness,
87            throw_velocity,
88            angular_velocity,
89            ..
90        }) = assets.get(element_meta.data).try_cast_ref()
91        {
92            hydrated.insert(spawner_ent, MapElementHydrated);
93
94            let entity = entities.create();
95            items.insert(entity, Item);
96            item_throws.insert(
97                entity,
98                ItemThrow::strength(*throw_velocity)
99                    .with_spin(*angular_velocity)
100                    .with_system(periscope_drop(entity, *max_ammo)),
101            );
102            item_grabs.insert(
103                entity,
104                ItemGrab {
105                    fin_anim: *fin_anim,
106                    sync_animation: false,
107                    grab_offset: *grab_offset,
108                },
109            );
110            periscopes.insert(
111                entity,
112                Periscope {
113                    ammo: *max_ammo,
114                    cooldown: Timer::new(Duration::from_millis(0), TimerMode::Once),
115                },
116            );
117            atlas_sprites.insert(entity, AtlasSprite::new(*atlas));
118            respawn_points.insert(entity, DehydrateOutOfBounds(spawner_ent));
119            transforms.insert(entity, transform);
120            element_handles.insert(entity, element_handle);
121            hydrated.insert(entity, MapElementHydrated);
122            bodies.insert(
123                entity,
124                KinematicBody {
125                    shape: ColliderShape::Rectangle { size: *body_size },
126                    has_mass: true,
127                    has_friction: true,
128                    can_rotate: *can_rotate,
129                    bounciness: *bounciness,
130                    gravity: game_meta.core.physics.gravity,
131                    ..default()
132                },
133            );
134            spawner_manager.create_spawner(spawner_ent, vec![entity])
135        }
136    }
137}

139fn update(
140    entities: Res<Entities>,
141    mut commands: Commands,
142    element_handles: Comp<ElementHandle>,
143    assets: Res<AssetServer>,
144
145    mut periscopes: CompMut<Periscope>,
146    transforms: CompMut<Transform>,
147    mut sprites: CompMut<AtlasSprite>,
148    mut audio_center: ResMut<AudioCenter>,
149
150    player_inventories: PlayerInventories,
151    mut items_used: CompMut<ItemUsed>,
152    items_dropped: CompMut<ItemDropped>,
153    time: Res<Time>,
154
155    mut bodies: CompMut<KinematicBody>,
156) {
157    for (entity, (periscope, element_handle)) in
158        entities.iter_with((&mut periscopes, &element_handles))
159    {
160        let element_meta = assets.get(element_handle.0);
161
162        let asset = assets.get(element_meta.data);
163        let Ok(PeriscopeMeta {
164            max_ammo,
165            shoot_fps,
166            shoot_atlas,
167            shoot_frames,
168            shoot_lifetime,
169            cooldown,
170            bullet_meta,
171            bullet_spawn_offset,
172            shoot_sound,
173            empty_shoot_sound,
174            shoot_sound_volume,
175            empty_shoot_sound_volume,
176            kickback,
177            ..
178        }) = asset.try_cast_ref()
179        else {
180            unreachable!();
181        };
182
183        periscope.cooldown.tick(time.delta());
184
185        // If the item is being held
186        if let Some(Inv { player, .. }) = player_inventories.find_item(entity) {
187            // If the item is being used
188            let item_used = items_used.remove(entity).is_some();
189            if item_used && periscope.cooldown.finished() {
190                // Reset fire cooldown
191                periscope.cooldown = Timer::new(*cooldown, TimerMode::Once);
192                // Empty
193                if periscope.ammo.eq(&0) {
194                    audio_center.play_sound(*empty_shoot_sound, *empty_shoot_sound_volume);
195                    continue;
196                }
197
198                // Subtract ammo
199                periscope.ammo = periscope.ammo.saturating_sub(1).clamp(0, periscope.ammo);
200                audio_center.play_sound(*shoot_sound, *shoot_sound_volume);
201
202                let player_sprite = sprites.get_mut(player).unwrap();
203                let player_flip_x = player_sprite.flip_x;
204                let player_body = bodies.get_mut(player).unwrap();
205
206                //Set kickback
207                player_body.velocity.x = if player_flip_x { 1.0 } else { -1.0 } * kickback;
208
209                let mut shoot_animation_transform = *transforms.get(entity).unwrap();
210                let bullet_spawn_offset = *bullet_spawn_offset;
211                shoot_animation_transform.translation.z += 1.0;
212                shoot_animation_transform.translation.y += bullet_spawn_offset.y;
213                shoot_animation_transform.translation.x += if player_sprite.flip_x {
214                    -bullet_spawn_offset.x
215                } else {
216                    bullet_spawn_offset.x
217                };
218
219                let shoot_fps = *shoot_fps;
220                let shoot_frames = *shoot_frames;
221                let shoot_lifetime = *shoot_lifetime;
222                let shoot_atlas = *shoot_atlas;
223
224                let bullet_meta = *bullet_meta;
225
226                commands.add(
227                    move |mut entities: ResMutInit<Entities>,
228                          mut lifetimes: CompMut<Lifetime>,
229                          mut sprites: CompMut<AtlasSprite>,
230                          mut transforms: CompMut<Transform>,
231                          mut bullets: CompMut<Bullet>,
232                          mut bullet_handles: CompMut<BulletHandle>,
233                          mut animated_sprites: CompMut<AnimatedSprite>| {
234                        // spawn fire animation
235                        {
236                            let ent = entities.create();
237                            transforms.insert(ent, shoot_animation_transform);
238                            sprites.insert(
239                                ent,
240                                AtlasSprite {
241                                    flip_x: player_flip_x,
242                                    atlas: shoot_atlas,
243                                    ..default()
244                                },
245                            );
246
247                            animated_sprites.insert(
248                                ent,
249                                AnimatedSprite {
250                                    frames: (0..shoot_frames).collect(),
251                                    fps: shoot_fps,
252                                    repeat: false,
253                                    ..default()
254                                },
255                            );
256                            lifetimes.insert(ent, Lifetime::new(shoot_lifetime));
257                        }
258
259                        // spawn bullet
260                        {
261                            let ent = entities.create();
262                            bullets.insert(
263                                ent,
264                                Bullet {
265                                    owner: player,
266                                    direction: if player_flip_x {
267                                        vec2(-1.0, 0.0)
268                                    } else {
269                                        vec2(1.0, 0.0)
270                                    },
271                                },
272                            );
273                            transforms.insert(ent, shoot_animation_transform);
274                            bullet_handles.insert(ent, BulletHandle(bullet_meta));
275                        }
276                    },
277                );
278            }
279        }
280
281        // If the item was dropped
282        if items_dropped.get(entity).is_some() {
283            // reload gun
284            periscope.ammo = *max_ammo;
285        }
286    }
287}

289fn periscope_drop(entity: Entity, max_ammo: u32) -> StaticSystem<(), ()> {
290    (move |mut periscopes: CompMut<Periscope>| {
291        periscopes.get_mut(entity).unwrap().ammo = max_ammo;
292    })
293    .system()
