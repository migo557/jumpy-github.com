31fn hydrate(
32    entities: Res<Entities>,
33    mut hydrated: CompMut<MapElementHydrated>,
34    element_handles: Comp<ElementHandle>,
35    assets: Res<AssetServer>,
36    mut player_spawners: CompMut<PlayerSpawner>,
37    mut spawner_manager: SpawnerManager,
38) {
39    let mut not_hydrated_bitset = hydrated.bitset().clone();
40    not_hydrated_bitset.bit_not();
41    not_hydrated_bitset.bit_and(element_handles.bitset());
42
43    for entity in entities.iter_with_bitset(&not_hydrated_bitset) {
44        let element_handle = element_handles.get(entity).unwrap();
45        let element_meta = assets.get(element_handle.0);
46
47        if let Ok(PlayerSpawner) = assets.get(element_meta.data).try_cast_ref() {
48            hydrated.insert(entity, MapElementHydrated);
49            player_spawners.insert(entity, PlayerSpawner);
50
51            spawner_manager.create_grouped_spawner(entity, vec![], &player_spawners, &entities);
52        }
53    }
54}

56fn update(
57    mut entities: ResMutInit<Entities>,
58    mut current_spawner: ResMutInit<CurrentSpawner>,
59    player_spawners: Comp<PlayerSpawner>,
60    mut player_indexes: CompMut<PlayerIdx>,
61    mut transforms: CompMut<Transform>,
62    player_inputs: Res<MatchInputs>,
63    mut spawner_manager: SpawnerManager,
64) {
65    let alive_players = entities
66        .iter_with(&player_indexes)
67        .map(|(_ent, pidx)| pidx.0)
68        .collect::<Vec<_>>();
69    let spawn_points = entities
70        .iter_with((&player_spawners, &transforms))
71        .map(|(_ent, (_spawner, transform))| transform.translation)
72        .collect::<Vec<_>>();
73
74    // For every player
75    for i in 0..MAX_PLAYERS {
76        let player = &player_inputs.players[i as usize];
77
78        // If the player is active, but not alive
79        if player.active && !alive_players.contains(&i) {
80            // Increment the spawner index
81            current_spawner.0 += 1;
82            current_spawner.0 %= spawn_points.len().max(1);
83
84            let Some(mut spawn_point) = spawn_points.get(current_spawner.0).copied() else {
85                return;
86            };
87
88            // Make sure each player spawns at a different z level ( give enough room for 10 players
89            // to fit between map layers )
90            spawn_point.z += i as f32 * MAP_LAYERS_GAP_DEPTH / 10.0;
91
92            let player_ent = entities.create();
93            player_indexes.insert(player_ent, PlayerIdx(i));
94            transforms.insert(player_ent, Transform::from_translation(spawn_point));
95
96            spawner_manager.insert_spawned_entity_into_grouped_spawner(
97                player_ent,
98                &player_spawners,
99                &entities,
100            );
101        }
102    }
