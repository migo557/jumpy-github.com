4#[type_data(metadata_asset("fish_school"))]
5#[repr(C)]
6pub struct FishSchoolMeta {
7    pub kinds: SVec<Handle<Atlas>>,
8    /// The default and most-likely to ocurr number of fish in a school
9    pub base_count: u32,
10    /// The ammount greater or less than the base number of fish that may spawn
11    pub count_variation: u32,
12    /// The distance from the spawn point on each axis that the individual fish in the school will be
13    /// initially spawned within
14    pub spawn_range: f32,
15    /// The distance that the fish wish to stay within the center of their school
16    pub school_size: f32,
17    // The distance a collider must be for the fish to run away
18    pub flee_range: f32,
19}

51pub fn hydrate(
52    rng: Res<GlobalRng>,
53    mut entities: ResMutInit<Entities>,
54    mut hydrated: CompMut<MapElementHydrated>,
55    mut element_handles: CompMut<ElementHandle>,
56    assets: Res<AssetServer>,
57    mut fish_schools: CompMut<FishSchool>,
58    mut fishes: CompMut<Fish>,
59    mut transforms: CompMut<Transform>,
60    mut actors: CompMut<Actor>,
61    mut colliders: CompMut<Collider>,
62    mut atlas_sprites: CompMut<AtlasSprite>,
63    mut animated_sprites: CompMut<AnimatedSprite>,
64    mut spawner_manager: SpawnerManager,
65) {
66    let mut not_hydrated_bitset = hydrated.bitset().clone();
67    not_hydrated_bitset.bit_not();
68    not_hydrated_bitset.bit_and(element_handles.bitset());
69
70    let spawner_entities = entities
71        .iter_with_bitset(&not_hydrated_bitset)
72        .collect::<Vec<_>>();
73
74    for spawner_ent in spawner_entities {
75        let transform = *transforms.get(spawner_ent).unwrap();
76        let element_handle = *element_handles.get(spawner_ent).unwrap();
77        let element_meta = assets.get(element_handle.0);
78        if let Ok(FishSchoolMeta {
79            kinds,
80            base_count,
81            count_variation,
82            spawn_range,
83            flee_range,
84            ..
85        }) = assets.get(element_meta.data).try_cast_ref()
86        {
87            // We can use the spawner entity itself for our
88            // school because there is no respawning.
89            let fish_school_ent = spawner_ent;
90
91            hydrated.insert(fish_school_ent, MapElementHydrated);
92            transforms.insert(fish_school_ent, transform);
93            element_handles.insert(fish_school_ent, element_handle);
94
95            let mut fish_ents = Vec::new();
96
97            let rand_bool = || rng.u8(0u8..2) == 0;
98
99            let mut fish_count = *base_count;
100            if rand_bool() {
101                for _ in 0..*count_variation {
102                    if rand_bool() {
103                        fish_count = if rand_bool() {
104                            fish_count.saturating_sub(1)
105                        } else {
106                            fish_count.saturating_add(1)
107                        };
108                    }
109                }
110            }
111
112            let spawn_point = transform.translation.xy();
113
114            for _ in 0..fish_count {
115                let spawn_point = spawn_point
116                    + vec2(
117                        rng.f32_normalized() * spawn_range,
118                        rng.f32_normalized() * spawn_range,
119                    );
120
121                let duration = {
122                    let min = 0.2;
123                    let max = 1.0;
124                    rng.f32() * (max - min) + min
125                };
126
127                let atlas = &kinds[rng.usize(0..kinds.len())];
128
129                let fish_ent = entities.create();
130                fishes.insert(
131                    fish_ent,
132                    Fish {
133                        state: FishState::Moving {
134                            from: spawn_point,
135                            to: spawn_point,
136                        },
137                        state_timer: Timer::new(
138                            Duration::from_secs_f32(duration),
139                            TimerMode::Repeating,
140                        ),
141                    },
142                );
143                transforms.insert(fish_ent, transform);
144                actors.insert(fish_ent, Actor);
145                colliders.insert(
146                    fish_ent,
147                    Collider {
148                        shape: ColliderShape::Circle {
149                            diameter: flee_range * 2.0,
150                        },
151                        ..default()
152                    },
153                );
154                atlas_sprites.insert(fish_ent, AtlasSprite::new(*atlas));
155                animated_sprites.insert(
156                    fish_ent,
157                    AnimatedSprite {
158                        frames: [0, 1, 2, 3].into_iter().collect(),
159                        fps: 6.0,
160                        repeat: true,
161                        ..default()
162                    },
163                );
164
165                fish_ents.push(fish_ent);
166            }
167            fish_schools.insert(
168                fish_school_ent,
169                FishSchool {
170                    fish: fish_ents.clone(),
171                },
172            );
173            spawner_manager.create_spawner(fish_school_ent, fish_ents);
174        }
175    }
176}

178pub fn update_fish_schools(
179    rng: Res<GlobalRng>,
180    time: Res<Time>,
181    entities: Res<Entities>,
182    fish_schools: Comp<FishSchool>,
183    mut fishes: CompMut<Fish>,
184    element_handles: CompMut<ElementHandle>,
185    assets: Res<AssetServer>,
186    mut atlas_sprites: CompMut<AtlasSprite>,
187    mut transforms: CompMut<Transform>,
188    collision_world: CollisionWorld,
189    bodies: Comp<KinematicBody>,
190) {
191    for (school_ent, school) in entities.iter_with(&fish_schools) {
192        let element_handle = element_handles.get(school_ent).unwrap();
193        let element_meta = assets.get(element_handle.0);
194        let asset = assets.get(element_meta.data);
195        let Ok(FishSchoolMeta { school_size, .. }) = asset.try_cast_ref() else {
196            continue;
197        };
198
199        let transform = transforms.get(school_ent).unwrap();
200        let mut fish_transforms = school
201            .fish
202            .iter()
203            .map(|entity| transforms.get(*entity).unwrap());
204        let Some(fish_transform) = fish_transforms.next() else {
205            continue;
206        };
207
208        let mut school_bounds_min = fish_transform.translation.xy();
209        let mut school_bounds_max = fish_transform.translation.xy();
210
211        for transform in fish_transforms {
212            let pos = &transform.translation.xy();
213
214            school_bounds_min.x = school_bounds_min.x.min(pos.x);
215            school_bounds_min.y = school_bounds_min.y.min(pos.y);
216            school_bounds_max.x = school_bounds_max.x.max(pos.x);
217            school_bounds_max.y = school_bounds_max.y.max(pos.y);
218        }
219
220        let size = school_bounds_max - school_bounds_min;
221        let center = school_bounds_min + size / 2.0;
222        let is_grouped = size.x.max(size.y) < *school_size;
223        let spawn_pos = transform.translation.xy();
224
225        for fish_ent in school.fish.iter() {
226            let flee = collision_world
227                .actor_collisions(*fish_ent)
228                .into_iter()
229                .find(|ent| {
230                    !fishes.contains(*ent)
231                        && bodies
232                            .get(*ent)
233                            .map(|x| x.velocity != Vec2::ZERO)
234                            .unwrap_or_default()
235                });
236
237            let fish = fishes.get_mut(*fish_ent).unwrap();
238            let sprite = atlas_sprites.get_mut(*fish_ent).unwrap();
239
240            let mut pos = transforms.get(*fish_ent).unwrap().translation.xy();
241
242            let rand_bool = || rng.u8(0u8..2) == 0;
243            let rand_range = |min: f32, max: f32| rng.f32() * (max - min) + min;
244
245            let pick_next_move = || {
246                if !is_grouped {
247                    let target_point = pos.lerp(center, rand_range(0.1, 0.4));
248
249                    (
250                        FishState::Moving {
251                            from: pos,
252                            to: target_point,
253                        },
254                        Timer::new(
255                            Duration::from_secs_f32(rand_range(0.2, 0.7)),
256                            TimerMode::Repeating,
257                        ),
258                    )
259                } else if rand_bool() {
260                    let target_point = vec2(
261                        pos.x + rand_range(-20.0, 20.0),
262                        pos.y + rand_range(-20.0, 20.0),
263                    );
264                    (
265                        FishState::Moving {
266                            from: pos,
267                            to: target_point,
268                        },
269                        Timer::new(
270                            Duration::from_secs_f32(rand_range(0.5, 1.5)),
271                            TimerMode::Repeating,
272                        ),
273                    )
274                } else {
275                    let target_point = pos.lerp(spawn_pos, rand_range(0.10, 0.25));
276                    (
277                        FishState::Moving {
278                            from: pos,
279                            to: target_point,
280                        },
281                        Timer::new(
282                            Duration::from_secs_f32(rand_range(0.5, 1.5)),
283                            TimerMode::Repeating,
284                        ),
285                    )
286                }
287            };
288
289            fish.state_timer.tick(time.delta());
290
291            if let Some(scary_thing) = flee {
292                let diff = pos - transforms.get(scary_thing).unwrap().translation.xy();
293                fish.state = FishState::Moving {
294                    from: pos,
295                    to: pos + diff.normalize() * rand_range(30.0, 60.0),
296                };
297                fish.state_timer = Timer::new(
298                    Duration::from_secs_f32(rand_range(0.2, 0.6)),
299                    TimerMode::Repeating,
300                );
301                // We tick the timer an extra time here to make sure that the fish gets moving
302                // immediately without waiting for an extra frame, because if we keep colliding we
303                // may just keep re-setting the timer and the fish get's stuck until it stops
304                // colliding.
305                fish.state_timer.tick(time.delta());
306            } else if fish.state_timer.finished() {
307                let (state, timer) = pick_next_move();
308                fish.state = state;
309                fish.state_timer = timer;
310            }
311
312            match &fish.state {
313                FishState::Moving { from, to } => {
314                    let lerp_progress = Ease {
315                        ease_in: false,
316                        ease_out: true,
317                        function: EaseFunction::Sinusoidial,
318                        progress: fish.state_timer.percent(),
319                    }
320                    .output();
321
322                    sprite.flip_x = from.x > to.x;
323                    sprite.index = (4.0 * lerp_progress).floor() as u32;
324
325                    pos = from.lerp(*to, lerp_progress);
326                }
327            }
328            let transform = transforms.get_mut(*fish_ent).unwrap();
329            transform.translation.x = pos.x;
330            transform.translation.y = pos.y;
331        }
332    }
