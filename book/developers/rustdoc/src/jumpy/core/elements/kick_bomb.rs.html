69impl KickBombCommand {
70    /// Command for spawning  a kick bomb.
71    /// If entity is provided, components are added to this entity, otherwise command spawns the entity
72    ///
73    /// `kick_bomb_handle` must cast to `Handle<KickBombMeta>` or `Handle<ElementMeta>` where [`ElementMeta`]
74    /// contains handle that casts to `Handle<KickBombMeta>`.
75    /// [`Handle::untyped`] should be used to convert to [`UntypedHandle`].
76    #[must_use]
77    pub fn spawn_kick_bomb(
78        entity: Option<Entity>,
79        transform: Transform,
80        kick_bomb_meta_handle: UntypedHandle,
81        lit: bool,
82        player_flip_f: Option<bool>,
83    ) -> StaticSystem<(), ()> {
84        (move |game_meta: Root<GameMeta>,
85               assets: Res<AssetServer>,
86               mut animated_sprites: CompMut<AnimatedSprite>,
87               mut atlas_sprites: CompMut<AtlasSprite>,
88               mut bodies: CompMut<KinematicBody>,
89               mut entities: ResMutInit<Entities>,
90               mut idle_bombs: CompMut<IdleKickBomb>,
91               mut lit_bombs: CompMut<LitKickBomb>,
92               mut items: CompMut<Item>,
93               mut item_throws: CompMut<ItemThrow>,
94               mut item_grabs: CompMut<ItemGrab>,
95               mut kick_bomb_handles: CompMut<KickBombHandle>,
96               mut transforms: CompMut<Transform>| {
97            // Unwrap entity or spawn if existing entity was not provided.
98            let entity = entity.unwrap_or_else(|| entities.create());
99
100            // Try to use handle as Handle<KickBombMeta>.
101            let kick_bomb_meta_handle =
102                match try_cast_meta_handle::<KickBombMeta>(kick_bomb_meta_handle, &assets) {
103                    Ok(handle) => handle,
104                    Err(err) => {
105                        error!("KickBombCommand::spawn_kick_bomb() failed: {err}");
106                        return;
107                    }
108                };
109
110            let KickBombMeta {
111                atlas,
112                fin_anim,
113                grab_offset,
114                body_diameter,
115                can_rotate,
116                bounciness,
117                throw_velocity,
118                angular_velocity,
119                arm_delay,
120                fuse_time,
121                ..
122            } = *assets.get(kick_bomb_meta_handle);
123
124            kick_bomb_handles.insert(entity, KickBombHandle(kick_bomb_meta_handle));
125            items.insert(entity, Item);
126            item_throws.insert(
127                entity,
128                ItemThrow::strength(throw_velocity).with_spin(angular_velocity),
129            );
130            item_grabs.insert(
131                entity,
132                ItemGrab {
133                    fin_anim,
134                    sync_animation: false,
135                    grab_offset,
136                },
137            );
138            atlas_sprites.insert(entity, AtlasSprite::new(atlas));
139            transforms.insert(entity, transform);
140            animated_sprites.insert(entity, default());
141            bodies.insert(
142                entity,
143                KinematicBody {
144                    shape: ColliderShape::Circle {
145                        diameter: body_diameter,
146                    },
147                    gravity: game_meta.core.physics.gravity,
148                    has_mass: true,
149                    has_friction: true,
150                    can_rotate,
151                    bounciness,
152                    ..default()
153                },
154            );
155
156            if lit {
157                lit_bombs.insert(
158                    entity,
159                    LitKickBomb {
160                        arm_delay: Timer::new(arm_delay, TimerMode::Once),
161                        fuse_time: Timer::new(fuse_time, TimerMode::Once),
162                        kicking: false,
163                        kicks: 0,
164                    },
165                );
166
167                if let Some(body) = bodies.get_mut(entity) {
168                    let horizontal_flip_factor = if player_flip_f.unwrap() {
169                        Vec2::new(-1.0, 1.0)
170                    } else {
171                        Vec2::ONE
172                    };
173
174                    body.velocity = Vec2::new(
175                        horizontal_flip_factor.x * throw_velocity,
176                        horizontal_flip_factor.y * throw_velocity / 2.5,
177                    );
178                    body.angular_velocity = angular_velocity;
179                }
180            } else {
181                idle_bombs.insert(entity, IdleKickBomb);
182            }
183        })
184        .system()
185    }
186}

188fn hydrate(
189    assets: Res<AssetServer>,
190    mut entities: ResMutInit<Entities>,
191    transforms: Comp<Transform>,
192    mut hydrated: CompMut<MapElementHydrated>,
193    mut element_handles: CompMut<ElementHandle>,
194    mut respawn_points: CompMut<DehydrateOutOfBounds>,
195    mut spawner_manager: SpawnerManager,
196    mut commands: Commands,
197) {
198    let mut not_hydrated_bitset = hydrated.bitset().clone();
199    not_hydrated_bitset.bit_not();
200    not_hydrated_bitset.bit_and(element_handles.bitset());
201
202    let spawner_entities = entities
203        .iter_with_bitset(&not_hydrated_bitset)
204        .collect::<Vec<_>>();
205
206    for spawner_ent in spawner_entities {
207        let transform = *transforms.get(spawner_ent).unwrap();
208        let element_handle = *element_handles.get(spawner_ent).unwrap();
209
210        // Check if spawner element handle is for kick bomb
211        let element_meta = assets.get(element_handle.0);
212        if assets
213            .get(element_meta.data)
214            .try_cast_ref::<KickBombMeta>()
215            .is_ok()
216        {
217            hydrated.insert(spawner_ent, MapElementHydrated);
218
219            let entity = entities.create();
220            hydrated.insert(entity, MapElementHydrated);
221            element_handles.insert(entity, element_handle);
222            respawn_points.insert(entity, DehydrateOutOfBounds(spawner_ent));
223            spawner_manager.create_spawner(spawner_ent, vec![entity]);
224
225            commands.add(KickBombCommand::spawn_kick_bomb(
226                Some(entity),
227                transform,
228                element_meta.data.untyped(),
229                false,
230                None,
231            ));
232        }
233    }
234}

236fn update_idle_kick_bombs(
237    entities: Res<Entities>,
238    mut commands: Commands,
239    mut items_used: CompMut<ItemUsed>,
240    mut audio_center: ResMut<AudioCenter>,
241    kick_bomb_handles: Comp<KickBombHandle>,
242    mut idle_bombs: CompMut<IdleKickBomb>,
243    assets: Res<AssetServer>,
244    mut animated_sprites: CompMut<AnimatedSprite>,
245) {
246    for (entity, (_kick_bomb, kick_bomb_handle)) in
247        entities.iter_with((&mut idle_bombs, &kick_bomb_handles))
248    {
249        let kick_bomb_meta = assets.get(kick_bomb_handle.0);
250
251        let KickBombMeta {
252            fuse_sound,
253            fuse_sound_volume,
254            arm_delay,
255            fuse_time,
256            lit_frames_start,
257            lit_frames_end,
258            lit_fps,
259            ..
260        } = *kick_bomb_meta;
261
262        if items_used.remove(entity).is_some() {
263            audio_center.play_sound(fuse_sound, fuse_sound_volume);
264            let animated_sprite = animated_sprites.get_mut(entity).unwrap();
265            animated_sprite.frames = (lit_frames_start..lit_frames_end).collect();
266            animated_sprite.repeat = true;
267            animated_sprite.fps = lit_fps;
268            commands.add(
269                move |mut idle: CompMut<IdleKickBomb>, mut lit: CompMut<LitKickBomb>| {
270                    idle.remove(entity);
271                    lit.insert(
272                        entity,
273                        LitKickBomb {
274                            arm_delay: Timer::new(arm_delay, TimerMode::Once),
275                            fuse_time: Timer::new(fuse_time, TimerMode::Once),
276                            kicking: false,
277                            kicks: 0,
278                        },
279                    );
280                },
281            );
282        }
283    }
284}

286fn update_lit_kick_bombs(
287    entities: Res<Entities>,
288    kick_bomb_handles: Comp<KickBombHandle>,
289    assets: Res<AssetServer>,
290    collision_world: CollisionWorld,
291    player_indexes: Comp<PlayerIdx>,
292    mut audio_center: ResMut<AudioCenter>,
293    mut trauma_events: ResMutInit<CameraTraumaEvents>,
294    mut lit_grenades: CompMut<LitKickBomb>,
295    mut sprites: CompMut<AtlasSprite>,
296    mut bodies: CompMut<KinematicBody>,
297    mut hydrated: CompMut<MapElementHydrated>,
298    player_inventories: PlayerInventories,
299    mut transforms: CompMut<Transform>,
300    mut commands: Commands,
301    time: Res<Time>,
302    spawners: Comp<DehydrateOutOfBounds>,
303    invincibles: CompMut<Invincibility>,
304) {
305    for (entity, (kick_bomb, kick_bomb_handle, spawner)) in
306        entities.iter_with((&mut lit_grenades, &kick_bomb_handles, &Optional(&spawners)))
307    {
308        let kick_bomb_meta = assets.get(kick_bomb_handle.0);
309        let KickBombMeta {
310            explosion_sound,
311            explosion_volume,
312            explode_on_contact,
313            kick_velocity,
314            kickable,
315            damage_region_lifetime,
316            damage_region_size,
317            explosion_lifetime,
318            explosion_atlas,
319            explosion_fps,
320            explosion_frames,
321            ..
322        } = *kick_bomb_meta;
323
324        kick_bomb.fuse_time.tick(time.delta());
325        kick_bomb.arm_delay.tick(time.delta());
326
327        let should_explode = 'should_explode: {
328            if kick_bomb.fuse_time.finished() {
329                break 'should_explode true;
330            }
331
332            if explode_on_contact {
333                let players = entities
334                    .iter_with(&player_indexes)
335                    .map(|x| x.0)
336                    .collect::<Vec<_>>();
337
338                let colliding_with_players = collision_world
339                    .actor_collisions_filtered(entity, |e| {
340                        players.contains(&e) && invincibles.get(e).is_none()
341                    })
342                    .into_iter()
343                    .collect::<Vec<_>>();
344
345                if !colliding_with_players.is_empty() && kick_bomb.arm_delay.finished() {
346                    break 'should_explode true;
347                }
348            }
349
350            // If the item is being held
351            if player_inventories.find_item(entity).is_some() {
352                kick_bomb.kicking = false;
353                break 'should_explode false;
354            }
355
356            if kickable {
357                // If the item is colliding with a non-invincible player
358                if let Some(player_entity) = collision_world
359                    .actor_collisions_filtered(entity, |e| !invincibles.contains(e))
360                    .into_iter()
361                    .find(|&x| player_indexes.contains(x))
362                {
363                    if !std::mem::replace(&mut kick_bomb.kicking, true) {
364                        kick_bomb.kicks += 1;
365                    }
366
367                    // Explode on the 3rd kick.
368                    // Dropping the bomb is detected as a kick so we explode when
369                    // the counter reaches 4.
370                    if kick_bomb.kicks > 3 {
371                        break 'should_explode true;
372                    }
373
374                    let body = bodies.get_mut(entity).unwrap();
375                    let translation = transforms.get_mut(entity).unwrap().translation;
376
377                    let player_sprite = sprites.get_mut(player_entity).unwrap();
378                    let player_translation = transforms.get(player_entity).unwrap().translation;
379
380                    let player_standing_left = player_translation.x <= translation.x;
381
382                    if body.velocity.x == 0.0 {
383                        body.velocity = kick_velocity;
384                        if player_sprite.flip_x {
385                            body.velocity.x *= -1.0;
386                        }
387                    } else if player_standing_left && !player_sprite.flip_x {
388                        body.velocity.x = kick_velocity.x;
389                        body.velocity.y = kick_velocity.y;
390                    } else if !player_standing_left && player_sprite.flip_x {
391                        body.velocity.x = -kick_velocity.x;
392                        body.velocity.y = kick_velocity.y;
393                    } else if kick_bomb.arm_delay.finished() {
394                        break 'should_explode true;
395                    }
396                } else {
397                    kick_bomb.kicking = false;
398                }
399            }
400            false
401        };
402
403        // If it's time to explode
404        if should_explode {
405            audio_center.play_sound(explosion_sound, explosion_volume);
406
407            trauma_events.send(7.5);
408
409            if let Some(spawner) = spawner {
410                // Cause the item to respawn by un-hydrating it's spawner.
411                hydrated.remove(**spawner);
412            }
413
414            let mut explosion_transform = *transforms.get(entity).unwrap();
415            explosion_transform.translation.z = -10.0; // On top of almost everything
416            explosion_transform.rotation = Quat::IDENTITY;
417
418            commands.add(
419                move |mut entities: ResMutInit<Entities>,
420                      mut transforms: CompMut<Transform>,
421                      mut damage_regions: CompMut<DamageRegion>,
422                      mut lifetimes: CompMut<Lifetime>,
423                      mut sprites: CompMut<AtlasSprite>,
424                      mut animated_sprites: CompMut<AnimatedSprite>| {
425                    // Despawn the kick bomb
426                    entities.kill(entity);
427
428                    // Spawn the damage region
429                    let ent = entities.create();
430                    transforms.insert(ent, explosion_transform);
431                    damage_regions.insert(
432                        ent,
433                        DamageRegion {
434                            size: damage_region_size,
435                        },
436                    );
437                    lifetimes.insert(ent, Lifetime::new(damage_region_lifetime));
438
439                    // Spawn the explosion animation
440                    let ent = entities.create();
441                    transforms.insert(ent, explosion_transform);
442                    sprites.insert(
443                        ent,
444                        AtlasSprite {
445                            atlas: explosion_atlas,
446                            ..default()
447                        },
448                    );
449                    animated_sprites.insert(
450                        ent,
451                        AnimatedSprite {
452                            frames: (0..explosion_frames).collect(),
453                            fps: explosion_fps,
454                            repeat: false,
455                            ..default()
456                        },
457                    );
458                    lifetimes.insert(ent, Lifetime::new(explosion_lifetime));
459                },
460            );
461        }
462    }
