55fn hydrate_crates(
56    game_meta: Root<GameMeta>,
57    mut entities: ResMutInit<Entities>,
58    mut hydrated: CompMut<MapElementHydrated>,
59    mut element_handles: CompMut<ElementHandle>,
60    assets: Res<AssetServer>,
61    mut idle_crates: CompMut<IdleCrate>,
62    mut atlas_sprites: CompMut<AtlasSprite>,
63    mut animated_sprites: CompMut<AnimatedSprite>,
64    mut bodies: CompMut<KinematicBody>,
65    mut transforms: CompMut<Transform>,
66    mut items: CompMut<Item>,
67    mut item_throws: CompMut<ItemThrow>,
68    mut item_grabs: CompMut<ItemGrab>,
69    mut respawn_points: CompMut<DehydrateOutOfBounds>,
70    mut spawner_manager: SpawnerManager,
71) {
72    let mut not_hydrated_bitset = hydrated.bitset().clone();
73    not_hydrated_bitset.bit_not();
74    not_hydrated_bitset.bit_and(element_handles.bitset());
75
76    let spawner_entities = entities
77        .iter_with_bitset(&not_hydrated_bitset)
78        .collect::<Vec<_>>();
79
80    for spawner_entity in spawner_entities {
81        let transform = *transforms.get(spawner_entity).unwrap();
82        let element_handle = *element_handles.get(spawner_entity).unwrap();
83        let element_meta = assets.get(element_handle.0);
84
85        let asset = assets.get(element_meta.data);
86        let Ok(CrateMeta {
87            atlas,
88            fin_anim,
89            grab_offset,
90            body_size,
91            throw_velocity,
92            bounciness,
93            ..
94        }) = asset.try_cast_ref()
95        else {
96            continue;
97        };
98
99        hydrated.insert(spawner_entity, MapElementHydrated);
100
101        let entity = entities.create();
102        items.insert(entity, Item);
103        idle_crates.insert(entity, IdleCrate);
104        item_throws.insert(entity, ItemThrow::strength(*throw_velocity));
105        item_grabs.insert(
106            entity,
107            ItemGrab {
108                fin_anim: *fin_anim,
109                sync_animation: false,
110                grab_offset: *grab_offset,
111            },
112        );
113        atlas_sprites.insert(entity, AtlasSprite::new(*atlas));
114        respawn_points.insert(entity, DehydrateOutOfBounds(spawner_entity));
115        transforms.insert(entity, transform);
116        element_handles.insert(entity, element_handle);
117        hydrated.insert(entity, MapElementHydrated);
118        animated_sprites.insert(entity, default());
119        bodies.insert(
120            entity,
121            KinematicBody {
122                shape: ColliderShape::Rectangle { size: *body_size },
123                has_mass: true,
124                can_rotate: false,
125                has_friction: true,
126                gravity: game_meta.core.physics.gravity,
127                bounciness: *bounciness,
128                ..default()
129            },
130        );
131        spawner_manager.create_spawner(spawner_entity, vec![entity])
132    }
133}

135fn update_idle_crates(
136    entities: Res<Entities>,
137    mut items_used: CompMut<ItemUsed>,
138    assets: Res<AssetServer>,
139    element_handles: Comp<ElementHandle>,
140    idle_crates: CompMut<IdleCrate>,
141    player_inventories: PlayerInventories,
142    mut commands: Commands,
143) {
144    for (entity, (_le_crate, element_handle)) in
145        entities.iter_with((&idle_crates, &element_handles))
146    {
147        let element_meta = assets.get(element_handle.0);
148
149        let asset = assets.get(element_meta.data);
150        let Ok(CrateMeta { break_timeout, .. }) = asset.try_cast_ref() else {
151            continue;
152        };
153
154        let break_timeout = *break_timeout;
155
156        if let Some(Inv { player, .. }) = player_inventories.find_item(entity) {
157            if items_used.remove(entity).is_some() {
158                commands.add(PlayerCommand::set_inventory(player, None));
159                commands.add(
160                    move |mut idle: CompMut<IdleCrate>, mut thrown: CompMut<ThrownCrate>| {
161                        idle.remove(entity);
162                        thrown.insert(
163                            entity,
164                            ThrownCrate {
165                                owner: player,
166                                damage_delay: Timer::new(
167                                    Duration::from_secs_f32(0.25),
168                                    TimerMode::Once,
169                                ),
170                                break_timeout: Timer::new(break_timeout, TimerMode::Once),
171                                was_colliding: false,
172                                crate_break_state: 0,
173                            },
174                        );
175                    },
176                );
177            }
178        }
179    }
180}

182fn update_thrown_crates(
183    entities: Res<Entities>,
184    mut hydrated: CompMut<MapElementHydrated>,
185    assets: Res<AssetServer>,
186    element_handles: Comp<ElementHandle>,
187    mut thrown_crates: CompMut<ThrownCrate>,
188    mut commands: Commands,
189    mut atlas_sprites: CompMut<AtlasSprite>,
190    players: Comp<PlayerIdx>,
191    collision_world: CollisionWorld,
192    mut bodies: CompMut<KinematicBody>,
193    mut audio_center: ResMut<AudioCenter>,
194    transforms: Comp<Transform>,
195    spawners: Comp<DehydrateOutOfBounds>,
196    invincibles: CompMut<Invincibility>,
197    time: Res<Time>,
198) {
199    for (entity, (thrown_crate, element_handle, transform, atlas_sprite, body, spawner)) in entities
200        .iter_with((
201            &mut thrown_crates,
202            &element_handles,
203            &transforms,
204            &mut atlas_sprites,
205            &mut bodies,
206            &spawners,
207        ))
208    {
209        let element_meta = assets.get(element_handle.0);
210
211        let asset = assets.get(element_meta.data);
212        let Ok(CrateMeta {
213            breaking_anim_frames,
214            breaking_atlas,
215            breaking_anim_fps,
216            break_sound,
217            break_sound_volume,
218            bounce_sound,
219            bounce_sound_volume,
220            crate_break_state_1,
221            crate_break_state_2,
222            ..
223        }) = asset.try_cast_ref()
224        else {
225            continue;
226        };
227
228        thrown_crate.damage_delay.tick(time.delta());
229        thrown_crate.break_timeout.tick(time.delta());
230
231        let colliding_with_tile = {
232            let collider = collision_world.get_collider(entity);
233            let aabb = collider.shape.compute_aabb(default());
234            let width = aabb.extents().x + 2.0;
235            let height = aabb.extents().y + 2.0;
236            collision_world.tile_collision(
237                Transform::from_translation(transform.translation),
238                ColliderShape::Rectangle {
239                    size: vec2(width, height),
240                },
241            ) != TileCollisionKind::Empty
242        };
243
244        if colliding_with_tile && !thrown_crate.was_colliding {
245            thrown_crate.was_colliding = true;
246            thrown_crate.crate_break_state += 1;
247            audio_center.play_sound(*bounce_sound, *bounce_sound_volume);
248        } else if !colliding_with_tile {
249            thrown_crate.was_colliding = false;
250        }
251
252        match thrown_crate.crate_break_state {
253            1 => {
254                atlas_sprite.atlas = *breaking_atlas;
255                atlas_sprite.index = *crate_break_state_1;
256            }
257            3 => {
258                atlas_sprite.index = *crate_break_state_2;
259            }
260            _ => {}
261        }
262
263        let colliding_with_players = collision_world
264            .actor_collisions_filtered(entity, |e| {
265                players.contains(e)
266                    && invincibles.get(e).is_none()
267                    && thrown_crate.damage_delay.finished()
268            })
269            .into_iter()
270            .collect::<Vec<_>>();
271
272        for player_entity in &colliding_with_players {
273            commands.add(PlayerCommand::kill(
274                *player_entity,
275                Some(transform.translation.xy()),
276            ));
277        }
278        let kill_nearby_colliding: bool = kill_all_colliding_if_freshly_thrown(
279            thrown_crate,
280            &collision_world,
281            &players,
282            &invincibles,
283            &mut commands,
284            transform,
285        );
286
287        if !colliding_with_players.is_empty()
288            || kill_nearby_colliding
289            || thrown_crate.break_timeout.finished()
290            || thrown_crate.crate_break_state >= 4
291            || body.is_on_ground && body.velocity.length_squared() < 0.1
292        {
293            hydrated.remove(**spawner);
294
295            let breaking_anim_frames = *breaking_anim_frames;
296            let breaking_anim_fps = *breaking_anim_fps;
297            let atlas = *breaking_atlas;
298
299            audio_center.play_sound(*break_sound, *break_sound_volume);
300
301            commands.add(
302                move |mut entities: ResMutInit<Entities>,
303                      mut transforms: CompMut<Transform>,
304                      mut animated_sprites: CompMut<AnimatedSprite>,
305                      mut lifetimes: CompMut<Lifetime>,
306                      mut atlas_sprites: CompMut<AtlasSprite>| {
307                    let pos = *transforms.get(entity).unwrap();
308                    entities.kill(entity);
309                    let breaking_anim_ent = entities.create();
310                    atlas_sprites.insert(breaking_anim_ent, AtlasSprite { atlas, ..default() });
311                    animated_sprites.insert(
312                        breaking_anim_ent,
313                        AnimatedSprite {
314                            repeat: false,
315                            fps: breaking_anim_fps,
316                            frames: (1..breaking_anim_frames).collect(),
317                            ..default()
318                        },
319                    );
320                    lifetimes.insert(breaking_anim_ent, Lifetime::new(1.0));
321                    transforms.insert(breaking_anim_ent, pos);
322                },
323            );
324        }
325    }
326}

328fn kill_all_colliding_if_freshly_thrown(
329    thrown_crate: &ThrownCrate,
330    collision_world: &CollisionWorld,
331    players: &Comp<PlayerIdx>,
332    invincibles: &CompMut<Invincibility>,
333    commands: &mut Commands,
334    transform: &Transform,
335) -> bool {
336    if thrown_crate.damage_delay.finished() {
337        return false;
338    }
339
340    let colliding_with_players = collision_world
341        .actor_collisions_filtered(thrown_crate.owner, |e| players.contains(e))
342        .into_iter()
343        .collect::<Vec<_>>();
344
345    if !colliding_with_players.is_empty() {
346        for player_entity in &colliding_with_players {
347            if invincibles.get(*player_entity).is_none() {
348                commands.add(PlayerCommand::kill(
349                    *player_entity,
350                    Some(transform.translation.xy()),
351                ));
352            }
353        }
354        commands.add(PlayerCommand::kill(
355            thrown_crate.owner,
356            Some(transform.translation.xy()),
357        ));
358        true
359    } else {
360        false
361    }
