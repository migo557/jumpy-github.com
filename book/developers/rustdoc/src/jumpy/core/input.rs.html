13/// The inputs for each player in this simulation frame.
14#[derive(Clone, Debug, HasSchema)]
15pub struct MatchInputs {
16    pub players: [PlayerInput; MAX_PLAYERS as usize],
17}
18impl MatchInputs {
19    pub fn get_control_source(&self, player_idx: usize) -> Option<ControlSource> {
20        self.players.get(player_idx).unwrap().control_source
21    }
22}
23impl Default for MatchInputs {
24    fn default() -> Self {
25        Self {
26            players: array::from_fn(|_| default()),
27        }
28    }
29}
30impl Controls<'_, PlayerControl> for MatchInputs {
31    fn get_control(&self, player_idx: usize) -> &PlayerControl {
32        &self.players.get(player_idx).unwrap().control
33    }
34
35    fn get_control_mut(&mut self, player_idx: usize) -> &mut PlayerControl {
36        &mut self.players.get_mut(player_idx).unwrap().control
37    }
38}

40/// Player input, not just controls, but also other status that comes from the player, such as the
41/// selected player and whether the player is actually active.
42#[derive(Default, Clone, Debug, HasSchema)]
43pub struct PlayerInput {
44    /// Whether or not the player is present.
45    pub active: bool,
46    /// The selected player skin.
47    pub selected_player: Handle<PlayerMeta>,
48    /// The selected player hat.
49    pub selected_hat: Option<Handle<HatMeta>>,
50    /// The player control input
51    pub control: PlayerControl,
52    /// The editor inputs the player is making, if any.
53    pub editor_input: Option<EditorInput>,
54    /// If this is [`None`] it means the player is an AI, or remote player in networked game.
55    pub control_source: Option<ControlSource>,
56
57    /// Whether or not this is an AI player.
58    pub is_ai: bool,
59}

73/// The editor inputs that a player may make.
74#[derive(Clone, Debug)]
75pub enum EditorInput {
76    /// Spawn an element onto the map.
77    SpawnElement {
78        /// The handle to the element that is being spawned.
79        handle: Handle<ElementMeta>,
80        /// The translation to spawn the element with.
81        translation: Vec2,
82        /// The map layer index to spawn the element on.
83        layer: u8,
84    },
85    MoveEntity {
86        /// The entity to move.
87        entity: Entity,
88        /// The amount to move the entity.
89        pos: Vec2,
90    },
91    DeleteEntity {
92        /// The entity to delete.
93        entity: Entity,
94    },
95    /// Create a new layer
96    CreateLayer {
97        /// The name of the layer.
98        id: String,
99    },
100    /// Rename a map layer.
101    RenameLayer {
102        /// The index of the layer to rename.
103        layer: u8,
104        /// The new name of the layer.
105        name: String,
106    },
107    DeleteLayer {
108        layer: u8,
109    },
110    /// Move a layer up or down.
111    MoveLayer {
112        /// The layer to move
113        layer: u8,
114        /// Whether or not to move the layer down. If false, move the layer up.
115        down: bool,
116    },
117    /// Update the tilemap of a layer.
118    SetTilemap {
119        /// The layer index of the layer to update.
120        layer: u8,
121        /// The handle to the tilemap to use or [`None`] to clear the tilemap.
122        handle: Option<Handle<Atlas>>,
123    },
124    SetTile {
125        /// The layer index of the layer to update
126        layer: u8,
127        /// The position of the tile to set
128        pos: UVec2,
129        /// The index in the tilemap to set the tile, or [`None`] to delete the tile.
130        tilemap_tile_idx: Option<u32>,
131        /// The tile collision kind
132        collision: TileCollisionKind,
133    },
134    RenameMap {
135        name: String,
136    },
137    RandomizeTiles {
138        tile_layers: Vec<LocatedTileLayer>,
139        element_layers: Vec<ElementLayer>,
140        tile_size: Vec2,
141    },
142}
