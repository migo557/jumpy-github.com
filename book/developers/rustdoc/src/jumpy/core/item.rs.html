7pub fn install(session: &mut SessionBuilder) {
8    Item::register_schema();
9    ItemThrow::register_schema();
10    ItemGrab::register_schema();
11    DropItem::register_schema();
12    ItemUsed::register_schema();
13
14    session
15        .stages
16        .add_system_to_stage(CoreStage::Last, grab_items)
17        .add_system_to_stage(CoreStage::Last, drop_items)
18        .add_system_to_stage(CoreStage::Last, throw_dropped_items);
19}

37/// A helper struct containing a player-inventory pair that indicates the given player is holding
38/// the other entity in their inventory.
39#[derive(Debug, Clone, Copy)]
40pub struct Inv {
41    pub player: Entity,
42    pub inventory: Entity,
43}
44
45/// System param that can be used to conveniently get the inventory of each player.
46#[derive(Deref, DerefMut, Debug)]
47pub struct PlayerInventories<'a>(&'a [Option<Inv>; MAX_PLAYERS as usize]);
48
49impl PlayerInventories<'_> {
50    pub fn find_item(&self, item: Entity) -> Option<Inv> {
51        self.0
52            .iter()
53            .find_map(|i| i.filter(|inv| inv.inventory == item))
54    }
55}
56
57impl<'a> SystemParam for PlayerInventories<'a> {
58    type State = [Option<Inv>; MAX_PLAYERS as usize];
59    type Param<'s> = PlayerInventories<'s>;
60
61    fn get_state(world: &World) -> Self::State {
62        world.run_system(
63            |entities: Res<Entities>,
64             player_indexes: Comp<PlayerIdx>,
65             inventories: Comp<Inventory>| {
66                let mut player_inventories = [None; MAX_PLAYERS as usize];
67                for (player, (idx, inventory)) in
68                    entities.iter_with((&player_indexes, &inventories))
69                {
70                    if let Some(inventory) = inventory.0 {
71                        player_inventories[idx.0 as usize] = Some(Inv { player, inventory });
72                    }
73                }
74
75                player_inventories
76            },
77            (),
78        )
79    }
80
81    fn borrow<'s>(_world: &'s World, state: &'s mut Self::State) -> Self::Param<'s> {
82        PlayerInventories(state)
83    }
84}

121/// Drop items that have the `DropItem` component added to them.
122pub fn drop_items(
123    mut commands: Commands,
124    mut drop_items: CompMut<DropItem>,
125    player_inventories: PlayerInventories,
126) {
127    for Inv { player, inventory } in player_inventories.iter().flatten() {
128        if drop_items.remove(*inventory).is_some() {
129            commands.add(PlayerCommand::set_inventory(*player, None));
130        }
131    }
132}

134pub fn grab_items(
135    entities: Res<Entities>,
136    item_grab: Comp<ItemGrab>,
137    items: Comp<Item>,
138    mut items_grabbed: CompMut<ItemGrabbed>,
139    mut bodies: CompMut<KinematicBody>,
140    mut attachments: CompMut<PlayerBodyAttachment>,
141    mut player_layers: CompMut<PlayerLayers>,
142) {
143    for (entity, (_item, item_grab)) in entities.iter_with((&items, &item_grab)) {
144        let ItemGrab {
145            fin_anim,
146            grab_offset,
147            sync_animation,
148        } = *item_grab;
149
150        if let Some(ItemGrabbed { player }) = items_grabbed.remove(entity) {
151            player_layers.get_mut(player).unwrap().fin_anim = fin_anim;
152
153            if let Some(body) = bodies.get_mut(entity) {
154                body.is_deactivated = true
155            }
156
157            attachments.insert(
158                entity,
159                PlayerBodyAttachment {
160                    player,
161                    sync_animation,
162                    sync_color: false,
163                    head: false,
164                    offset: grab_offset.extend(PlayerLayers::FIN_Z_OFFSET / 2.0),
165                },
166            );
167        }
168    }
169}

171/// Component defining the strength of the throw types when an item is dropped.
172///
173/// Mainly handled by the [`throw_dropped_items`] system which consumes the
174/// [`ItemDropped`] components for entities which have this component.
175/// [`Item`] is required for the system to take affect.
176#[derive(Clone, HasSchema)]
177#[repr(C)]
178pub struct ItemThrow {
179    normal: Vec2,
180    fast: Vec2,
181    up: Vec2,
182    drop: Vec2,
183    lob: Vec2,
184    roll: Vec2,
185    spin: f32,
186    #[schema(opaque)]
187    /// An optional system value that gets run once on throw.
188    system: Option<Arc<AtomicCell<StaticSystem<(), ()>>>>,
189}
190
191impl Default for ItemThrow {
192    fn default() -> Self {
193        Self::base()
194    }
195}
196
197impl ItemThrow {
198    /// The relative velocities of each different throw type.
199    ///
200    /// This is multiiplied by the desired throw strength in [`Self::strength`] to get a deafault
201    /// throw pattern from a single velocity.
202    pub fn base() -> Self {
203        Self {
204            normal: Vec2::new(1.5, 1.2).normalize() * 0.6,
205            fast: Vec2::new(1.5, 1.2).normalize(),
206            up: Vec2::new(0.0, 1.1),
207            drop: Vec2::new(0.0, 0.0),
208            lob: Vec2::new(1.0, 2.5).normalize() * 1.1,
209            roll: Vec2::new(0.4, -0.1),
210            spin: 0.0,
211            system: None,
212        }
213    }
214
215    /// [`Self::base`] with the throw values multiplied by `strength`.
216    pub fn strength(strength: f32) -> Self {
217        let base = Self::base();
218        Self {
219            normal: base.normal * strength,
220            fast: base.fast * strength,
221            up: base.up * strength,
222            drop: base.drop * strength,
223            lob: base.lob * strength,
224            roll: base.roll * strength,
225            spin: 0.0,
226            system: None,
227        }
228    }
229
230    pub fn with_spin(self, spin: f32) -> Self {
231        Self { spin, ..self }
232    }
233
234    pub fn with_system<Args, I>(self, system: I) -> Self
235    where
236        I: IntoSystem<Args, (), (), Sys = StaticSystem<(), ()>>,
237    {
238        Self {
239            system: Some(Arc::new(AtomicCell::new(system.system()))),
240            ..self
241        }
242    }
243
244    /// Chooses one of the throw values based on a [`PlayerControl`]
245    pub fn velocity_from_control(&self, player_control: &PlayerControl) -> Vec2 {
246        let PlayerControl { move_direction, .. } = player_control;
247        let y = move_direction.y;
248        let moving = move_direction.x.abs() > 0.0;
249        if y < 0.0 {
250            if moving {
251                return self.roll;
252            } else {
253                return self.drop;
254            }
255        }
256        if moving {
257            if y > 0.0 {
258                self.lob
259            } else {
260                self.fast
261            }
262        } else if y > 0.0 {
263            self.up
264        } else {
265            self.normal
266        }
267    }
268}

270pub fn throw_dropped_items(
271    entities: Res<Entities>,
272    item_throws: Comp<ItemThrow>,
273    items: Comp<Item>,
274    player_inputs: Res<MatchInputs>,
275    player_indexes: Comp<PlayerIdx>,
276    mut items_dropped: CompMut<ItemDropped>,
277    mut bodies: CompMut<KinematicBody>,
278    mut attachments: CompMut<PlayerBodyAttachment>,
279    mut sprites: CompMut<AtlasSprite>,
280    mut transforms: CompMut<Transform>,
281    item_spawners: Comp<DehydrateOutOfBounds>,
282    map_layers: Comp<SpawnedMapLayerMeta>,
283    player_spawnwers: Comp<PlayerSpawner>,
284    mut commands: Commands,
285) {
286    for (entity, (_items, item_throw, transform)) in
287        entities.iter_with((&items, &item_throws, &mut transforms))
288    {
289        if let Some(ItemDropped { player }) = items_dropped.get(entity).cloned() {
290            if let Some(system) = item_throw.system.clone() {
291                commands.add(move |world: &World| (system.borrow_mut().run)(world, ()));
292            }
293            items_dropped.remove(entity);
294            attachments.remove(entity);
295
296            let player_sprite = sprites.get_mut(player).unwrap();
297
298            let horizontal_flip_factor = if player_sprite.flip_x {
299                Vec2::new(-1.0, 1.0)
300            } else {
301                Vec2::ONE
302            };
303
304            let throw_velocity = item_throw.velocity_from_control(
305                &player_inputs.players[player_indexes.get(player).unwrap().0 as usize].control,
306            );
307
308            // Use the item's spawner depth as the drop depth
309            if let Some(item_spawner) = item_spawners.get(entity) {
310                let map_layer = map_layers.get(item_spawner.0).unwrap();
311                transform.translation.z = z_depth_for_map_layer(map_layer.layer_idx);
312            } else {
313                // Grab a random player spawner and use that for the z depth
314                let (_, (_, layer)) = entities
315                    .iter_with((&player_spawnwers, &map_layers))
316                    .next()
317                    .unwrap();
318                transform.translation.z = z_depth_for_map_layer(layer.layer_idx);
319            }
320
321            if let Some(body) = bodies.get_mut(entity) {
322                body.velocity = throw_velocity * horizontal_flip_factor;
323                body.angular_velocity =
324                    item_throw.spin * horizontal_flip_factor.x * throw_velocity.y.signum();
325
326                body.is_deactivated = false;
327            }
328        }
329    }
