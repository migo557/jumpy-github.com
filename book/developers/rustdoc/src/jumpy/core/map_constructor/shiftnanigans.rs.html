11/// Map randomizer using [Shiftnanigans](https://github.com/AustinHellerRepo/Shiftnanigans).
12pub struct ShiftnanigansMapConstructor {
13    pixel_board_randomizer: PixelBoardRandomizer<PixelType>,
14    compressed_top_height: usize,
15    compressed_left_width: usize,
16    tile_size: Vec2,
17}

19impl ShiftnanigansMapConstructor {
20    pub fn new(
21        map_size: UVec2,
22        tile_size: Vec2,
23        tile_layers: &[LocatedTileLayer],
24        element_layers: &[ElementLayer],
25    ) -> ShiftnanigansMapConstructor {
26        // . map tiles and elements into pixel types
27        // . run pixel randomizer
28        // . transform tiles and elements based on where they exist in the randomized map
29
30        let mut ungrouped_pixel_per_y_per_x: Vec<Vec<Option<UngroupedPixel>>> = Vec::new();
31        for x in 0..map_size.x {
32            let mut ungrouped_pixel_per_y: Vec<Option<UngroupedPixel>> = Vec::new();
33            for y in 0..map_size.y {
34                let ungrouped_pixel_option =
35                    UngroupedPixel::from_layers_and_location(tile_layers, element_layers, x, y);
36                ungrouped_pixel_per_y.push(ungrouped_pixel_option);
37            }
38            ungrouped_pixel_per_y_per_x.push(ungrouped_pixel_per_y);
39        }
40
41        let top_height;
42        'get_top_height: {
43            for y in 0..map_size.y as usize {
44                // if any column is missing a pixel, then the ceiling wall is as thick as the current value of 'y'
45                for ungrouped_pixel_per_y in ungrouped_pixel_per_y_per_x.iter() {
46                    if ungrouped_pixel_per_y[y].is_none() {
47                        top_height = y;
48                        break 'get_top_height;
49                    }
50                }
51            }
52            top_height = map_size.y as usize;
53        }
54
55        let left_width;
56        'get_left_width: {
57            for (x, ungrouped_pixel_per_y) in ungrouped_pixel_per_y_per_x.iter().enumerate() {
58                for ungrouped_pixel in ungrouped_pixel_per_y.iter() {
59                    if ungrouped_pixel.is_none() {
60                        left_width = x;
61                        break 'get_left_width;
62                    }
63                }
64            }
65            left_width = map_size.x as usize;
66        }
67
68        let bottom_height;
69        'get_bottom_height: {
70            for y in (0..map_size.y as usize).rev() {
71                for ungrouped_pixel_per_y in ungrouped_pixel_per_y_per_x.iter() {
72                    if ungrouped_pixel_per_y[y].is_none() {
73                        bottom_height = map_size.y as usize - y - 1;
74                        break 'get_bottom_height;
75                    }
76                }
77            }
78            bottom_height = 0; // pretend the bottom is missing since the top_height must also be the entire height
79        }
80
81        let right_width;
82        'get_right_width: {
83            for (x, ungrouped_pixel_per_y) in ungrouped_pixel_per_y_per_x.iter().enumerate().rev() {
84                for ungrouped_pixel in ungrouped_pixel_per_y.iter() {
85                    if ungrouped_pixel.is_none() {
86                        right_width = map_size.x as usize - x - 1;
87                        break 'get_right_width;
88                    }
89                }
90            }
91            right_width = 0; // pretend the right is missing since the left_width must also be the entire width
92        }
93
94        let composite_map_width =
95            map_size.x as usize - std::cmp::max(left_width, 1) - std::cmp::max(right_width, 1) + 2;
96        let composite_map_height =
97            map_size.y as usize - std::cmp::max(top_height, 1) - std::cmp::max(bottom_height, 1)
98                + 2;
99
100        let mut grouped_pixels_per_y_per_x: Vec<Vec<Vec<GroupedPixel>>> =
101            Vec::with_capacity(composite_map_width);
102        for _ in 0..composite_map_width {
103            let mut grouped_pixels_per_y: Vec<Vec<GroupedPixel>> =
104                Vec::with_capacity(composite_map_height);
105            for _ in 0..composite_map_height {
106                grouped_pixels_per_y.push(Vec::new());
107            }
108            grouped_pixels_per_y_per_x.push(grouped_pixels_per_y);
109        }
110
111        ungrouped_pixel_per_y_per_x
112            .into_iter()
113            .enumerate()
114            .for_each(|(x, ungrouped_pixel_per_y)| {
115                ungrouped_pixel_per_y
116                    .into_iter()
117                    .enumerate()
118                    .for_each(|(y, ungrouped_pixel)| {
119                        // the pixel board coordinates based on the map grid coordinates 'x' and 'y'
120                        let pixel_board_x;
121                        let pixel_board_y;
122
123                        if x < left_width {
124                            pixel_board_x = 0;
125                        } else if x >= map_size.x as usize - right_width {
126                            pixel_board_x = composite_map_width - 1;
127                        } else if left_width == 0 {
128                            pixel_board_x = x;
129                        } else {
130                            pixel_board_x = x - left_width + 1;
131                        }
132
133                        if y < top_height {
134                            pixel_board_y = 0;
135                        } else if y >= map_size.y as usize - bottom_height {
136                            pixel_board_y = composite_map_height - 1;
137                        } else if top_height == 0 {
138                            pixel_board_y = y;
139                        } else {
140                            pixel_board_y = y - top_height + 1
141                        }
142
143                        // add the existing pixel to the grouped pixels (changing from None to an instance if applicable)
144                        // at this point we know that (x, y) in ungrouped pixels equates to (pixel_board_x, pixel_board_y) in grouped pixels
145
146                        if let Some(ungrouped_pixel) = ungrouped_pixel {
147                            grouped_pixels_per_y_per_x[pixel_board_x][pixel_board_y].push(
148                                GroupedPixel {
149                                    ungrouped_pixel,
150                                    ungrouped_pixel_location: UVec2 {
151                                        x: x as u32,
152                                        y: y as u32,
153                                    },
154                                },
155                            );
156                        }
157                    });
158            });
159
160        // at this point all of the grouped pixels are known and so the pixel types can now be added to the pixel board
161
162        let mut pixel_board: PixelBoard<PixelType> =
163            PixelBoard::new(composite_map_width, composite_map_height);
164
165        grouped_pixels_per_y_per_x
166            .into_iter()
167            .enumerate()
168            .for_each(|(x, grouped_pixels_per_y)| {
169                grouped_pixels_per_y
170                    .into_iter()
171                    .enumerate()
172                    .for_each(|(y, grouped_pixels)| {
173                        if !grouped_pixels.is_empty() {
174                            let pixel_type = PixelType { grouped_pixels };
175                            pixel_board.set(x, y, Rc::new(RefCell::new(pixel_type)));
176                        }
177                    });
178            });
179
180        ShiftnanigansMapConstructor {
181            pixel_board_randomizer: PixelBoardRandomizer::new(pixel_board),
182            compressed_top_height: top_height,
183            compressed_left_width: left_width,
184            tile_size,
185        }
186    }
187}

189impl MapConstructor for ShiftnanigansMapConstructor {
190    fn construct_map(&self, map_manager: &mut MapManager) {
191        let random_pixel_board = self.pixel_board_randomizer.get_random_pixel_board();
192
193        // remove all tiles
194        map_manager.clear_tiles();
195
196        // remove all elements
197        map_manager.clear_elements();
198
199        // place all tiles and elements
200        for y in 0..random_pixel_board.get_height() {
201            for x in 0..random_pixel_board.get_width() {
202                if random_pixel_board.exists(x, y) {
203                    let wrapped_pixel = random_pixel_board.get(x, y).unwrap();
204                    let borrowed_pixel: &PixelType = &wrapped_pixel.borrow();
205                    let top_left_position: UVec2 = borrowed_pixel
206                        .grouped_pixels
207                        .first()
208                        .unwrap()
209                        .ungrouped_pixel_location;
210
211                    // calculate the x and y that the grouped pixels uncompress to
212                    let uncompressed_y: usize = if y == 0 {
213                        0
214                    } else {
215                        let top_offset: usize = if self.compressed_top_height == 0 {
216                            0
217                        } else {
218                            self.compressed_top_height - 1
219                        };
220                        y + top_offset
221                    };
222                    let uncompressed_x: usize = if x == 0 {
223                        0
224                    } else {
225                        let left_offset: usize = if self.compressed_left_width == 0 {
226                            0
227                        } else {
228                            self.compressed_left_width - 1
229                        };
230                        x + left_offset
231                    };
232
233                    borrowed_pixel.grouped_pixels.iter().for_each(|gp| {
234                        gp.ungrouped_pixel
235                            .layer_pixel_entity_types
236                            .iter()
237                            .for_each(|lpet| match lpet {
238                                LayerPixelEntityType::Tile(tile) => {
239                                    let position = UVec2 {
240                                        x: uncompressed_x as u32 + gp.ungrouped_pixel_location.x
241                                            - top_left_position.x,
242                                        y: uncompressed_y as u32 + gp.ungrouped_pixel_location.y
243                                            - top_left_position.y,
244                                    };
245                                    map_manager.set_tile(
246                                        tile.layer_index,
247                                        position,
248                                        &Some(tile.tilemap_tile_index),
249                                        tile.tile_collision_kind,
250                                    );
251                                }
252                                LayerPixelEntityType::Element(element) => {
253                                    let position = Vec2 {
254                                        x: (uncompressed_x as f32
255                                            + gp.ungrouped_pixel_location.x as f32
256                                            - top_left_position.x as f32
257                                            + element.position.x)
258                                            * self.tile_size.x,
259                                        y: (uncompressed_y as f32
260                                            + gp.ungrouped_pixel_location.y as f32
261                                            - top_left_position.y as f32
262                                            + element.position.y)
263                                            * self.tile_size.y,
264                                    };
265                                    map_manager.create_element(
266                                        &element.element_meta_handle,
267                                        &position,
268                                        element.layer_index,
269                                    );
270                                }
271                            });
272                    });
273                }
274            }
275        }
276    }
277}

299/// This contains the layer pixels
300struct UngroupedPixel {
301    layer_pixel_entity_types: Vec<LayerPixelEntityType>,
302}
303
304impl UngroupedPixel {
305    fn from_layers_and_location(
306        tile_layers: &[LocatedTileLayer],
307        element_layers: &[ElementLayer],
308        x: u32,
309        y: u32,
310    ) -> Option<Self> {
311        // iterate over each layer at the provided location, creating a vector of MapEntityType
312        let mut entity_types: Vec<LayerPixelEntityType> = Vec::new();
313        tile_layers.iter().for_each(|tile_layer| {
314            tile_layer.located_tiles.iter().for_each(|tile| {
315                if tile.0.x == x && tile.0.y == y {
316                    entity_types.push(LayerPixelEntityType::Tile(Tile {
317                        layer_index: tile_layer.layer_index,
318                        tilemap_tile_index: tile.1,
319                        tile_collision_kind: tile.2,
320                    }));
321                }
322            });
323        });
324        element_layers.iter().for_each(|element_layer| {
325            element_layer.located_elements.iter().for_each(|element| {
326                // grab elements nearby this location
327                if element.0.x as u32 == x && element.0.y as u32 == y {
328                    // TODO determine which element this is in 'element' for custom spacing requirements
329                    entity_types.push(LayerPixelEntityType::Element(Element {
330                        layer_index: element_layer.layer_index,
331                        element_meta_handle: element.1,
332                        position: Vec2::new(element.0.x - x as f32, element.0.y - y as f32),
333                    }));
334                }
335            });
336        });
337        if entity_types.is_empty() {
338            None
339        } else {
340            Some(UngroupedPixel {
341                layer_pixel_entity_types: entity_types,
342            })
343        }
344    }

347/// This can contain one or more ungrouped pixels, allowing for combining adjacent pixels into one grouped pixel
348///     This is useful for when thick walls need to be considered as one-pixel-wide walls
349struct GroupedPixel {
350    ungrouped_pixel: UngroupedPixel,
351    ungrouped_pixel_location: UVec2,
352}
353
354/// The pixel provided to the pixel board randomizer
355struct PixelType {

357}
