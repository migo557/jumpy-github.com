18/// Resource that can produce deterministic, pseudo-random numbers.
19///
20/// Access in a system with [`Res<GlobalRng>`].
21#[derive(Clone, HasSchema, Deref, DerefMut)]
22#[type_data(SchemaLuaEcsRefMetatable(lua_metatable))]
23pub struct GlobalRng(AtomicRng);
24
25pub const DEFAULT_RANDOM_SEED: u32 = 7;
26
27impl Default for GlobalRng {
28    fn default() -> Self {
29        Self(AtomicRng::with_seed(DEFAULT_RANDOM_SEED as u64))
30    }
31}
32
33fn lua_metatable(ctx: lua::Context) -> lua::Table {
34    let metatable = lua::Table::new(&ctx);
35
36    let f32_fn = ctx.registry().stash(
37        &ctx,
38        Callback::from_fn(&ctx, |ctx, _fuel, mut stack| {
39            let this: &EcsRef = stack.consume(ctx)?;
40            let mut b = this.borrow_mut();
41            let global_rng = b.schema_ref_mut()?.cast_into_mut::<GlobalRng>();
42            let n = global_rng.0.f32();
43            stack.replace(ctx, n);
44            Ok(lua::CallbackReturn::Return)
45        }),
46    );
47    metatable
48        .set(
49            ctx,
50            "__index",
51            Callback::from_fn(&ctx, move |ctx, _fuel, mut stack| {
52                let (_this, key): (lua::Value, lua::String) = stack.consume(ctx)?;
53
54                #[allow(clippy::single_match)]
55                match key.as_bytes() {
56                    b"f32" => {
57                        stack.push_front(ctx.registry().fetch(&f32_fn).into());
58                    }
59                    _ => (),
60                }
61                Ok(lua::CallbackReturn::Return)
62            }),
63        )
64        .unwrap();
65
66    metatable
67}
