14/// Track state of ragdoll.
15#[derive(HasSchema, Clone, Default)]
16pub struct PlayerRagdollState {
17    /// If ticking, cannot twitch
18    twitch_timer: Timer,
19
20    // Last frame that started twitch animation
21    last_twitch_anim_frame: u64,
22}

24pub fn handle_player_state(
25    entities: Res<Entities>,
26    time: Res<Time>,
27    mut player_states: CompMut<PlayerState>,
28    player_indexes: Comp<PlayerIdx>,
29    mut transform: CompMut<Transform>,
30    assets: Res<AssetServer>,
31    player_inputs: Res<MatchInputs>,
32    atlas_sprites: Comp<AtlasSprite>,
33    mut animations: CompMut<AnimationBankSprite>,
34    mut bodies: CompMut<KinematicBody>,
35    mut dynamic_bodies: CompMut<DynamicBody>,
36    mut ragdoll_states: CompMut<PlayerRagdollState>,
37    game_meta: Root<GameMeta>,
38    mut collision_world: CollisionWorld,
39    mut commands: Commands,
40) {
41    for (player_ent, (state, transform, animation, player_idx, atlas_sprite)) in
42        entities.iter_with((
43            &mut player_states,
44            &mut transform,
45            &mut animations,
46            &player_indexes,
47            &atlas_sprites,
48        ))
49    {
50        if state.current != *ID {
51            continue;
52        };
53
54        let meta_handle = player_inputs.players[player_idx.0 as usize].selected_player;
55        let meta = &*assets.get(meta_handle);
56        let player_physics = &game_meta.core.physics.player;
57        let control = player_inputs.players[player_idx.0 as usize].control;
58
59        match state.age {
60            0 => {
61                // TODO find right animation
62                animation.current = "ragdoll".into();
63
64                // drop item
65                commands.add(PlayerCommand::set_inventory(player_ent, None));
66
67                // Knock the player's hat off if they had one.
68                commands.add(PlayerCommand::drop_hat(player_ent));
69
70                // Set to simulate physics
71                let dynamic_body =
72                    dynamic_bodies.get_mut_or_insert(player_ent, DynamicBody::default);
73                dynamic_body.is_dynamic = true;
74
75                let body = bodies.get_mut(player_ent).unwrap();
76                let player_physics = &game_meta.core.physics.player;
77                let pop_vel = Vec2::new(0.0, player_physics.ragdoll_initial_pop);
78                let ang_vel = player_physics.ragdoll_initial_ang_vel;
79                let sprite_dir = atlas_sprite.flip_x;
80                let current_vel = body.velocity;
81                let additional_mass = player_physics.ragdoll_additional_mass;
82
83                dynamic_body.push_simulation_command(Box::new(
84                    move |body: &mut rapier::RigidBody| {
85                        // Transfer current kinematic velocity to dynamic
86                        body.set_linvel((current_vel).into(), true);
87
88                        // Apply initial pop upward
89                        body.apply_impulse((pop_vel * body.mass()).into(), true);
90
91                        // Apply angular velocity
92                        let angular_dir = if sprite_dir { 1.0 } else { -1.0 };
93                        body.apply_torque_impulse(
94                            ang_vel
95                                * angular_dir
96                                * body.mass_properties().effective_angular_inertia(),
97                            true,
98                        );
99
100                        body.set_additional_mass(additional_mass, true);
101                    },
102                ));
103
104                ragdoll_states.insert(
105                    player_ent,
106                    PlayerRagdollState {
107                        twitch_timer: Timer::from_seconds(
108                            player_physics.ragdoll_twitch_delay,
109                            TimerMode::Once,
110                        ),
111                        last_twitch_anim_frame: 0,
112                    },
113                );
114                use_ragdoll_collider(player_ent, meta, &mut collision_world)
115            }
116            n if n > 1 => {
117                let ragdoll_state = ragdoll_states.get_mut(player_ent).unwrap();
118                if control.ragdoll_just_pressed {
119                    state.current = *idle::ID;
120                    animation.current = ustr("idle");
121
122                    // Switch back to kinematic
123                    dynamic_bodies.get_mut(player_ent).unwrap().is_dynamic = false;
124                    transform.rotation = Quat::default();
125
126                    let pop_vel = Vec2::new(0.0, player_physics.ragdoll_initial_pop);
127                    let body = bodies.get_mut(player_ent).unwrap();
128                    body.velocity = pop_vel;
129
130                    restore_player_collider(player_ent, meta, &mut collision_world);
131                }
132                // Handle twitching on input
133                else if control.move_direction != Vec2::ZERO {
134                    // Update timer determining if we may twitch on input
135                    let timer = &mut ragdoll_state.twitch_timer;
136                    timer.tick(time.delta());
137                    let can_twitch = timer.finished();
138
139                    animation.current = "ragdoll_twitch".into();
140                    ragdoll_state.last_twitch_anim_frame = n;
141
142                    // Check if touching solid, no twitching in air
143                    // TODO: Maybe it's fun to allow the minor air control?
144                    // let touching_solid = collision_world.tile_collision(*transform, body.shape)
145                    //     != TileCollisionKind::Empty;
146
147                    // Don't twitch when holding, only tap.
148                    if can_twitch && control.just_moved
149                    /*&& touching_solid*/
150                    {
151                        let dynamic_body = dynamic_bodies.get_mut(player_ent).unwrap();
152                        let twitch_vel = player_physics.ragdoll_twitch_vel;
153
154                        dynamic_body.push_simulation_command(Box::new(
155                            move |body: &mut rapier::RigidBody| {
156                                let mut lin_vel = *body.linvel();
157                                let dir_x = control.move_direction.x;
158                                lin_vel += vector!(dir_x * twitch_vel, 3.0 * twitch_vel);
159                                body.set_linvel(lin_vel, true);
160                            },
161                        ));
162
163                        timer.reset();
164                    }
165                } else {
166                    // Not twitching, back to normal animation if twitch anim held for at least 2 frames
167                    if n >= ragdoll_state.last_twitch_anim_frame + 4 {
168                        animation.current = "ragdoll".into();
169                    }
170                }
171            }
172            _ => {}
173        }
174    }
175}

177/// Switch entity to a capsule collider based on collider
178/// from [`PlayerMeta`].
179pub fn use_ragdoll_collider(
180    entity: Entity,
181    player_meta: &PlayerMeta,
182    collision_world: &mut CollisionWorld,
183) {
184    // Build capsule from player size.
185    let radius = player_meta.body_size.x / 2.0;
186    let half_length = player_meta.body_size.y * 0.5 - radius;
187    let capsule = ColliderShape::CapsuleY {
188        half_length,
189        radius,
190    };
191    let mut builder = build_actor_rapier_collider(entity, capsule.shared_shape());
192
193    // Shift capsule slightly to lineup with animation
194    let translation = rapier::Translation::new(0.0, 5.0);
195    builder = builder.position(translation.into());
196
197    collision_world.set_actor_shape_from_builder(entity, builder, capsule);
198}

200/// Restore player shape back to default from `PlayerMeta`.
201fn restore_player_collider(
202    player_ent: Entity,
203    player_meta: &PlayerMeta,
204    collision_world: &mut CollisionWorld,
205) {
206    collision_world.set_actor_shape(player_ent, player_collider_shape(player_meta));
