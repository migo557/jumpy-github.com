11pub fn player_state_transition(
12    entities: Res<Entities>,
13    player_inputs: Res<MatchInputs>,
14    player_indexes: Comp<PlayerIdx>,
15    mut player_states: CompMut<PlayerState>,
16    assets: Res<AssetServer>,
17    mut bodies: CompMut<KinematicBody>,
18    mut transforms: CompMut<Transform>,
19) {
20    for (_ent, (state, player_idx, body, transform)) in entities.iter_with((
21        &mut player_states,
22        &player_indexes,
23        &mut bodies,
24        &mut transforms,
25    )) {
26        let player_input = &player_inputs.players[player_idx.0 as usize];
27        let meta_handle = player_input.selected_player;
28        let meta = assets.get(meta_handle);
29
30        // Reset the body size and position if we stop sliding
31        if state.last == *ID && state.current != *ID {
32            if let ColliderShape::Rectangle { size } = &body.shape {
33                if *size != meta.body_size {
34                    body.shape = ColliderShape::Rectangle {
35                        size: meta.body_size,
36                    };
37                    let offset = (meta.body_size.y - meta.slide_body_size.y) / 2.0;
38                    let direction = player_input.control.move_direction.x.signum();
39                    transform.translation.x += offset * direction;
40                    transform.translation.y += offset;
41                }
42            }
43        }
44
45        if state.current != *ID {
46            continue;
47        }
48        let control = &player_inputs.players[player_idx.0 as usize].control;
49
50        if control.ragdoll_just_pressed {
51            state.current = *ragdoll::ID;
52        } else if !body.is_on_ground || control.move_direction.y > -0.5 {
53            state.current = *idle::ID;
54        }
55    }
56}

58pub fn handle_player_state(
59    entities: Res<Entities>,
60    player_inputs: Res<MatchInputs>,
61    player_states: Comp<PlayerState>,
62    player_indexes: Comp<PlayerIdx>,
63    mut animations: CompMut<AnimationBankSprite>,
64    mut bodies: CompMut<KinematicBody>,
65    assets: Res<AssetServer>,
66    mut transforms: CompMut<Transform>,
67) {
68    for (_player_ent, (state, player_idx, animation, body, transform)) in entities.iter_with((
69        &player_states,
70        &player_indexes,
71        &mut animations,
72        &mut bodies,
73        &mut transforms,
74    )) {
75        if state.current != *ID {
76            continue;
77        }
78        let player_input = &player_inputs.players[player_idx.0 as usize];
79        let meta_handle = player_input.selected_player;
80        let meta = assets.get(meta_handle);
81
82        if body.velocity.x == 0.0 {
83            animation.current = "crouch".into();
84            if let ColliderShape::Rectangle { size } = &body.shape {
85                if *size != meta.body_size {
86                    body.shape = ColliderShape::Rectangle {
87                        size: meta.body_size,
88                    };
89                    let offset = (meta.body_size.y - meta.slide_body_size.y) / 2.0;
90                    let direction = player_input.control.move_direction.x.signum();
91                    transform.translation.x += offset * direction;
92                    transform.translation.y += offset;
93                }
94            }
95        } else if let ColliderShape::Rectangle { size } = &body.shape {
96            animation.current = "slide".into();
97
98            if *size != meta.slide_body_size {
99                body.shape = ColliderShape::Rectangle {
100                    size: meta.slide_body_size,
101                };
102                let offset = (meta.body_size.y - meta.slide_body_size.y) / 2.0;
103                transform.translation.x -= offset;
104                transform.translation.y -= offset;
105            }
106        }
107
108        let control = &player_inputs.players[player_idx.0 as usize].control;
109
110        if control.jump_just_pressed {
111            body.fall_through = true;
112        }
113    }
