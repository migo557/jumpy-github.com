31impl SystemStage for PlayerStateStageImpl {
32    fn id(&self) -> Ulid {
33        Self::ID
34    }
35
36    fn name(&self) -> String {
37        "PlayerStateStage".into()
38    }
39
40    fn run(&mut self, world: &World) {
41        trace!("Starting player state transitions");
42        loop {
43            // Get the current player states
44            let last_player_states = world.run_system(
45                |entities: Res<Entities>,
46                 player_indexes: Comp<PlayerIdx>,
47                 player_states: Comp<PlayerState>| {
48                    let mut states: [Option<Ustr>; MAX_PLAYERS as usize] =
49                        std::array::from_fn(|_| None);
50                    for (_ent, (idx, state)) in
51                        entities.iter_with((&player_indexes, &player_states))
52                    {
53                        states[idx.0 as usize] = Some(state.current);
54                    }
55
56                    states
57                },
58                (),
59            );
60            trace!(?last_player_states, "Checcking current states");
61
62            trace!("Running state transitions");
63            // Run all of the player state systems
64            for system in &mut self.systems {
65                system.run(world, ());
66            }
67
68            // Get whether the states have changed
69            let has_changed = world.run_system(
70                move |entities: Res<Entities>,
71                      player_indexes: Comp<PlayerIdx>,
72                      mut player_states: CompMut<PlayerState>| {
73                    let mut has_changed = false;
74                    for (_ent, (idx, state)) in
75                        entities.iter_with((&player_indexes, &mut player_states))
76                    {
77                        let old_state = last_player_states[idx.0 as usize].unwrap();
78
79                        if old_state != state.current {
80                            state.last = old_state;
81                            state.age = 0;
82                            has_changed = true;
83                        }
84                    }
85
86                    has_changed
87                },
88                (),
89            );
90
91            // If the states haven't changed
92            if !has_changed {
93                trace!("No state changes, done with state transition loop");
94                // Then we are finished applying player state transitions
95                break;
96            }
97        }
98    }
99
100    fn add_system(&mut self, system: StaticSystem<(), ()>) {
101        self.systems.push(system.system())
102    }
103
104    fn remove_all_systems(&mut self) {
105        self.systems.clear();
106    }
