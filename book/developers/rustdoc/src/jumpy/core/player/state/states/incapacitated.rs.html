9const SLOWING_SPEED: f32 = 0.3;
10
11pub fn handle_player_state(
12    entities: Res<Entities>,
13    mut player_states: CompMut<PlayerState>,
14    player_indexes: Comp<PlayerIdx>,
15    assets: Res<AssetServer>,
16    player_inputs: Res<MatchInputs>,
17    atlas_sprites: Comp<AtlasSprite>,
18    mut animations: CompMut<AnimationBankSprite>,
19    mut bodies: CompMut<KinematicBody>,
20) {
21    for (player_ent, (state, animation, body, player_idx, atlas_sprite)) in entities.iter_with((
22        &mut player_states,
23        &mut animations,
24        &mut bodies,
25        &player_indexes,
26        &atlas_sprites,
27    )) {
28        if state.current != *ID {
29            continue;
30        };
31
32        let meta_handle = player_inputs.players[player_idx.0 as usize].selected_player;
33        let meta = assets.get(meta_handle);
34
35        match state.age {
36            0 => {
37                // TODO find right animation
38                animation.current = "rise".into();
39                PlayerCommand::set_inventory(player_ent, None);
40
41                if body.velocity.x.abs() < meta.stats.walk_speed {
42                    body.velocity.x = 5. * if atlas_sprite.flip_x { -1.0f32 } else { 1.0 };
43                }
44            }
45            n if n < 80 => {
46                if body.velocity.x.abs() < SLOWING_SPEED {
47                    body.velocity.x = 0.;
48                } else {
49                    body.velocity.x -= body.velocity.x.signum() * SLOWING_SPEED
50                }
51            }
52            n if n >= 80 => {
53                state.current = *idle::ID;
54                animation.current = ustr("idle");
55            }
56            _ => (),
57        }
58    }
59}
