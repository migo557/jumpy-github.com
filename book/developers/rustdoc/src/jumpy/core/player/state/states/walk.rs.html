11pub fn player_state_transition(
12    entities: Res<Entities>,
13    player_inputs: Res<MatchInputs>,
14    player_indexes: Comp<PlayerIdx>,
15    mut player_states: CompMut<PlayerState>,
16    bodies: Comp<KinematicBody>,
17) {
18    for (_ent, (player_idx, player_state, body)) in
19        entities.iter_with((&player_indexes, &mut player_states, &bodies))
20    {
21        if player_state.current != *ID {
22            continue;
23        }
24
25        let control = &player_inputs.players[player_idx.0 as usize].control;
26
27        if control.ragdoll_just_pressed {
28            player_state.current = *ragdoll::ID;
29        } else if !body.is_on_ground {
30            player_state.current = *midair::ID;
31        } else if control.move_direction.y < -0.5 {
32            player_state.current = *crouch::ID;
33        } else if control.move_direction.x == 0.0 {
34            player_state.current = *idle::ID;
35        }
36    }
37}

39pub fn handle_player_state(
40    entities: Res<Entities>,
41    player_inputs: Res<MatchInputs>,
42    player_indexes: Comp<PlayerIdx>,
43    player_states: Comp<PlayerState>,
44    assets: Res<AssetServer>,
45    mut sprites: CompMut<AtlasSprite>,
46    mut animations: CompMut<AnimationBankSprite>,
47    mut bodies: CompMut<KinematicBody>,
48    mut audio_center: ResMut<AudioCenter>,
49) {
50    let players = entities.iter_with((
51        &player_states,
52        &player_indexes,
53        &mut animations,
54        &mut sprites,
55        &mut bodies,
56    ));
57    for (_player_ent, (player_state, player_idx, animation, sprite, body)) in players {
58        if player_state.current != *ID {
59            continue;
60        }
61        let meta_handle = player_inputs.players[player_idx.0 as usize].selected_player;
62        let meta = assets.get(meta_handle);
63        let control = &player_inputs.players[player_idx.0 as usize].control;
64
65        // If this is the first frame of this state
66        if player_state.age == 0 {
67            // set our animation
68            animation.current = "walk".into();
69        }
70
71        // If we are jumping
72        if control.jump_just_pressed {
73            audio_center.play_sound(meta.sounds.jump, meta.sounds.jump_volume);
74
75            // Move up
76            body.velocity.y = meta.stats.jump_speed;
77        }
78
79        // Walk in movement direction
80        body.velocity.x += meta.stats.accel_walk_speed * control.move_direction.x;
81        if control.move_direction.x.is_sign_positive() {
82            body.velocity.x = body
83                .velocity
84                .x
85                .min(meta.stats.walk_speed * control.move_direction.x);
86        } else {
87            body.velocity.x = body
88                .velocity
89                .x
90                .max(meta.stats.walk_speed * control.move_direction.x);
91        }
92
93        // Point in movement direction
94        if control.move_direction.x > 0.0 {
95            sprite.flip_x = false;
96        } else if control.move_direction.x < 0.0 {
97            sprite.flip_x = true;
98        }
99    }
