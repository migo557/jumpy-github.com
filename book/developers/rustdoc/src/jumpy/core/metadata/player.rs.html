117fn deserialize_body_animations<'de, D: Deserializer<'de>>(
118    deserializer: D,
119) -> Result<BodyAnimationsMeta, D::Error> {
120    #[derive(Clone, Deserialize)]
121    #[serde(deny_unknown_fields)]
122    struct AnimatedBodySprite {
123        #[serde(default)]
124        pub index: u32,
125        #[serde(deserialize_with = "deserialize_arc_slice")]
126        #[serde(serialize_with = "serialize_arc_slice")]
127        pub frames: Arc<[BodyFrame]>,
128        /// The frames per second to play the animation at.
129        pub fps: f32,
130        /// The amount of time the current frame has been playing
131        #[serde(default)]
132        pub timer: f32,
133        /// Whether or not to repeat the animation
134        #[serde(default = "default_true")]
135        pub repeat: bool,
136    }
137
138    #[derive(Clone, Deserialize)]
139    #[serde(deny_unknown_fields)]
140    struct BodyFrame {
141        pub idx: u32,
142        #[serde(default)]
143        pub offset: Vec2,
144        #[serde(default)]
145        pub head_offset: Vec2,
146    }
147
148    let body_sprite_anmations = <HashMap<String, AnimatedBodySprite>>::deserialize(deserializer)?;
149
150    let offsets = body_sprite_anmations
151        .iter()
152        .map(|(k, v)| {
153            (
154                ustr(k),
155                v.frames
156                    .iter()
157                    .map(|x| Offsets {
158                        body: x.offset,
159                        head: x.head_offset,
160                    })
161                    .collect::<SVec<_>>(),
162            )
163        })
164        .collect();
165    let frames = body_sprite_anmations
166        .into_iter()
167        .map(|(k, v)| {
168            (
169                ustr(&k),
170                AnimatedSprite {
171                    index: v.index,
172                    frames: v.frames.iter().map(|x| x.idx).collect(),
173                    fps: v.fps,
174                    timer: v.timer,
175                    repeat: v.repeat,
176                },
177            )
178        })
179        .collect();
180
181    Ok(BodyAnimationsMeta { offsets, frames })
182}
