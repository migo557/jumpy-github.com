61impl MapMeta {
62    /// Checks if the given position is out of the bounds of the map.
63    pub fn is_out_of_bounds(&self, pos: &Vec3) -> bool {
64        const KILL_ZONE_BORDER: f32 = 500.0;
65        let map_width = self.grid_size.x as f32 * self.tile_size.x;
66        let left_kill_zone = -KILL_ZONE_BORDER;
67        let right_kill_zone = map_width + KILL_ZONE_BORDER;
68        let bottom_kill_zone = -KILL_ZONE_BORDER;
69        pos.x < left_kill_zone || pos.x > right_kill_zone || pos.y < bottom_kill_zone
70    }
