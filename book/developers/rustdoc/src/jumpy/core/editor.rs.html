15impl_system_param! {
16    /// A system parameter for editing the map.
17    ///
18    /// [`MapManager`] provides an interface for editing the map contents easily while the match is
19    /// running. This can be used both for manual map editing, and by algorithms for generating or
20    /// randomizing maps.
21    ///
22    /// Map generators implement the [`MapConstructor`]
23    /// trait, which is given a [`MapManager`] to make its changes with.
24    pub struct MapManager<'a> {
25        commands: Commands<'a>,
26        entities: ResMutInit<'a, Entities>,
27        spawned_map_meta: ResMutInit<'a, SpawnedMapMeta>,
28        element_handles: CompMut<'a, ElementHandle>,
29        transforms: CompMut<'a, Transform>,
30        spawned_map_layer_metas: CompMut<'a, SpawnedMapLayerMeta>,
31        tile_layers: CompMut<'a, TileLayer>,
32        tiles: CompMut<'a, Tile>,
33        tile_collisions: CompMut<'a, TileCollisionKind>,
34        map: Res<'a, LoadedMap>,
35        element_kill_callbacks: Comp<'a, ElementKillCallback>,
36        spawner_manager: SpawnerManager<'a>,
37    }
38}
39
40impl<'a> MapManager<'a> {
41    /// Create a new map element at the given location on the given layer.
42    pub fn create_element(
43        &mut self,
44        element_meta_handle: &Handle<ElementMeta>,
45        translation: &Vec2,
46        layer_index: u32,
47    ) {
48        let entity = self.entities.create();
49        // TODO remove element handles as the underlying elements are removed
50        self.element_handles
51            .insert(entity, ElementHandle(*element_meta_handle));
52        let z_depth = z_depth_for_map_layer(layer_index);
53        self.transforms.insert(
54            entity,
55            Transform::from_translation(translation.extend(z_depth)),
56        );
57        self.spawned_map_layer_metas.insert(
58            entity,
59            SpawnedMapLayerMeta {
60                layer_idx: layer_index,
61            },
62        );
63    }
64    /// Create a new layer with the given name.
65    pub fn create_layer(&mut self, name: Ustr) {
66        let entity = self.entities.create();
67        let layer_index = self.spawned_map_meta.layer_names.len() as u32;
68        self.spawned_map_meta.layer_names = self
69            .spawned_map_meta
70            .layer_names
71            .clone()
72            .iter()
73            .cloned()
74            .chain([name])
75            .collect();
76        self.spawned_map_layer_metas.insert(
77            entity,
78            SpawnedMapLayerMeta {
79                layer_idx: layer_index,
80            },
81        );
82        self.tile_layers.insert(
83            entity,
84            TileLayer::new(
85                self.spawned_map_meta.grid_size,
86                self.spawned_map_meta.tile_size,
87                default(),
88            ),
89        );
90        self.transforms.insert(
91            entity,
92            Transform::from_translation(Vec3::new(0.0, 0.0, z_depth_for_map_layer(layer_index))),
93        );
94    }
95    /// Delete the layer with the given index.
96    pub fn delete_layer(&mut self, layer_index: u32) {
97        let layer_count = self.spawned_map_meta.layer_names.len() as u32;
98        let layers_to_decrement = layer_count - layer_index;
99        self.spawned_map_meta.layer_names = self
100            .spawned_map_meta
101            .layer_names
102            .clone()
103            .iter()
104            .cloned()
105            .enumerate()
106            .filter_map(|(i, name)| {
107                if i as u32 == layer_index {
108                    None
109                } else {
110                    Some(name)
111                }
112            })
113            .collect();
114        let mut to_kill = Vec::new();
115        self.entities
116            .iter_with(&mut self.spawned_map_layer_metas)
117            .for_each(|(entity, layer)| {
118                if layer.layer_idx == layer_index {
119                    to_kill.push(entity);
120
121                    if let Some(tile_layer) = self.tile_layers.get(entity) {
122                        for tile in tile_layer.tiles.iter().flatten() {
123                            to_kill.push(*tile);
124                        }
125                    }
126                };
127
128                if layer.layer_idx > layer_count - layers_to_decrement {
129                    layer.layer_idx -= 1;
130                }
131            });
132        to_kill.into_iter().for_each(|ent| {
133            self.entities.kill(ent);
134        });
135    }
136    /// Rename the layer with the given index.
137    pub fn rename_layer(&mut self, layer_index: u32, name: &str) {
138        self.spawned_map_meta.layer_names = self
139            .spawned_map_meta
140            .layer_names
141            .clone()
142            .iter()
143            .cloned()
144            .enumerate()
145            .map(|(i, n)| {
146                if i as u32 == layer_index {
147                    ustr(name)
148                } else {
149                    n
150                }
151            })
152            .collect();
153    }
154    /// Move an element to a new position on the map.
155    pub fn move_element(&mut self, entity: Entity, position: &Vec2) {
156        let transform = self.transforms.get_mut(entity).unwrap();
157        transform.translation.x = position.x;
158        transform.translation.y = position.y;
159    }
160    /// Delete an element off of the map.
161    pub fn delete_element(&mut self, entity: Entity) {
162        if let Some(element_kill_callback) = self.element_kill_callbacks.get(entity) {
163            let system = element_kill_callback.system.clone();
164            self.commands
165                .add(move |world: &World| (system.lock().unwrap().run)(world, ()));
166        } else if self.spawner_manager.is_entity_a_spawner(entity) {
167            // if the entity is a spawner, there are specific rules around how child entities may be killed
168            self.spawner_manager.kill_spawner_entity(
169                entity,
170                &mut self.entities,
171                &self.element_kill_callbacks,
172                &mut self.commands,
173            );
174        } else {
175            self.entities.kill(entity);
176        }
177    }
178    /// Set the tilemap for the given layer.
179    pub fn set_layer_tilemap(&mut self, layer_index: u32, tilemap: &Option<Handle<Atlas>>) {
180        if let Some((_, (tile_layer, _))) = self
181            .entities
182            .iter_with((&mut self.tile_layers, &self.spawned_map_layer_metas))
183            .find(|x| x.1 .1.layer_idx == layer_index)
184        {
185            if let Some(handle) = tilemap {
186                tile_layer.atlas = *handle;
187            } else {
188                tile_layer.atlas = default();
189            }
190        };
191    }
192    /// Set the tile index of a tile on the given layer.
193    pub fn set_tile(
194        &mut self,
195        layer_index: u32,
196        position: UVec2,
197        tilemap_tile_index: &Option<u32>,
198        tile_collision_kind: TileCollisionKind,
199    ) {
200        if let Some((_, (tile_layer, _))) = self
201            .entities
202            .iter_with((&mut self.tile_layers, &self.spawned_map_layer_metas))
203            .find(|x| x.1 .1.layer_idx == layer_index)
204        {
205            if let Some(entity) = tile_layer.get(position) {
206                if let Some(idx) = tilemap_tile_index.as_ref() {
207                    self.tiles.get_mut(entity).unwrap().idx = *idx;
208
209                    // TODO: technically setting the collision to empty should be
210                    // equivalent to removing the component, but it isn't working like
211                    // that right now.
212                    if tile_collision_kind != TileCollisionKind::Empty {
213                        self.tile_collisions.insert(entity, tile_collision_kind);
214                    } else {
215                        self.tile_collisions.remove(entity);
216                    }
217                } else {
218                    self.entities.kill(entity);
219                    tile_layer.set(position, None);
220                }
221            } else if let Some(idx) = tilemap_tile_index {
222                let entity = self.entities.create();
223                tile_layer.set(position, Some(entity));
224                self.tiles.insert(
225                    entity,
226                    Tile {
227                        idx: *idx,
228                        ..default()
229                    },
230                );
231
232                // TODO: technically setting the collision to empty should be
233                // equivalent to removing the component, but it isn't workidesired_keyng like
234                // that right now.
235                if tile_collision_kind != TileCollisionKind::Empty {
236                    self.tile_collisions.insert(entity, tile_collision_kind);
237                } else {
238                    self.tile_collisions.remove(entity);
239                }
240            }
241
242            self.commands
243                .add(move |mut collision_world: CollisionWorld| {
244                    collision_world.update_tile(layer_index, position);
245                });
246        };
247    }
248    /// Swap the position of two layers.
249    pub fn swap_layer(&mut self, layer_index: u32, is_downward: bool) {
250        let origin_layer_index = layer_index;
251        let other_layer_index = if is_downward {
252            origin_layer_index + 1
253        } else {
254            origin_layer_index - 1
255        };
256        let mut layer_names = self.spawned_map_meta.layer_names.to_vec();
257        layer_names.swap(origin_layer_index as usize, other_layer_index as usize);
258        self.spawned_map_meta.layer_names = layer_names.into_iter().collect();
259
260        for (_, (transform, layer_meta)) in self
261            .entities
262            .iter_with((&mut self.transforms, &mut self.spawned_map_layer_metas))
263        {
264            if layer_meta.layer_idx == origin_layer_index {
265                layer_meta.layer_idx = other_layer_index;
266                transform.translation.z = z_depth_for_map_layer(other_layer_index);
267            } else if layer_meta.layer_idx == other_layer_index {
268                layer_meta.layer_idx = origin_layer_index;
269                transform.translation.z = z_depth_for_map_layer(origin_layer_index);
270            }
271        }
272    }
273    /// Rename the map.
274    pub fn rename_map(&mut self, name: &str) {
275        self.spawned_map_meta.name = ustr(name);
276    }
277    /// Get the size of the map.
278    pub fn get_size(&self) -> UVec2 {
279        self.spawned_map_meta.grid_size
280    }
281    /// The the number of layers.
282    pub fn get_layers_total(&self) -> usize {
283        self.spawned_map_meta.layer_names.len()
284    }
285    /// Clear all the tiles on the map.
286    pub fn clear_tiles(&mut self) {
287        let empty_tile: Option<u32> = Option::None;
288        for y in 0..self.spawned_map_meta.grid_size.y {
289            for x in 0..self.spawned_map_meta.grid_size.x {
290                let position = UVec2 { x, y };
291                for layer_index in 0..self.spawned_map_meta.layer_names.len() as u32 {
292                    self.set_tile(layer_index, position, &empty_tile, TileCollisionKind::Empty);
293                }
294            }
295        }
296    }
297    /// Clear all of the elements on the map.
298    pub fn clear_elements(&mut self) {
299        let mut to_kill: Vec<Entity> = Vec::new();
300        self.entities
301            .iter_with(&mut self.element_handles)
302            .for_each(|(entity, _)| {
303                to_kill.push(entity);
304            });
305
306        to_kill.into_iter().for_each(|entity| {
307            self.delete_element(entity);
308        });
309    }
310}

312/// Handles user input comming from the editor and makes the required changes to the map.
313fn handle_editor_input(player_inputs: Res<MatchInputs>, mut map_manager: MapManager) {
314    for player in &player_inputs.players {
315        if let Some(editor_input) = &player.editor_input {
316            match editor_input {
317                EditorInput::SpawnElement {
318                    handle,
319                    translation,
320                    layer,
321                } => {
322                    map_manager.create_element(handle, translation, *layer as u32);
323                }
324                EditorInput::CreateLayer { id } => {
325                    map_manager.create_layer(ustr(id));
326                }
327                EditorInput::DeleteLayer { layer } => {
328                    map_manager.delete_layer(*layer as u32);
329                }
330                EditorInput::RenameLayer {
331                    layer,
332                    name: new_name,
333                } => map_manager.rename_layer(*layer as u32, new_name),
334                EditorInput::MoveEntity { entity, pos } => {
335                    map_manager.move_element(*entity, pos);
336                }
337                EditorInput::DeleteEntity { entity } => {
338                    map_manager.delete_element(*entity);
339                }
340                EditorInput::SetTilemap { layer, handle } => {
341                    map_manager.set_layer_tilemap(*layer as u32, handle);
342                }
343                EditorInput::SetTile {
344                    layer,
345                    pos,
346                    tilemap_tile_idx,
347                    collision,
348                } => {
349                    map_manager.set_tile(*layer as u32, *pos, tilemap_tile_idx, *collision);
350                }
351                EditorInput::MoveLayer { layer, down } => {
352                    map_manager.swap_layer(*layer as u32, *down)
353                }
354                EditorInput::RenameMap { name } => {
355                    map_manager.rename_map(name);
356                }
357                EditorInput::RandomizeTiles {
358                    tile_layers,
359                    element_layers,
360                    tile_size,
361                } => {
362                    let map_constructor = ShiftnanigansMapConstructor::new(
363                        map_manager.get_size(),
364                        *tile_size,
365                        tile_layers,
366                        element_layers,
367                    );
368                    map_constructor.construct_map(&mut map_manager);
369                }
370            }
371        }
372    }
