3#[derive(Clone, Debug, HasSchema)]
4#[schema(no_default)]
5pub struct MapPool {
6    pub maps: Vec<Handle<MapMeta>>,
7    pub current_map: Handle<MapMeta>,
8}
9
10impl MapPool {
11    /// Convert to [`MapPoolNetwork`] which is serializable for replication.
12    pub fn into_network(&self, assets: &AssetServer) -> MapPoolNetwork {
13        MapPoolNetwork {
14            maps: self.maps.iter().map(|h| h.network_handle(assets)).collect(),
15            current_map: self.current_map.network_handle(assets),
16        }
17    }
18
19    /// Convert [`MapPoolNetwork`] into a [`MapPool`]
20    pub fn from_network(map_pool: MapPoolNetwork, assets: &AssetServer) -> MapPool {
21        MapPool {
22            maps: map_pool
23                .maps
24                .iter()
25                .map(|h| h.into_handle(assets))
26                .collect(),
27            current_map: map_pool.current_map.into_handle(assets),
28        }
29    }
30
31    /// Make a `MapPool` consisting of single map.
32    pub fn from_single_map(map: Handle<MapMeta>) -> Self {
33        Self {
34            maps: vec![map],
35            current_map: map,
36        }
37    }
38
39    /// Construct `MapPool` from slice of maps.
40    /// Current map is first in array.
41    pub fn from_slice(maps: &[Handle<MapMeta>]) -> Self {
42        Self {
43            maps: maps.into(),
44            current_map: maps[0],
45        }
46    }
47
48    /// Randomize current map. Updates `curent_map` on self and returns `Handle<MapMeta>`.
49    pub fn randomize_current_map(&mut self, rng: &GlobalRng) -> Handle<MapMeta> {
50        self.current_map = *rng.sample(&self.maps).unwrap();
51        self.current_map
52    }
53
54    /// Return a random map handle from pool
55    pub fn get_random_map(&self, rng: &GlobalRng) -> Handle<MapMeta> {
56        *rng.sample(&self.maps).unwrap()
57    }
58}
