1//! Damage / kill regions.
2//!
3//! Any player that intersects a damage region will be killed.
4
5use crate::prelude::*;
6
7use super::utils::Rect;
8
9/// Install this module.
10pub fn install(session: &mut SessionBuilder) {
11    DamageRegion::register_schema();
12    DamageRegionOwner::register_schema();
13
14    session.add_system_to_stage(CoreStage::PostUpdate, kill_players_in_damage_region);
15}
16
17/// A rectangular damage region.
18///
19/// While this _might_ change in the future, damage regions will kill players immediately, so there
20/// is no "damage" field.
21#[derive(Debug, Clone, Default, HasSchema)]
22#[repr(C)]
23pub struct DamageRegion {
24    /// The size of the damage region in pixels
25    pub size: Vec2,
26}
27
28impl DamageRegion {
29    /// Get the collision rectangle of this damage region, given it's transform.
30    pub fn collider_rect(&self, position: Vec3) -> Rect {
31        Rect::new(position.x, position.y, self.size.x, self.size.y)
32    }
33}
34
35/// A component that may be added to a damage region entity to indicate the triggering entity.
36///
37/// If this entity is a player, it will not be harmed by the damage region.
38// TODO: Make `DamageRegionOwner` a part of the `DamageRegion` component?
39#[derive(Debug, Clone, HasSchema, Default)]
40#[repr(C)]
41pub struct DamageRegionOwner(pub Entity);
42
43/// System that will eliminate players that are intersecting with a damage region.
44fn kill_players_in_damage_region(
45    entities: Res<Entities>,
46    mut commands: Commands,
47    player_indexes: Comp<PlayerIdx>,
48    transforms: Comp<Transform>,
49    damage_regions: Comp<DamageRegion>,
50    damage_region_owners: Comp<DamageRegionOwner>,
51    bodies: Comp<KinematicBody>,
52    invincibles: CompMut<Invincibility>,
53) {
54    let mut bitset = player_indexes.bitset().clone();
55    bitset.bit_and(transforms.bitset());
56    bitset.bit_and(bodies.bitset());
57    bitset.bit_andnot(invincibles.bitset());
58
59    for player_ent in entities.iter_with_bitset(&bitset) {
60        let transform = transforms.get(player_ent).unwrap();
61        let body = bodies.get(player_ent).unwrap();
62
63        let player_rect = body.bounding_box(*transform);
64        for (ent, (damage_region, transform)) in entities.iter_with((&damage_regions, &transforms))
65        {
66            let owner = damage_region_owners.get(ent);
67            // Don't damage the player that owns this damage region
68            if let Some(owner) = owner {
69                if owner.0 == player_ent {
70                    continue;
71                }
72            }
73
74            let damage_rect = damage_region.collider_rect(transform.translation);
75            if player_rect.overlaps(&damage_rect) {
76                commands.add(PlayerCommand::kill(
77                    player_ent,
78                    Some(transform.translation.xy()),
79                ));
80            }
81        }
82    }
83}
