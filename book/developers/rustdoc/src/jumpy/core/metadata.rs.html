14/// Extension trait for the bones [`AssetServer`].
15pub trait MatchAssetServerExt {
16    /// Register the default assets from `bones_framework`.
17    fn register_match_assets(self) -> Self;
18}
19impl MatchAssetServerExt for &mut AssetServer {
20    fn register_match_assets(self) -> Self {
21        CoreMeta::register_schema();
22        PlayerMeta::register_schema();
23        ElementMeta::register_schema();
24        BulletMeta::register_schema();
25        MapMeta::register_schema();
26        HatMeta::register_schema();
27
28        self
29    }
30}
