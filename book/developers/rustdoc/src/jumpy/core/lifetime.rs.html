1//! Entity lifetimes for deleting an entity after a period of time.

3use std::time::Duration;

7pub fn install(session: &mut SessionBuilder) {
8    Lifetime::register_schema();
9
10    session
11        .add_system_to_stage(CoreStage::PostUpdate, lifetime_system)
12        .add_system_to_stage(CoreStage::PostUpdate, invincibility);
13}

15/// The lifetime state of an entity
16///
17/// > **Note:** The age represents how long the entity has had the [`Lifetime`] component on it, not
18/// > necessarily how long since the entity was spawned, if the [`Lifetime`] component was added
19/// > later, after it was spawned.
20/// >
21/// > Also, the age and lifetime are public, subject to other system's modification.
22#[derive(Copy, Clone, Default, HasSchema)]
23#[repr(C)]
24pub struct Lifetime {
25    /// How long the entity should be allowed to live in seconds.
26    pub lifetime: f32,
27    /// How long the entity has lived in seconds.
28    pub age: f32,
29}
30
31impl Lifetime {
32    pub fn new(lifetime: f32) -> Self {
33        Self {
34            lifetime,
35            ..default()
36        }
37    }
38}
39
40/// Despawns entities that have an expired lifetime
41fn lifetime_system(mut entities: ResMutInit<Entities>, mut lifetimes: CompMut<Lifetime>) {
42    let mut to_kill = Vec::new();
43    for (entity, lifetime) in &mut entities.iter_with(&mut lifetimes) {
44        lifetime.age += 1.0 / FPS;
45        if lifetime.age > lifetime.lifetime {
46            to_kill.push(entity);
47        }
48    }
49    for entity in to_kill {
50        entities.kill(entity);
51    }
52}

54/// A timer that can be used to make an entity invincible for a certain amount of time
55///
56/// This is a general purpose invinvibility timer, but will serve as our spawn protection timer.
57#[derive(Clone, Default, HasSchema, Debug)]
58pub struct Invincibility(Timer);
59
60impl Invincibility {
61    pub fn new(duration: Duration) -> Self {
62        Self(Timer::new(duration, TimerMode::Once))
63    }
64}
65
66fn invincibility(
67    time: Res<Time>,
68    mut commands: Commands,
69    entities: ResMutInit<Entities>,
70    mut invincibles: CompMut<Invincibility>,
71) {
72    for (player_ent, invincible) in &mut entities.iter_with(&mut invincibles) {
73        invincible.0.tick(time.delta());
74
75        if invincible.0.finished() {
76            commands.add(move |mut invincibles: CompMut<Invincibility>| {
77                invincibles.remove(player_ent);
78            });
79        }
80    }
81}
