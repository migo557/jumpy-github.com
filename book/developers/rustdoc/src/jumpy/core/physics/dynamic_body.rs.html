1//! DynamicBody component allows simultaing phyiscs of a body.
2
3use std::sync::Mutex;
4
5use crate::prelude::*;
6
7pub type SimulationCommand = dyn FnOnce(&mut rapier::RigidBody) + 'static + Send;
8
9/// Bodies with `DynamicBody` will be simulated dynamically
10/// if `is_dynamic` is True. Currently `DynamicBody` requires component
11/// [`KinematicBody`], as it shares shape.
12///
13/// `if is_dynamic` is false, body is treated as kinematic.
14#[derive(Default, Clone, HasSchema)]
15#[repr(C)]
16pub struct DynamicBody {
17    /// Is body simulating or kinematic mode?
18    pub is_dynamic: bool,
19
20    /// This transform is used to determine if position has moved in gameplay code outside of rapier sim.
21    /// Is only relevant when `is_dynamic` = true. It is updated whenever transform is synced to or from rapier.
22    ///
23    /// If this does not match entity `Transform` at beginning of physics update, rapier simulation body
24    /// will be teleported to match gameplay.
25    ///
26    /// See [`Self::simulation_transform_needs_update`] for context.
27    last_rapier_synced_transform: Transform,
28
29    /// Simulation command queue. See [`Self::push_simulation_command`] for details.
30    #[schema(opaque)]
31    command_queue: Arc<Mutex<Vec<Box<SimulationCommand>>>>,
32}
33
34impl DynamicBody {
35    /// Create new `DynamicBody`. is_dynamic = true simulates physics, false remains kinematic.
36    pub fn new(is_dynamic: bool) -> Self {
37        Self {
38            is_dynamic,
39            ..Default::default()
40        }
41    }
42    /// Check if rapier simulation's body's transform is dirty (If it was modified outside of
43    /// physics simulation). Always returns true if object is not simulating.
44    ///
45    /// Updates `last_rapier_synced_transform` if called, should only be called before updating rapier positions from gameplay.
46    pub fn simulation_transform_needs_update(&mut self, transform: &Transform) -> bool {
47        if !self.is_dynamic {
48            // Body not simulating, we should always update transform in sim
49            return true;
50        }
51
52        // Only use 2D translation + rotation, as this is what rapier sim uses.
53        let last_translation = self.last_rapier_synced_transform.translation.xy();
54        let current_translation = transform.translation.xy();
55        let last_rotation = self
56            .last_rapier_synced_transform
57            .rotation
58            .to_euler(EulerRot::XYZ)
59            .2;
60        let current_rotation = transform.rotation.to_euler(EulerRot::XYZ).2;
61
62        // transform was modified in gameplay outside of rapier sim
63        if last_translation != current_translation || last_rotation != current_rotation {
64            self.last_rapier_synced_transform.translation = transform.translation;
65            self.last_rapier_synced_transform.rotation = transform.rotation;
66
67            return true;
68        }
69
70        // Transform was not modified outside of rapier sim, is not dirty
71        false
72    }
73
74    /// Update `last_rapier_synced_transform` from simulation output so we can determine when gameplay has modified transform
75    /// outside of rapier for [`Self::simulation_transform_needs_update`].
76    ///
77    /// This does NOT update position of entity.
78    pub fn update_last_rapier_synced_transform(&mut self, translation: Vec3, rotation: Quat) {
79        self.last_rapier_synced_transform.translation = translation;
80        self.last_rapier_synced_transform.rotation = rotation;
81    }
82
83    /// Push command that is executed before next physics step, after body is guranteed to be initialized in rapier
84    /// and set to [`rapier::RigidBodyType::Dynamic`] (if transitioning from Kinematic to Dynamic).
85    ///
86    /// Commands are only executed if `is_dynamic` = true (body is simulating) next frame! Otherwise they are discarded.
87    /// This restriction is in place to prevent switching back to kinematic and modifying simulation state in an unintentional way.
88    ///
89    /// Commands are executed in order they are pushed. If you need to immediately apply changes, use [`CollisionWorld::mutate_rigidbody`].
90    pub fn push_simulation_command(&mut self, command: Box<SimulationCommand>) {
91        self.command_queue.lock().unwrap().push(command);
92    }
93
94    /// Retrieve [`SimulationCommand`] queue for processing. Values are moved out of self, flushing queue.
95    pub fn simulation_commands(&mut self) -> Vec<Box<SimulationCommand>> {
96        let mut dummy: Vec<Box<SimulationCommand>> = vec![];
97        let mut queue = self.command_queue.lock().unwrap();
98        std::mem::swap(&mut *queue, &mut dummy);
99        dummy
100    }
101}
