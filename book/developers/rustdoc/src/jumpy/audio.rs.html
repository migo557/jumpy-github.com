7pub fn game_plugin(game: &mut Game) {

9
10    let modified_session = game.sessions.modify_and_replace_existing_session(
11        SessionNames::AUDIO,
12        |session: &mut SessionBuilder| {
13            session.stages().add_system_to_stage(First, music_system);
14        },
15    );
16
17    if modified_session.is_none() {
18        panic!("Audio plugin failed to find existing bones audio session, make sure jumpy audio plugin is installed after bones default plugins.")
19    }
