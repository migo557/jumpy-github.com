8/// The music playback state.
9#[derive(HasSchema, Default, PartialEq, Eq)]
10#[schema(no_clone)]
11pub enum MusicState {
12    /// Music is not playing.
13    #[default]
14    None,
15    /// Playing the main menu music.
16    MainMenu,
17    /// Playing the character select music.
18    CharacterSelect,
19    /// Playing the credits music.
20    Credits,
21    /// Playing the fight music.
22    Fight {
23        /// The index of the song in the shuffled playlist.
24        idx: usize,
25    },
26}
27
28/// Bevy resource containing the in-game music playlist shuffled.
29#[derive(HasSchema, Deref, DerefMut, Clone, Default)]
30#[repr(C)]
31pub struct ShuffledPlaylist(pub SVec<Handle<AudioSource>>);
32
33/// The amount of time to spend fading the music in and out.
34pub const MUSIC_FADE_DURATION: Duration = Duration::from_millis(500);
35
36pub const MUSIC_VOLUME: f64 = 0.1;
37
38/// System that plays music according to the game mode.
39pub(super) fn music_system(
40    meta: Root<GameMeta>,
41    mut audio: ResMut<AudioCenter>,
42    mut shuffled_fight_music: ResMutInit<ShuffledPlaylist>,
43    mut music_state: ResMutInit<MusicState>,
44    ctx: Res<EguiCtx>,
45    sessions: Res<Sessions>,
46) {
47    if shuffled_fight_music.is_empty() {
48        let mut songs = meta.music.fight.clone();
49        THREAD_RNG.with(|rng| rng.shuffle(&mut songs));
50        **shuffled_fight_music = songs;
51    }
52
53    let tween = Tween {
54        start_time: kira::StartTime::Immediate,
55        duration: MUSIC_FADE_DURATION,
56        easing: kira::tween::Easing::Linear,
57    };
58    let play_settings = StaticSoundSettings::default()
59        .volume(MUSIC_VOLUME)
60        .fade_in_tween(tween);
61
62    // If we are in a game
63    if sessions.get(SessionNames::GAME).is_some() {
64        if let MusicState::Fight { idx } = &mut *music_state {
65            if let Some(PlaybackState::Stopped) = audio.music_state() {
66                *idx = (*idx + 1) % shuffled_fight_music.len();
67                audio.play_music_from_settings(shuffled_fight_music[*idx], play_settings, true);
68            }
69        } else if let Some(song) = shuffled_fight_music.get(0) {
70            audio.play_music_from_settings(*song, play_settings, false);
71            *music_state = MusicState::Fight { idx: 0 };
72        }
73
74    // If we are on a menu page
75    } else if sessions.get(SessionNames::MAIN_MENU).is_some() {
76        let menu_page = ctx.get_state::<MenuPage>();
77        match menu_page {
78            MenuPage::PlayerSelect | MenuPage::MapSelect { .. } | MenuPage::NetworkGame => {
79                if *music_state != MusicState::CharacterSelect {
80                    audio.play_music_from_settings(
81                        meta.music.title_screen,
82                        play_settings.loop_region(Region::default()),
83                        false,
84                    );
85                    *music_state = MusicState::CharacterSelect;
86                }
87            }
88            MenuPage::Home | MenuPage::Settings => {
89                if *music_state != MusicState::MainMenu {
90                    audio.play_music_from_settings(meta.music.title_screen, play_settings, false);
91                    *music_state = MusicState::MainMenu;
92                }
93            }
94            MenuPage::Credits => {
95                if *music_state != MusicState::Credits {
96                    audio.play_music_from_settings(meta.music.credits, play_settings, false);
97                    *music_state = MusicState::Credits;
98                }
99            }
100        }
101    }
102}
