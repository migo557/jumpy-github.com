1//! Debug tools and menus.

3use crate::prelude::*;
4use bones_framework::debug::frame_time_diagnostics_plugin;
5
6#[cfg(not(target_arch = "wasm32"))]
7use bones_framework::networking::debug::network_debug_window;
8
9pub fn game_plugin(game: &mut Game) {
10    game.sessions
11        .create_with(SessionNames::DEBUG, |builder: &mut SessionBuilder| {
12            builder
13                .install_plugin(session_plugin)
14                .install_plugin(frame_time_diagnostics_plugin);
15        });
16}
17
18fn session_plugin(session: &mut SessionBuilder) {
19    session
20        .stages
21        .add_system_to_stage(CoreStage::First, debug_menu);
22
23    #[cfg(not(target_arch = "wasm32"))]
24    session
25        .stages
26        .add_system_to_stage(CoreStage::First, network_debug_window);
27}

29#[derive(HasSchema, Clone, Debug, Default)]
30struct DebugMenuState {
31    pub show_menu: bool,
32    pub snapshot: Option<World>,
33}
34
35fn debug_menu(
36    mut sessions: ResMut<Sessions>,
37    keyboard_inputs: Res<KeyboardInputs>,
38    mut state: ResMutInit<DebugMenuState>,
39    ctx: ResMut<EguiCtx>,
40    localization: Localization<GameMeta>,
41) {
42    let DebugMenuState {
43        snapshot,
44        show_menu,
45    } = &mut *state;
46
47    let toggle_debug = keyboard_inputs
48        .key_events
49        .iter()
50        .any(|x| x.key_code.option() == Some(KeyCode::F12) && !x.button_state.pressed());
51
52    if toggle_debug {
53        *show_menu = !*show_menu;
54    }
55    let game_session = sessions.get_mut(SessionNames::GAME);
56
57    // Delete the snapshot if there is one and we are not in the middle of a game.
58    if game_session.is_none() && snapshot.is_some() {
59        *snapshot = None;
60    }
61
62    egui::Window::new(localization.get("debug-tools"))
63        .id(egui::Id::from("debug"))
64        .open(show_menu)
65        .show(&ctx, |ui| {
66            ui.horizontal(|ui| {
67                ui.set_enabled(game_session.is_some());
68
69                if ui.button(localization.get("take-snapshot")).clicked() {
70                    if let Some(session) = game_session {
71                        *snapshot = Some(session.snapshot());
72                    }
73                } else if ui.button(localization.get("restore-snapshot")).clicked() {
74                    if let Some(session) = game_session {
75                        if let Some(mut snapshot) = snapshot.clone() {
76                            session.restore(&mut snapshot);
77                        }
78                    }
79                }
80
81                let show_frame_time_window = &mut ctx
82                    .get_state::<bones_framework::debug::FrameTimeWindowState>()
83                    .open;
84                if ui
85                    .button(localization.get("frame-time-diagnostics"))
86                    .clicked()
87                {
88                    *show_frame_time_window = !*show_frame_time_window;
89                    ctx.set_state(bones_framework::debug::FrameTimeWindowState {
90                        open: *show_frame_time_window,
91                    });
92                }
93
94                // Show net diagnostics button
95                #[cfg(not(target_arch = "wasm32"))]
96                {
97                    let show_network_debug = &mut ctx.get_state::<NetworkDebugMenuState>().open;
98                    if ui.button(localization.get("network-debug")).clicked() {
99                        *show_network_debug = !*show_network_debug;
100                        // Set state so other debug menus may access
101                        ctx.set_state(NetworkDebugMenuState {
102                            open: *show_network_debug,
103                        });
104                    }
105                }
106
107                // Network debug disabled in wasm.
108                #[cfg(target_arch = "wasm32")]
109                ui.add_enabled_ui(false, |ui| {
110                    let _ = ui.button(localization.get("network-debug"));
111                });
112            })
113        });
114}
