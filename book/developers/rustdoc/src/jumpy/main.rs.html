100impl GameMeta {
101    /// Get the lua plugins loaded by the game.
102    pub fn get_plugins(&self, asset_server: &AssetServer) -> Arc<Vec<Handle<LuaPlugin>>> {
103        let mut plugins = Vec::new();
104        plugins.extend(self.plugins.iter().copied());
105        plugins.extend(
106            self.core
107                .map_elements
108                .iter()
109                .map(|eh| asset_server.get(*eh).plugin)
110                .filter(|plugin_handle| plugin_handle != &Handle::default()),
111        );
112
113        for pack in asset_server.packs() {
114            let pack_meta = asset_server.get(pack.root.typed::<PackMeta>());
115            plugins.extend(pack_meta.plugins.iter().copied());
116            plugins.extend(
117                pack_meta
118                    .map_elements
119                    .iter()
120                    .map(|eh| asset_server.get(*eh).plugin)
121                    .filter(|plugin_handle| plugin_handle != &Handle::default()),
122            );
123        }
124        Arc::new(plugins)
125    }
126}

138fn main() {
139    // Init logging
140    setup_logs!("org", "fishfolk", "jumpy");
141
142    // Initialize the Bevy task pool manually so that we can use it during startup.
143    bevy_tasks::IoTaskPool::init(bevy_tasks::TaskPool::new);
144
145    // Register types that we will load from persistent storage.
146    settings::Settings::register_schema();
147
148    // First create bones game.
149    let mut game = Game::new();
150
151    // Register our game and pack meta types
152    GameMeta::register_schema();
153    PackMeta::register_schema();
154
155    game
156        // Install game plugins
157        .install_plugin(DefaultGamePlugin)
158        .install_plugin(audio::game_plugin)
159        .install_plugin(settings::game_plugin)
160        .install_plugin(fullscreen::game_plugin)
161        .install_plugin(input::game_plugin)
162        .install_plugin(core::game_plugin)
163        .install_plugin(debug::game_plugin)
164        .install_plugin(profiler::game_plugin)
165        .install_plugin(ui::scoring::game_plugin)
166        // We initialize the asset server and register asset types
167        .init_shared_resource::<AssetServer>()
168        .register_default_assets();
169
170    // Create a new session for the game menu. Each session is it's own bones world with it's own
171    // plugins, systems, and entities.
172    game.sessions.start_menu();
173
174    // Create a new session for the pause menu, which sits in the background by default and only
175    // does anything while the game is running.
176    game.sessions
177        .create_with(SessionNames::PAUSE_MENU, |builder: &mut SessionBuilder| {
178            builder.install_plugin(ui::pause_menu::session_plugin);
179        });
180
181    // Set priority to ensure pause menu comes before scoring menu (drawn on top)
182    game.sessions
183        .get_mut(SessionNames::PAUSE_MENU)
184        .unwrap()
185        .priority = 1;
186
187    // Scoring menu plugin, activated by game between round tarnsitions when appropriate
188    game.sessions
189        .create_with(SessionNames::SCORING, |builder: &mut SessionBuilder| {
190            builder.install_plugin(ui::scoring::session_plugin);
191        });
192
193    // session for pop-ups / nofication UI
194    game.sessions.create_with(
195        SessionNames::NOTIFICATION,
196        |builder: &mut SessionBuilder| {
197            builder.install_plugin(ui::notification::session_plugin);
198        },
199    );
200
201    // Create a bevy renderer for the bones game and run it.
202    BonesBevyRenderer {
203        game,
204        pixel_art: true,
205        game_version: Version::new(
206            env!("CARGO_PKG_VERSION_MAJOR").parse().unwrap(),
207            env!("CARGO_PKG_VERSION_MINOR").parse().unwrap(),
208            env!("CARGO_PKG_VERSION_PATCH").parse().unwrap(),
209        ),
210        app_namespace: ("org".into(), "fishfolk".into(), "jumpy".into()),
211        asset_dir: std::env::var("JUMPY_ASSETS")
212            .unwrap_or_else(|_| "assets".into())
213            .into(),
214        packs_dir: std::env::var("JUMPY_ASSET_PACKS")
215            .unwrap_or_else(|_| "packs".into())
216            .into(),
217        custom_load_progress: Some(Box::new(load_progress)),
218        preload: true,
219    }
220    .app()
221    .run();
222}

224fn load_progress(assets: &AssetServer, ctx: &egui::Context) {
225    let errored = assets.load_progress.errored();
226    egui::CentralPanel::default()
227        .frame(egui::Frame::default().fill(egui::Color32::from_rgb(0x26, 0x2b, 0x44)))
228        .show(ctx, |ui| {
229            let height = ui.available_height();
230            let ctx = ui.ctx().clone();
231
232            let space_size = 0.03;
233            let spinner_size = 0.10;
234            let text_size = 0.034;
235            ui.vertical_centered(|ui| {
236                ui.add_space(height * 0.3);
237
238                if errored > 0 {
239                    let err_color = egui::Color32::RED;
240                    ui.label(
241                        egui::RichText::new("⚠")
242                            .color(err_color)
243                            .size(height * spinner_size),
244                    );
245                    ui.add_space(height * space_size);
246                    ui.label(
247                        egui::RichText::new(format!(
248                            "Error loading {errored} asset{}.",
249                            if errored > 1 { "s" } else { "" }
250                        ))
251                        .color(err_color)
252                        .size(height * text_size * 0.75),
253                    );
254                } else {
255                    let rect = ui
256                        .label(
257                            egui::RichText::new("⚓")
258                                .color(egui::Color32::WHITE)
259                                .size(height * spinner_size),
260                        )
261                        .rect;
262                    egui::Spinner::new().paint_at(ui, rect.expand(spinner_size * height * 0.2));
263                    ui.add_space(height * space_size);
264                    ui.label(
265                        egui::RichText::new("Loading")
266                            .color(egui::Color32::WHITE)
267                            .size(height * text_size),
268                    );
269                }
270            });
271
272            ctx.data_mut(|d| {
273                d.insert_temp(ui.id(), (spinner_size, space_size, text_size));
274            })
275        });
