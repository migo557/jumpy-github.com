74impl SessionPlugin for MatchPlugin {
75    fn install(self, session: &mut SessionBuilder) {
76        session
77            .install_plugin(DefaultSessionPlugin)
78            .install_plugin(LuaPluginLoaderSessionPlugin(self.plugins));
79
80        physics::install(session);
81        input::install(session);
82        map::install(session);
83        player::plugin(session);
84        elements::session_plugin(session);
85        damage::install(session);
86        camera::install(session);
87        lifetime::install(session);
88        random::plugin(session);
89        debug::plugin(session);
90        item::install(session);
91        attachment::install(session);
92        bullet::session_plugin(session);
93        editor::install(session);
94        scoring::session_plugin(session);
95
96        let current_map = self.maps.current_map;
97        session.insert_resource(self.maps);
98
99        // Initialize LoadedMap on startup as we cannot access AssetServer during MatchPlugin install
100        // to get map meta.
101        session.add_startup_system(
102            move |mut loaded_map: ResMutInit<LoadedMap>, assets: Res<AssetServer>| {
103                let map_meta = assets.get(current_map).clone();
104                *loaded_map = LoadedMap(Arc::new(map_meta))
105            },
106        );
107
108        session.insert_resource(MatchInputs {
109            players: self.player_info,
110        });
111        session.insert_resource(self.score);
112        session.set_session_runner(self.session_runner);
113    }
114}

116#[derive(Default)]
117pub struct JumpyDefaultMatchRunner {
118    /// The jumpy match runner has it's own input collector instead of using the global one, because
119    /// it needs different `just_[input]` behavior due to it running on a fixed update.
120    pub input_collector: PlayerInputCollector,
121    pub accumulator: f64,
122    pub last_run: Option<Instant>,
123    /// Disables local input for session.
124    disable_local_input: bool,
125}
126
127impl SessionRunner for JumpyDefaultMatchRunner {
128    fn step(&mut self, frame_start: Instant, world: &mut World, stages: &mut SystemStages) {
129        pub const STEP: f64 = 1.0 / FPS as f64;
130        let last_run = self.last_run.unwrap_or(frame_start);
131        let delta = (frame_start - last_run).as_secs_f64();
132
133        self.input_collector.apply_inputs(world);
134
135        let mut run = || {
136            // Advance the world time
137            world
138                .resource_mut::<Time>()
139                .advance_exact(Duration::from_secs_f64(STEP));
140
141            self.input_collector.update_just_pressed();
142            let input = self.input_collector.get_current_controls();
143            {
144                let mut player_inputs = world.resource_mut::<MatchInputs>();
145                (0..MAX_PLAYERS as usize).for_each(|i| {
146                    let player_input = &mut player_inputs.players[i];
147                    let Some(source) = &player_input.control_source else {
148                        return;
149                    };
150                    if !self.disable_local_input {
151                        player_input.control = *input.get(source).unwrap();
152                    } else {
153                        // This runner is only used for offline play, use default control for all to
154                        // disable input by using default (no input).
155                        player_input.control = PlayerControl::default();
156                    }
157                });
158            }
159
160            // Mark inputs as consumed for this frame
161            self.input_collector.advance_frame();
162
163            // Advance the simulation
164            stages.run(world);
165        };
166
167        self.accumulator += delta;
168
169        let loop_start = Instant::now();
170        loop {
171            if self.accumulator >= STEP {
172                let loop_too_long = (Instant::now() - loop_start).as_secs_f64() > STEP;
173
174                if loop_too_long {
175                    warn!("Frame took too long: couldn't keep up with fixed update.");
176                    self.accumulator = 0.0;
177                    break;
178                } else {
179                    self.accumulator -= STEP;
180                    run()
181                }
182            } else {
183                break;
184            }
185        }
186
187        self.last_run = Some(frame_start);
188    }
189
190    fn restart_session(&mut self) {
191        *self = JumpyDefaultMatchRunner::default();
192    }
193
194    fn disable_local_input(&mut self, disable_input: bool) {
195        self.disable_local_input = disable_input;
196    }
197}
