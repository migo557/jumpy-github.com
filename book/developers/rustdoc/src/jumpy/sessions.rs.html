15pub trait SessionExt {
16    fn start_menu(&mut self);
17    fn end_game(&mut self);
18    /// Optionally provide map_pool to restart game with. If not provided,
19    /// current game's [`MapPool`] will be used.
20    fn restart_game(&mut self, map_pool: Option<MapPool>, reset_score: bool);
21    fn start_game(&mut self, match_plugin: crate::core::MatchPlugin);
22}
23
24impl SessionExt for Sessions {
25    fn start_menu(&mut self) {
26        self.create_with(SessionNames::MAIN_MENU, |builder: &mut SessionBuilder| {
27            builder.install_plugin(crate::ui::main_menu::session_plugin);
28        });
29    }
30
31    fn end_game(&mut self) {
32        self.delete(SessionNames::GAME);
33
34        // Make sure that scoring menu closed if was open
35        if let Some(scoring_session) = self.get_mut(SessionNames::SCORING) {
36            if let Some(mut state) = scoring_session.world.get_resource_mut::<ScoringMenuState>() {
37                state.active = false;
38            }
39        }
40    }
41
42    #[track_caller]
43    fn restart_game(&mut self, map_pool: Option<MapPool>, reset_score: bool) {
44        if let Some((existing_map_pool, player_info, plugins, mut session_runner, score)) =
45            self.get_mut(SessionNames::GAME).map(|session| {
46                let existing_map_pool = (*session.world.resource::<MapPool>()).clone();
47                let match_inputs = session.world.resource::<MatchInputs>();
48                let score = (*session.world.resource::<MatchScore>()).clone();
49
50                // Take ownership of session runner (we want to preserve socket and such for network runner)
51                // by swapping a dummy one with session.
52                let mut session_runner: Box<dyn SessionRunner> =
53                    Box::<JumpyDefaultMatchRunner>::default();
54                std::mem::swap(&mut session.runner, &mut session_runner);
55
56                (
57                    existing_map_pool,
58                    match_inputs.players.clone(),
59                    session.world.resource::<LuaPlugins>().0.clone(),
60                    session_runner,
61                    score,
62                )
63            })
64        {
65            self.end_game();
66
67            // Reset session runner
68            session_runner.restart_session();
69
70            let map_pool = map_pool.unwrap_or(existing_map_pool);
71            let score = if reset_score {
72                MatchScore::default()
73            } else {
74                score
75            };
76
77            self.create_with(SessionNames::GAME, |builder: &mut SessionBuilder| {
78                builder.install_plugin(crate::core::MatchPlugin {
79                    maps: map_pool,
80                    player_info,
81                    plugins,
82                    session_runner,
83                    score,
84                });
85            });
86        } else {
87            panic!("Cannot restart game when game is not running");
88        }
89    }
90
91    fn start_game(&mut self, match_plugin: crate::core::MatchPlugin) {
92        self.create_with(SessionNames::GAME, |builder: &mut SessionBuilder| {
93            builder.install_plugin(match_plugin);
94        });
95    }
96}
